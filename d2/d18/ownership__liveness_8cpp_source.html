21void addUse(ValueSet &values, const std::shared_ptr<Value> &value,
22 const BorrowProvenance &provenance) {
23 if (tracksOwnership(value)) {
24 values.insert(value.get());

30void addInstructionUses(ValueSet &values, const Instruction &instruction,
31 const BorrowProvenance &provenance) {
32 switch (instruction.getOpCode()) {
33 case OpCode::Alloca:
34 case OpCode::Alloc:
35 case OpCode::Br:
36 case OpCode::Phi:
37 return;
38 case OpCode::Load:
39 addUse(values, static_cast<const LoadInst &>(instruction).getSource(),
40 provenance);
41 return;
42 case OpCode::Store: {
43 const auto &store = static_cast<const StoreInst &>(instruction);
44 addUse(values, store.getSource(), provenance);
45 addUse(values, store.getDestination(), provenance);
46 return;
47 }
48 case OpCode::Add:
49 case OpCode::Sub:
50 case OpCode::Mul:
51 case OpCode::SDiv:
52 case OpCode::UDiv:
53 case OpCode::SRem:
54 case OpCode::URem:
55 case OpCode::Shl:
56 case OpCode::LShr:
57 case OpCode::AShr:
58 case OpCode::BitAnd:
59 case OpCode::BitOr:
60 case OpCode::BitXor: {
61 const auto &binary = static_cast<const BinaryInst &>(instruction);
62 addUse(values, binary.getLhs(), provenance);
63 addUse(values, binary.getRhs(), provenance);
64 return;
65 }
66 case OpCode::Cmp: {
67 const auto &comparison = static_cast<const CmpInst &>(instruction);
68 addUse(values, comparison.getLhs(), provenance);
69 addUse(values, comparison.getRhs(), provenance);
70 return;
71 }
72 case OpCode::CondBr:
73 addUse(values,
74 static_cast<const CondBranchInst &>(instruction).getCondition(),
75 provenance);
76 return;
77 case OpCode::Ret:
78 addUse(values, static_cast<const ReturnInst &>(instruction).getValue(),
79 provenance);
80 return;
81 case OpCode::Call: {
82 const auto &call = static_cast<const CallInst &>(instruction);
83 addUse(values, call.getCalleeValue(), provenance);
84 for (const auto &argument : call.getArguments()) {
85 addUse(values, argument, provenance);
86 }
87 addUse(values, call.getVariadicPack(), provenance);
88 return;
89 }
90 case OpCode::Copy:
91 addUse(values, static_cast<const CopyInst &>(instruction).getSource(),
92 provenance);
93 return;
94 case OpCode::Move:
95 addUse(values, static_cast<const MoveInst &>(instruction).getSource(),
96 provenance);
97 return;
98 case OpCode::Borrow:
99 addUse(values, static_cast<const BorrowInst &>(instruction).getOwner(),
100 provenance);
101 return;
102 case OpCode::Destroy:
103 addUse(values, static_cast<const DestroyInst &>(instruction).getValue(),
104 provenance);
105 return;
106 case OpCode::GetElementPtr: {
107 const auto &gep = static_cast<const GetElementPtrInst &>(instruction);
108 addUse(values, gep.getPointer(), provenance);
109 addUse(values, gep.getIndexValue(), provenance);
110 return;
111 }
112 case OpCode::Cast:
113 addUse(values, static_cast<const CastInst &>(instruction).getSource(),
114 provenance);
115 return;
116 case OpCode::WeakLock:
117 addUse(values,
118 static_cast<const WeakLockInst &>(instruction).getWeakValue(),
119 provenance);
120 return;
121 case OpCode::WeakAlive:
122 addUse(values,
123 static_cast<const WeakAliveInst &>(instruction).getWeakValue(),
124 provenance);
125 return;
126 case OpCode::InlineAsm: {
127 const auto &inlineAsm = static_cast<const InlineAsmInst &>(instruction);
128 for (const auto &operand : inlineAsm.getOutputs()) {
129 addUse(values, operand.value, provenance);
130 }
131 for (const auto &operand : inlineAsm.getInputs()) {
132 addUse(values, operand.value, provenance);
133 }
134 return;
135 }

289OwnershipLiveness analyzeOwnershipLiveness(const Module &module,
290 const Function &function,
291 const ControlFlowGraph &cfg) {
292 OwnershipLiveness result;
293 result.borrowProvenance_ = analyzeBorrowProvenance(module, function, cfg);
294
295 bool changed = true;
296 while (changed) {
297 changed = false;
298 for (auto blockIt = function.getBlocks().rbegin();
299 blockIt != function.getBlocks().rend(); ++blockIt) {
300 if (!*blockIt) {
301 continue;
302 }
303 const auto &block = **blockIt;
304 ValueSet liveOut;
305 for (const auto *successor : cfg.successors().at(&block)) {
306 ValueSet edgeState = result.entryStates_[successor];
307 for (const auto &instruction : successor->getInstructions()) {
308 if (!instruction || instruction->getOpCode() != OpCode::Phi) {
309 continue;
310 }
311 const auto &phi = static_cast<const PhiInst &>(*instruction);
312 for (const auto &[label, value] : phi.getIncoming()) {
313 if (label == block.label) {
314 replacePhiResultWithIncoming(edgeState, phi.getResult(), value,
315 result.borrowProvenance_);
316 break;
317 }
318 }
319 }
320 for (const auto &instruction : successor->getInstructions()) {
321 if (!instruction || instruction->getOpCode() != OpCode::Load) {
322 continue;
323 }
324 const auto &load = static_cast<const LoadInst &>(*instruction);
325 if (result.borrowProvenance_.isEntryLoad(*successor,
326 load.getResult())) {
327 replaceLoadResultWithStorage(edgeState, load, block, *successor,
328 result.borrowProvenance_);
329 }
330 }
331 if (result.edgeStates_[&block][successor] != edgeState) {
332 result.edgeStates_[&block][successor] = edgeState;
333 changed = true;
334 }
335 unionInto(liveOut, edgeState);
336 }
337
338 ValueSet live = std::move(liveOut);
339 OwnershipLiveness::InstructionStates afterStates;
340 for (size_t i = block.getInstructions().size(); i-- > 0;) {
341 const auto &instruction = block.getInstructions()[i];
342 if (!instruction) {
343 continue;
344 }
345 afterStates.emplace(i, live);
346 if (const auto value = instructionResult(*instruction);
347 tracksOwnership(value)) {
348 live.erase(value.get());
349 }
350 addInstructionUses(live, *instruction, result.borrowProvenance_);
351 }
352 if (result.entryStates_[&block] != live) {
353 result.entryStates_[&block] = std::move(live);
354 changed = true;
355 }
356 result.afterStates_[&block] = std::move(afterStates);
357 }
358 }
359 return result;
360}
