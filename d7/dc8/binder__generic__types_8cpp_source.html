674 if (failureReason) {
675 *failureReason = "conflicting generic type inference";
676 }
677 return {};
678 }
679 }

682 if (bindings.find(name) == bindings.end()) {
683 bool filledFromDefault = false;
684 if (decl) {
685 auto paramIndex = static_cast<size_t>(std::distance(
686 function.genericParameterNames.begin(),
687 std::find(function.genericParameterNames.begin(),
688 function.genericParameterNames.end(), name)));
689 if (paramIndex < decl->genericParams_.size() &&
690 decl->genericParams_[paramIndex] &&
691 decl->genericParams_[paramIndex]->defaultType) {
692 activeGenericBindingsStack_.push_back(bindings);
693 auto mapped = mapType(*decl->genericParams_[paramIndex]->defaultType);
694 activeGenericBindingsStack_.pop_back();
695 if (mapped) {
696 bindings[name] = mapped;
697 filledFromDefault = true;
698 }
699 }
700 }
701 if (!filledFromDefault) {
702 if (failureReason) {
703 *failureReason = "cannot infer generic type parameter '" + name + "'";
704 }
705 return {};
706 }
707 }
708 }
709
710 if (decl) {
711 if (!validateGenericConstraints(decl->genericConstraints_, bindings,
712 failureReason)) {
713 return {};
714 }
715 }
716
717 return bindings;
718}
