33void Binder::validateAndApplyTypeAttributes(
34 const TopLevel &node, const std::shared_ptr<TypeSymbol> &symbol,
35 bool allowErrorAttribute) {
36 if (!symbol) {
37 return;
38 }
39
40 warnUnknownAttributes(node);
41
42 bool seenError = false;
43 bool seenRepr = false;
44 bool seenPacked = false;
45
46 for (const auto &attr : node.attributes_) {
47 if (attr.name == "error") {
48 if (!allowErrorAttribute) {
49 error(attr.span, "attribute 'error' cannot be applied to this type");
50 continue;
51 }
52 if (seenError) {
53 error(attr.span, "duplicate attribute 'error'");
54 continue;
55 }
56 if (attr.hasArguments()) {
57 error(attr.span, "attribute 'error' does not accept arguments");
58 continue;
59 }
60 seenError = true;
61 symbol->isErrorType = true;
62 continue;
63 }
64
65 if (attr.name == "repr") {
66 if (seenRepr) {
67 error(attr.span, "duplicate attribute 'repr'");
68 continue;
69 }
70 seenRepr = true;
71
72 if (attr.arguments.size() != 1 ||
73 attr.arguments[0].kind != AttributeArgumentKind::Positional) {
74 error(
75 attr.span,
76 "attribute 'repr' expects exactly one positional string argument");
77 continue;
78 }
79
80 auto *str = dynamic_cast<ConstString *>(attr.arguments[0].value.get());
81 if (!str) {
82 error(attr.span, "attribute 'repr' expects a string literal argument");
83 continue;
84 }
85
86 if (str->value_ != "C") {
87 error(attr.span,
88 "invalid argument for attribute 'repr': expected \"C\"");
89 continue;
90 }
91
92 symbol->hasReprC = true;
93 symbol->reprValue = str->value_;
94 continue;
95 }
96
97 if (attr.name == "packed") {
98 if (seenPacked) {
99 error(attr.span, "duplicate attribute 'packed'");
100 continue;
101 }
102 if (attr.hasArguments()) {
103 error(attr.span, "attribute 'packed' does not accept arguments");
104 continue;
105 }
106 auto recordType =
107 std::dynamic_pointer_cast<zir::RecordType>(symbol->type);
108 if (!recordType || symbol->type->getKind() != zir::TypeKind::Record) {
109 error(attr.span,
110 "attribute 'packed' can only be applied to record types");
111 continue;
112 }
113 seenPacked = true;
114 symbol->isPacked = true;
115 recordType->isPacked = true;
116 continue;
117 }
118
119 if (attr.name == "extern" || attr.name == "noMangle") {
120 error(attr.span, "attribute '" + attr.name +
121 "' cannot be applied to type declarations");
122 continue;
123 }
124 }

195std::shared_ptr<zir::Type> Binder::mapType(const TypeNode &typeNode) {
196 // Cache simple types when not in a generic instantiation context.
197 // Skip cache for array types with size expressions (they evaluate AST nodes)
198 // and for failable types (they may emit errors on first call).
199 bool canCache = activeGenericBindingsStack_.empty() && !typeNode.isVarArgs &&
200 !typeNode.isFailable &&
201 !(typeNode.isArray && typeNode.arraySize);
202 if (canCache) {
203 auto it = mapTypeCache_.find(&typeNode);
204 if (it != mapTypeCache_.end())
205 return it->second;
206 }
207
208 auto doMap = [&]() -> std::shared_ptr<zir::Type> {
209 if (typeNode.isVarArgs) {
210 if (!typeNode.baseType)
211 return nullptr;
212 return mapType(*typeNode.baseType);
213 }
214
215 if (typeNode.isFailable) {
216 if (!typeNode.baseType || !typeNode.errorType) {
217 error(typeNode.span, "Invalid failable type declaration.");
218 return nullptr;
219 }
220
221 auto valueType = mapType(*typeNode.baseType);
222 auto errorType = mapType(*typeNode.errorType);
223
224 if (!valueType || !errorType) {
225 return nullptr;
226 }
227
228 if (!typeNode.errorType->qualifiers.empty() ||
229 !typeNode.errorType->typeName.empty()) {
230 std::vector<std::string> errParts = typeNode.errorType->qualifiers;
231 errParts.push_back(typeNode.errorType->typeName);
232 auto errSymbol = resolveQualifiedSymbol(
233 errParts, typeNode.errorType->span, SymbolKind::Type);
234 if (errSymbol && errSymbol->getKind() == SymbolKind::Type) {
235 auto errTypeSymbol = std::static_pointer_cast<TypeSymbol>(errSymbol);
236 if (!errTypeSymbol->isErrorType) {
237 error(typeNode.errorType->span,
238 "Type '" + typeNode.errorType->qualifiedName() +
239 "' used as failable error type must be annotated with "
240 "@error.");
241 }
242 }
243 }
244
245 return makeFailableType(valueType, errorType);
246 }
247
248 if (!activeGenericBindingsStack_.empty() && typeNode.qualifiers.empty() &&
249 typeNode.genericArgs.empty()) {
250 const auto &bindings = activeGenericBindingsStack_.back();
251 auto it = bindings.find(typeNode.typeName);
252 if (it != bindings.end()) {
253 auto mapped = it->second;
254 if (typeNode.isWeak) {
255 if (!mapped || mapped->getKind() != zir::TypeKind::Class) {
256 error(typeNode.span, "'weak' can only be used with class types.");
257 return nullptr;
258 }
259 auto weakType = std::make_shared<zir::ClassType>(
260 *std::static_pointer_cast<zir::ClassType>(mapped));
261 weakType->setWeak(true);
262 return weakType;
263 }
264 return mapped;
265 }
266 }
267
268 if (typeNode.isArray) {
269 if (!typeNode.baseType)
270 return nullptr;
271 auto base = mapType(*typeNode.baseType);
272
273 if (!typeNode.arraySize) {
274 return makeVariadicViewType(base);
275 }
276
277 size_t size = 0;
278 typeNode.arraySize->accept(*this);
279 if (!expressionStack_.empty()) {
280 auto boundSize = std::move(expressionStack_.top());
281 expressionStack_.pop();
282 auto evaluated = evaluateConstantInt(boundSize.get());
283 if (evaluated) {
284 size = static_cast<size_t>(*evaluated);
285 } else {
286 error(typeNode.span,
287 "Array size must be a constant integer expression.");
288 }
289 }
290
291 return std::make_shared<zir::ArrayType>(std::move(base), size);
292 }
293
294 if (typeNode.isPointer) {
295 if (!typeNode.baseType)
296 return nullptr;
297 auto base = mapType(*typeNode.baseType);
298 return std::make_shared<zir::PointerType>(std::move(base));
299 }
300
301 if (typeNode.isFunPtr) {
302 std::vector<std::shared_ptr<zir::Type>> params;
303 for (const auto &p : typeNode.funPtrParams) {
304 auto mapped = mapType(*p);
305 if (!mapped)
306 return nullptr;
307 params.push_back(std::move(mapped));
308 }
309 auto ret =
310 typeNode.funPtrReturn
311 ? mapType(*typeNode.funPtrReturn)
312 : std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
313 if (!ret)
314 return nullptr;
315 std::vector<zir::ParameterOwnership> ownership;
316 std::vector<zir::ParameterEscape> escape;
317 ownership.reserve(params.size());
318 escape.reserve(params.size());
319 for (size_t i = 0; i < params.size(); ++i) {
320 if (!zir::containsManagedValues(params[i])) {
321 ownership.push_back(zir::ParameterOwnership::Borrow);
322 } else if (i < typeNode.funPtrParamSinks.size() &&
323 typeNode.funPtrParamSinks[i]) {
324 ownership.push_back(zir::ParameterOwnership::Sink);
325 } else {
326 ownership.push_back(zir::ParameterOwnership::Transfer);
327 }
328 const bool noescape = i < typeNode.funPtrParamNoEscapes.size() &&
329 typeNode.funPtrParamNoEscapes[i];
330 if (noescape && ownership.back() != zir::ParameterOwnership::Borrow) {
331 error(typeNode.span,
332 "A transferring function pointer parameter cannot have a "
333 "'noescape' contract.");
334 }
335 escape.push_back(noescape ? zir::ParameterEscape::NoEscape
336 : zir::ParameterEscape::Unspecified);
337 }
338 zir::ResultBorrowContract resultBorrow;
339 if (typeNode.funPtrResultBorrowSource) {
340 const auto &source = *typeNode.funPtrResultBorrowSource;
341 const bool numeric =
342 !source.empty() &&
343 std::all_of(source.begin(), source.end(),
344 [](unsigned char c) { return std::isdigit(c) != 0; });
345 size_t sourceIndex = params.size();
346 if (numeric) {
347 try {
348 sourceIndex = static_cast<size_t>(std::stoull(source));
349 } catch (const std::exception &) {
350 sourceIndex = params.size();
351 }
352 }
353 if (!numeric || sourceIndex >= params.size()) {
354 error(typeNode.span,
355 "Function pointer 'borrows' source must be a valid parameter "
356 "index.");
357 } else if (typeNode.funPtrReturnsRef ||
358 ret->getIntrinsicKind() !=
359 zir::IntrinsicTypeKind::StringView) {
360 error(typeNode.span,
361 "Function pointer 'borrows' requires a StringView result.");
362 } else if (params[sourceIndex]->getIntrinsicKind() !=
363 zir::IntrinsicTypeKind::StringView) {
364 error(typeNode.span,
365 "Function pointer 'borrows' currently requires a StringView "
366 "source parameter.");
367 } else if (escape[sourceIndex] == zir::ParameterEscape::NoEscape) {
368 error(typeNode.span,
369 "A noescape function pointer parameter cannot back the "
370 "result.");
371 } else {
372 resultBorrow =
373 zir::ResultBorrowContract::fromParameter(sourceIndex);
374 }
375 }
376 return std::make_shared<zir::FunctionPointerType>(
377 std::move(params), std::move(ret), std::move(ownership),
378 std::move(escape), resultBorrow, typeNode.funPtrReturnsRef);
379 }
380
381 std::vector<std::string> parts = typeNode.qualifiers;
382 parts.push_back(typeNode.typeName);
383 auto symbol =
384 resolveQualifiedSymbol(parts, typeNode.span, SymbolKind::Type);
385 if (symbol && symbol->getKind() == SymbolKind::Type) {
386 auto typeSymbol = std::static_pointer_cast<TypeSymbol>(symbol);
387 if (!typeSymbol->genericParameterNames.empty()) {
388 typeSymbol = instantiateGenericTypeSymbol(typeSymbol, typeNode);
389 if (!typeSymbol) {
390 return nullptr;
391 }
392 } else if (!typeNode.genericArgs.empty()) {
393 error(typeNode.span,
394 "Type '" + typeNode.qualifiedName() + "' is not generic.");
395 return nullptr;
396 }
397 if (typeSymbol->isUnsafe) {
398 requireUnsafeContext(typeNode.span, "unsafe struct types");
399 }
400 if (typeNode.isWeak) {
401 if (!typeSymbol->isClass ||
402 typeSymbol->type->getKind() != zir::TypeKind::Class) {
403 error(typeNode.span, "'weak' can only be used with class types.");
404 return nullptr;
405 }
406 auto weakType = std::make_shared<zir::ClassType>(
407 *std::static_pointer_cast<zir::ClassType>(typeSymbol->type));
408 weakType->setWeak(true);
409 return weakType;
410 }
411
412 auto resolvedType = typeSymbol->type;
413 if (typeNode.errorType && !typeSymbol->isErrorType) {
414 error(typeNode.span,
415 "Type '" + typeNode.qualifiedName() +
416 "' used as failable error type must be annotated with "
417 "@error.");
418 }
419 return resolvedType;
420 }
421
422 return nullptr;
423 }; // end doMap lambda
424
425 auto result = doMap();
426 if (canCache)
427 mapTypeCache_[&typeNode] = result;
428 return result;
429}
