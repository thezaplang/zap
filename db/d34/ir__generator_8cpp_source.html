11std::string renderTypeForUser(const std::shared_ptr<Type> &type) {
12 if (!type) {

122std::shared_ptr<Value> BoundIRGenerator::lowerConstantExpression(
123 const sema::BoundExpression &expression,
124 std::set<const sema::VariableSymbol *> &resolvingConstants) {
125 if (auto unary =
126 dynamic_cast<const sema::BoundUnaryExpression *>(&expression)) {
127 if (unary->op != "&") {
128 return nullptr;
129 }
130
131 if (auto variable = dynamic_cast<const sema::BoundVariableExpression *>(
132 unary->expr.get())) {
133 return std::make_shared<GlobalAddress>(variable->symbol->linkName,
134 unary->type);
135 }
136
137 if (auto index =
138 dynamic_cast<const sema::BoundIndexAccess *>(unary->expr.get())) {
139 auto variable = dynamic_cast<const sema::BoundVariableExpression *>(
140 index->left.get());
141 auto indexValue =
142 lowerConstantExpression(*index->index, resolvingConstants);
143 auto constantIndex = std::dynamic_pointer_cast<Constant>(indexValue);
144 if (!variable || !constantIndex) {
145 return nullptr;
146 }
147 try {
148 return std::make_shared<GlobalAddress>(
149 variable->symbol->linkName, unary->type,
150 static_cast<size_t>(std::stoull(constantIndex->getLiteral())));
151 } catch (const std::exception &) {
152 return nullptr;
153 }
154 }
155 return nullptr;
156 }
157
158 if (auto literal = dynamic_cast<const sema::BoundLiteral *>(&expression)) {
159 return std::make_shared<Constant>(literal->value, literal->type);
160 }
161
162 if (auto variable =
163 dynamic_cast<const sema::BoundVariableExpression *>(&expression)) {
164 auto symbol = variable->symbol;
165 if (symbol && symbol->isCompileTimeConstant() && symbol->constant_value) {
166 if (!resolvingConstants.insert(symbol.get()).second) {
167 return nullptr;
168 }
169 auto value =
170 lowerConstantExpression(*symbol->constant_value, resolvingConstants);
171 resolvingConstants.erase(symbol.get());
172 return value;
173 }
174 return nullptr;
175 }
176
177 if (auto binary =
178 dynamic_cast<const sema::BoundBinaryExpression *>(&expression)) {
179 auto left = lowerConstantExpression(*binary->left, resolvingConstants);
180 auto right = lowerConstantExpression(*binary->right, resolvingConstants);
181 auto leftConstant = std::dynamic_pointer_cast<Constant>(left);
182 auto rightConstant = std::dynamic_pointer_cast<Constant>(right);
183 if (!leftConstant || !rightConstant) {
184 return nullptr;
185 }
186
187 try {
188 if (binary->type->isFloatingPoint()) {
189 const double lhs = std::stod(leftConstant->getLiteral());
190 const double rhs = std::stod(rightConstant->getLiteral());
191 std::optional<double> result;
192 if (binary->op == "+")
193 result = lhs + rhs;
194 else if (binary->op == "-")
195 result = lhs - rhs;
196 else if (binary->op == "*")
197 result = lhs * rhs;
198 else if (binary->op == "/" && rhs != 0.0)
199 result = lhs / rhs;
200 if (result) {
201 return std::make_shared<Constant>(std::to_string(*result),
202 binary->type);
203 }
204 return nullptr;
205 }
206
207 if ((binary->type->getIntrinsicKind() == IntrinsicTypeKind::String ||
208 binary->type->getIntrinsicKind() == IntrinsicTypeKind::StringView) &&
209 binary->op == "+") {
210 return std::make_shared<Constant>(leftConstant->getLiteral() +
211 rightConstant->getLiteral(),
212 binary->type);
213 }
214
215 if (!binary->type->isInteger()) {
216 return nullptr;
217 }
218 const int64_t lhs = std::stoll(leftConstant->getLiteral(), nullptr, 0);
219 const int64_t rhs = std::stoll(rightConstant->getLiteral(), nullptr, 0);
220 std::optional<int64_t> result;
221 if (binary->op == "+")
222 result = lhs + rhs;
223 else if (binary->op == "-")
224 result = lhs - rhs;
225 else if (binary->op == "*")
226 result = lhs * rhs;
227 else if (binary->op == "/" && rhs != 0)
228 result = lhs / rhs;
229 else if (binary->op == "%" && rhs != 0)
230 result = lhs % rhs;
231 else if (binary->op == "&")
232 result = lhs & rhs;
233 else if (binary->op == "|")
234 result = lhs | rhs;
235 else if (binary->op == "^")
236 result = lhs ^ rhs;
237 else if (binary->op == "<<" && rhs >= 0)
238 result = lhs << rhs;
239 else if (binary->op == ">>" && rhs >= 0)
240 result = lhs >> rhs;
241 if (result) {
242 return std::make_shared<Constant>(std::to_string(*result),
243 binary->type);
244 }
245 } catch (const std::exception &) {
246 }
247 return nullptr;
248 }
249
250 if (auto unary =
251 dynamic_cast<const sema::BoundUnaryExpression *>(&expression)) {
252 auto value = lowerConstantExpression(*unary->expr, resolvingConstants);
253 auto constant = std::dynamic_pointer_cast<Constant>(value);
254 if (!constant || !unary->type->isInteger()) {
255 return nullptr;
256 }
257 try {
258 const int64_t operand = std::stoll(constant->getLiteral(), nullptr, 0);
259 if (unary->op == "-") {
260 return std::make_shared<Constant>(std::to_string(-operand),
261 unary->type);
262 }
263 if (unary->op == "~") {
264 return std::make_shared<Constant>(std::to_string(~operand),
265 unary->type);
266 }
267 } catch (const std::exception &) {
268 }
269 return nullptr;
270 }
271
272 if (auto array = dynamic_cast<const sema::BoundArrayLiteral *>(&expression)) {
273 std::vector<std::shared_ptr<Value>> elements;
274 elements.reserve(array->elements.size());
275 for (const auto &element : array->elements) {
276 auto value = lowerConstantExpression(*element, resolvingConstants);
277 if (!value) {
278 return nullptr;
279 }
280 elements.push_back(std::move(value));
281 }
282 return std::make_shared<ArrayConstant>(array->type, std::move(elements));
283 }
284
285 if (auto record =
286 dynamic_cast<const sema::BoundStructLiteral *>(&expression)) {
287 std::vector<AggregateConstant::FieldValue> fields;
288 fields.reserve(record->fields.size());
289 for (const auto &field : record->fields) {
290 auto value = lowerConstantExpression(*field.second, resolvingConstants);
291 if (!value) {
292 return nullptr;
293 }
294 fields.push_back({field.first, std::move(value)});
295 }
296 return std::make_shared<AggregateConstant>(record->type, std::move(fields));
297 }
298
299 if (auto cast = dynamic_cast<const sema::BoundCast *>(&expression)) {
300 auto value = lowerConstantExpression(*cast->expression, resolvingConstants);
301 if (auto constant = std::dynamic_pointer_cast<Constant>(value)) {
302 return std::make_shared<Constant>(constant->getLiteral(), cast->type);
303 }
304 return value;
305 }
306
307 return nullptr;
308}

310std::shared_ptr<Value> BoundIRGenerator::emitFailableFieldLoad(
311 const std::shared_ptr<Value> &value, int fieldIndex,
312 const std::shared_ptr<Type> &fieldType) {
313 if (fieldType && fieldType->getKind() == TypeKind::Void) {
314 return std::make_shared<Constant>("0", fieldType);
315 }
316 auto fieldAddr = createRegister(std::make_shared<PointerType>(fieldType));
317 currentBlock_->addInstruction(
318 std::make_unique<GetElementPtrInst>(fieldAddr, value, fieldIndex));
319 auto loaded = createRegister(fieldType);
320 currentBlock_->addInstruction(std::make_unique<LoadInst>(loaded, fieldAddr));
321 if (!containsManagedValues(fieldType)) {
322 return loaded;
323 }
324
325 // A failable aggregate owns its managed payload. The aggregate is released
326 // after its selected field is read, so the extracted value needs its own
327 // reference before that release.
328 auto copied = createRegister(fieldType, ownedForType(fieldType));
329 currentBlock_->addInstruction(std::make_unique<CopyInst>(copied, loaded));
330 return copied;
331}

396 symbol->linkName, symbol->returnType, symbol->ownerTypeCodegenName,
397 symbol->isDestructor, symbol->vtableSlot, symbol->isCVariadic);
398 func->returnsRef = symbol->returnsRef;
399 func->resultBorrow = symbol->resultBorrow;

401

403 currentBlock_ = entryBlock.get();
404 currentFunction_->addBlock(std::move(entryBlock));
405
406 for (const auto &paramSymbol : symbol->parameters) {
407 auto argType = paramSymbol->is_ref
408 ? std::static_pointer_cast<Type>(
409 std::make_shared<PointerType>(paramSymbol->type))
410 : paramSymbol->type;
411 const size_t parameterIndex = currentFunction_->arguments.size();
412 const auto parameterOwnership =
413 parameterOwnershipFor(*symbol, parameterIndex);
414 auto arg = std::make_shared<Argument>(

417 parameterOwnership, parameterEscapeFor(*symbol, parameterIndex));
418 const bool borrowedSelf =
419 !symbol->ownerTypeCodegenName.empty() && paramSymbol->name == "self";
420 if (!paramSymbol->is_ref && !paramSymbol->is_variadic_pack &&
421 !borrowedSelf && containsManagedValues(argType)) {
422 arg->setOwnership(ownedForType(argType));
423 }
424 currentFunction_->arguments.push_back(arg);
425

427 ? std::make_shared<PointerType>(paramSymbol->type)
428 : paramSymbol->type;

435 symbolMap_[paramSymbol] = allocaReg;
436 }
437

440 }

443 if (symbol->returnType->getKind() == TypeKind::Void) {
444 emitReturn();
445 } else {
446 auto dummy = std::make_shared<Constant>("0", symbol->returnType);
447 emitReturn(dummy);
448 }
449 }

451 module_->addFunction(std::move(func));
452 currentFunction_ = nullptr;
453 currentBlock_ = nullptr;
454 symbolMap_.clear();

699 std::shared_ptr<Value> addr = nullptr;
700 auto localIt = symbolMap_.find(node.symbol);
701 if (localIt != symbolMap_.end()) {
702 addr = localIt->second;
703 } else {
704 auto globalIt = globalSymbolMap_.find(node.symbol);
705 if (globalIt != globalSymbolMap_.end()) {
706 addr = globalIt->second;
707 }
708 }
709 if (!addr) {
710 std::cerr << "Error: Symbol " << node.symbol->name
711 << " not found in IR symbol map\n";
712 return;
713 }
714 if (node.symbol->is_ref) {
715 auto ptrReg =
716 createRegister(std::make_shared<PointerType>(node.symbol->type));
717 currentBlock_->addInstruction(std::make_unique<LoadInst>(ptrReg, addr));
718 if (evaluateAsAddress_) {
719 valueStack_.push(ptrReg);
720 return;
721 }
722 auto valueReg = createRegister(node.type);
723 currentBlock_->addInstruction(std::make_unique<LoadInst>(valueReg, ptrReg));
724 valueStack_.push(valueReg);
725 return;
726 }
727 if (evaluateAsAddress_) {
728 valueStack_.push(addr);
729 return;
730 }
731 auto storedType =
732 std::static_pointer_cast<PointerType>(addr->getType())->getBaseType();
733 auto loaded = createRegister(storedType);
734 currentBlock_->addInstruction(std::make_unique<LoadInst>(loaded, addr));
735 if (storedType->toString() == node.type->toString()) {
736 valueStack_.push(loaded);
737 return;
738 }
739 auto reg = createRegister(node.type);
740 currentBlock_->addInstruction(
741 std::make_unique<CastInst>(reg, loaded, node.type));
742 valueStack_.push(reg);
743}

755void BoundIRGenerator::visit(sema::BoundBinaryExpression &node) {
756 if (node.op == "&&") {
757 auto rhsLabel = createBlockLabel("and.rhs");
758 auto mergeLabel = createBlockLabel("and.merge");
759
760 node.left->accept(*this);
761 auto leftVal = valueStack_.top();
762 valueStack_.pop();
763 std::string leftBlockLabel = currentBlock_->label;
764
765 currentBlock_->addInstruction(
766 std::make_unique<CondBranchInst>(leftVal, rhsLabel, mergeLabel));
767
768 auto rhsBlock = std::make_unique<BasicBlock>(rhsLabel);
769 auto *rhsBlockPtr = rhsBlock.get();
770 currentFunction_->addBlock(std::move(rhsBlock));
771 currentBlock_ = rhsBlockPtr;
772
773 node.right->accept(*this);
774 auto rightVal = valueStack_.top();
775 valueStack_.pop();
776 std::string actualRhsBlockLabel = currentBlock_->label;
777 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
778
779 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
780 auto *mergeBlockPtr = mergeBlock.get();
781 currentFunction_->addBlock(std::move(mergeBlock));
782 currentBlock_ = mergeBlockPtr;
783
784 std::vector<std::pair<std::string, std::shared_ptr<Value>>> incoming;
785 incoming.push_back(
786 {leftBlockLabel, std::make_shared<Constant>("false", node.type)});
787 incoming.push_back({actualRhsBlockLabel, rightVal});
788
789 auto res = createRegister(node.type, ownershipForPhi(node.type, incoming));
790 currentBlock_->addInstruction(std::make_unique<PhiInst>(res, incoming));
791 valueStack_.push(res);
792 return;
793 }
794
795 if (node.op == "||") {
796 auto rhsLabel = createBlockLabel("or.rhs");
797 auto mergeLabel = createBlockLabel("or.merge");
798
799 node.left->accept(*this);
800 auto leftVal = valueStack_.top();
801 valueStack_.pop();
802 std::string leftBlockLabel = currentBlock_->label;
803
804 currentBlock_->addInstruction(
805 std::make_unique<CondBranchInst>(leftVal, mergeLabel, rhsLabel));
806
807 auto rhsBlock = std::make_unique<BasicBlock>(rhsLabel);
808 auto *rhsBlockPtr = rhsBlock.get();
809 currentFunction_->addBlock(std::move(rhsBlock));
810 currentBlock_ = rhsBlockPtr;
811
812 node.right->accept(*this);
813 auto rightVal = valueStack_.top();
814 valueStack_.pop();
815 std::string actualRhsBlockLabel = currentBlock_->label;
816 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
817
818 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
819 auto *mergeBlockPtr = mergeBlock.get();
820 currentFunction_->addBlock(std::move(mergeBlock));
821 currentBlock_ = mergeBlockPtr;
822
823 std::vector<std::pair<std::string, std::shared_ptr<Value>>> incoming;
824 incoming.push_back(
825 {leftBlockLabel, std::make_shared<Constant>("true", node.type)});
826 incoming.push_back({actualRhsBlockLabel, rightVal});
827
828 auto res = createRegister(node.type, ownershipForPhi(node.type, incoming));
829 currentBlock_->addInstruction(std::make_unique<PhiInst>(res, incoming));
830 valueStack_.push(res);
831 return;
832 }
833
834 node.left->accept(*this);
835 auto left = valueStack_.top();
836 valueStack_.pop();
837
838 node.right->accept(*this);
839 auto right = valueStack_.top();
840 valueStack_.pop();
841
842 const bool ownsStringConcat =
843 node.op == "+" && containsManagedValues(node.type) &&
844 (node.left->type->getIntrinsicKind() == IntrinsicTypeKind::String ||
845 node.right->type->getIntrinsicKind() == IntrinsicTypeKind::String ||
846 node.left->type->getKind() == TypeKind::Char ||
847 node.right->type->getKind() == TypeKind::Char);
848 auto reg =
849 createRegister(node.type, ownsStringConcat ? ValueOwnership::OwnedStrong
850 : ValueOwnership::Borrowed);
851 bool isUnsigned = node.left->type->isUnsigned();
852 if (node.op == "==" || node.op == "!=" || node.op == "<" ||
853 (node.op == ">") || (node.op == "<=") || (node.op == ">=")) {
854 std::string pred;
855 if (node.op == "==")
856 pred = "eq";
857 else if (node.op == "!=")
858 pred = "ne";
859 else if (node.op == "<")
860 pred = isUnsigned ? "ult" : "slt";
861 else if (node.op == ">")
862 pred = isUnsigned ? "ugt" : "sgt";
863 else if (node.op == "<=")
864 pred = isUnsigned ? "ule" : "sle";
865 else if (node.op == ">=")
866 pred = isUnsigned ? "uge" : "sge";
867
868 currentBlock_->addInstruction(
869 std::make_unique<CmpInst>(pred, reg, left, right));
870 } else {
871 OpCode op;
872 if (node.op == "+")
873 op = OpCode::Add;
874 else if (node.op == "-")
875 op = OpCode::Sub;
876 else if (node.op == "*")
877 op = OpCode::Mul;
878 else if (node.op == "/")
879 op = isUnsigned ? OpCode::UDiv : OpCode::SDiv;
880 else if (node.op == "%")
881 op = isUnsigned ? OpCode::URem : OpCode::SRem;
882 else if (node.op == "<<")
883 op = OpCode::Shl;
884 else if (node.op == ">>")
885 op = isUnsigned ? OpCode::LShr : OpCode::AShr;
886 else if (node.op == "&")
887 op = OpCode::BitAnd;
888 else if (node.op == "|")
889 op = OpCode::BitOr;
890 else if (node.op == "^")
891 op = OpCode::BitXor;
892 else
893 op = OpCode::Add;
894
895 currentBlock_->addInstruction(
896 std::make_unique<BinaryInst>(op, reg, left, right));
897 }
898 valueStack_.push(reg);
899}

948void BoundIRGenerator::visit(sema::BoundFunctionCall &node) {
949 std::vector<std::shared_ptr<Value>> args;
950 for (size_t i = 0; i < node.arguments.size(); ++i) {
951 bool oldEvaluateAsAddress = evaluateAsAddress_;
952 evaluateAsAddress_ =
953 (i < node.argumentIsRef.size() && node.argumentIsRef[i]);
954 node.arguments[i]->accept(*this);
955 evaluateAsAddress_ = oldEvaluateAsAddress;
956 auto argument = valueStack_.top();
957 valueStack_.pop();
958 const auto parameterOwnership = parameterOwnershipFor(*node.symbol, i);
959 prepareCallArgument(argument, parameterOwnership);
960 args.push_back(std::move(argument));
961 }
962
963 std::shared_ptr<Value> variadicPack = nullptr;
964 if (node.variadicPack) {
965 node.variadicPack->accept(*this);
966 variadicPack = valueStack_.top();
967 valueStack_.pop();
968 }
969
970 // ref-returning functions return a pointer in LLVM IR
971 auto resultType = node.symbol->returnsRef
972 ? std::static_pointer_cast<zir::Type>(
973 std::make_shared<PointerType>(node.type))
974 : node.type;
975 const bool ownsResult =
976 !node.symbol->returnsRef && containsManagedValues(node.type);
977 auto reg = createRegister(resultType, ownsResult ? ownedForType(node.type)
978 : ValueOwnership::Borrowed);
979 currentBlock_->addInstruction(std::make_unique<CallInst>(
980 reg, node.symbol->linkName, args, node.argumentIsRef, variadicPack));
981
982 // If ref-returning function is used as value (not address), load it
983 if (node.symbol->returnsRef && !evaluateAsAddress_) {
984 auto loadReg = createRegister(node.type);
985 currentBlock_->addInstruction(std::make_unique<LoadInst>(loadReg, reg));
986 valueStack_.push(loadReg);
987 } else {
988 valueStack_.push(reg);
989 }
990}

998void BoundIRGenerator::visit(sema::BoundIndirectCall &node) {
999 node.callee->accept(*this);
1000 auto calleeVal = valueStack_.top();
1001 valueStack_.pop();
1002
1003 std::vector<std::shared_ptr<Value>> args;
1004 const auto functionType =
1005 std::static_pointer_cast<FunctionPointerType>(calleeVal->getType());
1006 for (size_t i = 0; i < node.arguments.size(); ++i) {
1007 auto &arg = node.arguments[i];
1008 arg->accept(*this);
1009 auto argument = valueStack_.top();
1010 valueStack_.pop();
1011 const auto parameterOwnership =
1012 i < functionType->getParameterOwnership().size()
1013 ? functionType->getParameterOwnership()[i]
1014 : ParameterOwnership::Borrow;
1015 prepareCallArgument(argument, parameterOwnership);
1016 args.push_back(std::move(argument));
1017 }
1018
1019 const bool returnsRef = functionType->returnsRef();
1020 auto resultType = returnsRef ? std::static_pointer_cast<zir::Type>(
1021 std::make_shared<PointerType>(node.type))
1022 : node.type;
1023 const bool ownsResult = !returnsRef && containsManagedValues(node.type);
1024 auto reg = createRegister(resultType, ownsResult ? ownedForType(node.type)
1025 : ValueOwnership::Borrowed);
1026 currentBlock_->addInstruction(
1027 std::make_unique<CallInst>(reg, calleeVal, std::move(args)));
1028 if (returnsRef && !evaluateAsAddress_) {
1029 auto loadReg = createRegister(node.type);
1030 currentBlock_->addInstruction(std::make_unique<LoadInst>(loadReg, reg));
1031 valueStack_.push(loadReg);
1032 } else {
1033 valueStack_.push(reg);
1034 }
1035}

1062 currentBlock_->addInstruction(std::make_unique<StoreInst>(

1079void BoundIRGenerator::prepareCallArgument(
1080 std::shared_ptr<Value> &value, ParameterOwnership parameterOwnership) {
1081 if (!transfersOwnership(parameterOwnership) || !value ||
1082 !containsManagedValues(value->getType())) {
1083 return;
1084 }
1085
1086 const bool moveOwnedValue = parameterOwnership == ParameterOwnership::Sink &&
1087 isOwned(value->getOwnership());
1088 const auto resultOwnership =
1089 moveOwnedValue ? value->getOwnership() : ownedForType(value->getType());
1090 auto prepared = createRegister(value->getType(), resultOwnership);
1091 if (moveOwnedValue) {
1092 currentBlock_->addInstruction(std::make_unique<MoveInst>(prepared, value));
1093 } else {
1094 currentBlock_->addInstruction(std::make_unique<CopyInst>(prepared, value));
1095 }
1096 value = std::move(prepared);
1097}

1113void BoundIRGenerator::visit(sema::BoundUnaryExpression &node) {
1114 if (node.op == "&") {
1115 bool oldEvaluateAsAddress = evaluateAsAddress_;
1116 evaluateAsAddress_ = true;
1117 node.expr->accept(*this);
1118 evaluateAsAddress_ = oldEvaluateAsAddress;
1119 return;
1120 }
1121
1122 if (node.op == "*") {
1123 bool oldEvaluateAsAddress = evaluateAsAddress_;
1124 evaluateAsAddress_ = false;
1125 node.expr->accept(*this);
1126 evaluateAsAddress_ = oldEvaluateAsAddress;
1127
1128 auto ptr = valueStack_.top();
1129 valueStack_.pop();
1130 if (evaluateAsAddress_) {
1131 valueStack_.push(ptr);
1132 return;
1133 }
1134
1135 auto reg = createRegister(node.type);
1136 currentBlock_->addInstruction(std::make_unique<LoadInst>(reg, ptr));
1137 valueStack_.push(reg);
1138 return;
1139 }
1140
1141 node.expr->accept(*this);
1142 auto expr = valueStack_.top();
1143 valueStack_.pop();
1144
1145 // Global initializers are lowered as constants only. Fold unary operators on
1146 // constants instead of emitting runtime IR outside functions.
1147 if (!currentFunction_ || !currentBlock_) {
1148 if (auto c = std::dynamic_pointer_cast<Constant>(expr)) {
1149 const auto &lit = c->getLiteral();
1150
1151 if (node.op == "+") {
1152 valueStack_.push(c);
1153 return;
1154 }
1155
1156 if (node.op == "-") {
1157 if (!lit.empty() && lit != "true" && lit != "false" && lit != "null" &&
1158 lit[0] != '\'' && lit[0] != '\\') {
1159 try {
1160 if (c->getType() && c->getType()->isFloatingPoint()) {
1161 auto v = std::stod(lit);
1162 valueStack_.push(
1163 std::make_shared<Constant>(std::to_string(-v), node.type));
1164 return;
1165 }
1166 if (c->getType() && c->getType()->isInteger()) {
1167 if (c->getType()->isUnsigned()) {
1168 auto v = std::stoull(lit);
1169 auto out = static_cast<int64_t>(-(static_cast<int64_t>(v)));
1170 valueStack_.push(
1171 std::make_shared<Constant>(std::to_string(out), node.type));
1172 } else {
1173 auto v = std::stoll(lit);
1174 valueStack_.push(
1175 std::make_shared<Constant>(std::to_string(-v), node.type));
1176 }
1177 return;
1178 }
1179 } catch (...) {
1180 }
1181 }
1182 }
1183
1184 if (node.op == "!") {
1185 if (lit == "true") {
1186 valueStack_.push(std::make_shared<Constant>("false", node.type));
1187 return;
1188 }
1189 if (lit == "false") {
1190 valueStack_.push(std::make_shared<Constant>("true", node.type));
1191 return;
1192 }
1193 }
1194
1195 if (node.op == "~") {
1196 if (!lit.empty() && lit != "true" && lit != "false" && lit != "null" &&
1197 lit[0] != '\'' && lit[0] != '\\') {
1198 try {
1199 auto v = std::stoll(lit);
1200 valueStack_.push(
1201 std::make_shared<Constant>(std::to_string(~v), node.type));
1202 return;
1203 } catch (...) {
1204 }
1205 }
1206 }
1207 }
1208
1209 valueStack_.push(expr);
1210 return;
1211 }
1212
1213 if (node.op == "+") {
1214 valueStack_.push(expr);
1215 return;
1216 }
1217
1218 if (node.op == "-") {
1219 auto zero = std::make_shared<Constant>("0", node.type);
1220 auto reg = createRegister(node.type);
1221 currentBlock_->addInstruction(
1222 std::make_unique<BinaryInst>(OpCode::Sub, reg, zero, expr));
1223 valueStack_.push(reg);
1224 return;
1225 }
1226
1227 if (node.op == "!") {
1228 auto zero = std::make_shared<Constant>(
1229 "false", std::make_shared<PrimitiveType>(TypeKind::Bool));
1230 auto reg = createRegister(node.type);
1231 currentBlock_->addInstruction(
1232 std::make_unique<CmpInst>("eq", reg, expr, zero));
1233 valueStack_.push(reg);
1234 return;
1235 }
1236
1237 if (node.op == "~") {
1238 auto allOnes = std::make_shared<Constant>("-1", node.type);
1239 auto reg = createRegister(node.type);
1240 currentBlock_->addInstruction(
1241 std::make_unique<BinaryInst>(OpCode::BitXor, reg, expr, allOnes));
1242 valueStack_.push(reg);
1243 return;
1244 }
1245
1246 valueStack_.push(expr);
1247}

1285void BoundIRGenerator::visit(sema::BoundMemberAccess &node) {
1286 if (node.left->type->getKind() == zir::TypeKind::Enum) {
1287 node.left->accept(*this);
1288 auto left = std::move(valueStack_.top());
1289 valueStack_.pop();
1290
1291 auto enumType = std::static_pointer_cast<zir::EnumType>(left->getType());
1292 int64_t value = enumType->getVariantDiscriminant(node.member);
1293 if (value != -1) {
1294 valueStack_.push(std::make_shared<Constant>(
1295 std::to_string(value),
1296 std::make_shared<zir::PrimitiveType>(zir::TypeKind::Int)));
1297 return;
1298 }
1299 }
1300 if (node.left->type->getKind() == zir::TypeKind::TaggedUnion &&
1301 node.member == "tag") {
1302 bool oldEvaluateAsAddress = evaluateAsAddress_;
1303 evaluateAsAddress_ = true;
1304 node.left->accept(*this);
1305 evaluateAsAddress_ = oldEvaluateAsAddress;
1306
1307 auto left = std::move(valueStack_.top());
1308 valueStack_.pop();
1309 auto fieldAddr = createRegister(std::make_shared<PointerType>(
1310 std::make_shared<PrimitiveType>(TypeKind::Int32)));
1311 currentBlock_->addInstruction(
1312 std::make_unique<GetElementPtrInst>(fieldAddr, left, 0));
1313 if (evaluateAsAddress_) {
1314 valueStack_.push(fieldAddr);
1315 } else {
1316 auto result = createRegister(node.type);
1317 currentBlock_->addInstruction(
1318 std::make_unique<LoadInst>(result, fieldAddr));
1319 valueStack_.push(result);
1320 }
1321 return;
1322 }
1323
1324 bool oldEvaluateAsAddress = evaluateAsAddress_;
1325 evaluateAsAddress_ = !(node.left->type->getKind() == zir::TypeKind::Class ||
1326 node.left->type->getKind() == zir::TypeKind::Pointer);
1327 node.left->accept(*this);
1328 evaluateAsAddress_ = oldEvaluateAsAddress;
1329
1330 auto left = std::move(valueStack_.top());
1331 valueStack_.pop();
1332
1333 if (node.left->type->getKind() == zir::TypeKind::Class) {
1334 auto classType = std::static_pointer_cast<zir::ClassType>(node.left->type);
1335 int fieldIndex = -1;
1336 const auto &fields = classType->getFields();
1337 for (size_t i = 0; i < fields.size(); ++i) {
1338 if (fields[i].name == node.member) {
1339 fieldIndex = static_cast<int>(i);
1340 break;
1341 }
1342 }
1343
1344 if (fieldIndex != -1) {
1345 auto fieldAddr = createRegister(
1346 std::make_shared<PointerType>(fields[fieldIndex].type));
1347 currentBlock_->addInstruction(
1348 std::make_unique<GetElementPtrInst>(fieldAddr, left, fieldIndex));
1349 if (evaluateAsAddress_) {
1350 valueStack_.push(fieldAddr);
1351 } else {
1352 auto result = createRegister(fields[fieldIndex].type);
1353 currentBlock_->addInstruction(
1354 std::make_unique<LoadInst>(result, fieldAddr));
1355 valueStack_.push(result);
1356 }
1357 return;
1358 }
1359 } else if (left->getType()->getKind() == zir::TypeKind::Pointer) {
1360 auto baseType = std::static_pointer_cast<zir::PointerType>(left->getType())
1361 ->getBaseType();
1362 if (baseType->getKind() == zir::TypeKind::Class) {
1363 auto classType = std::static_pointer_cast<zir::ClassType>(baseType);
1364 int fieldIndex = -1;
1365 const auto &fields = classType->getFields();
1366 for (size_t i = 0; i < fields.size(); ++i) {
1367 if (fields[i].name == node.member) {
1368 fieldIndex = static_cast<int>(i);
1369 break;
1370 }
1371 }
1372
1373 if (fieldIndex != -1) {
1374 auto fieldAddr = createRegister(
1375 std::make_shared<PointerType>(fields[fieldIndex].type));
1376 currentBlock_->addInstruction(
1377 std::make_unique<GetElementPtrInst>(fieldAddr, left, fieldIndex));
1378 if (evaluateAsAddress_) {
1379 valueStack_.push(fieldAddr);
1380 } else {
1381 auto result = createRegister(fields[fieldIndex].type);
1382 currentBlock_->addInstruction(
1383 std::make_unique<LoadInst>(result, fieldAddr));
1384 valueStack_.push(result);
1385 }
1386 return;
1387 }
1388 } else if (baseType->getKind() == zir::TypeKind::Record) {
1389 auto recordType = std::static_pointer_cast<zir::RecordType>(baseType);
1390 int fieldIndex = -1;
1391 const auto &fields = recordType->getFields();
1392 for (size_t i = 0; i < fields.size(); ++i) {
1393 if (fields[i].name == node.member) {
1394 fieldIndex = static_cast<int>(i);
1395 break;
1396 }
1397 }
1398
1399 if (fieldIndex != -1) {
1400 auto fieldAddr = createRegister(
1401 std::make_shared<PointerType>(fields[fieldIndex].type));
1402 currentBlock_->addInstruction(
1403 std::make_unique<GetElementPtrInst>(fieldAddr, left, fieldIndex));
1404 if (evaluateAsAddress_) {
1405 valueStack_.push(fieldAddr);
1406 } else {
1407 auto result = createRegister(fields[fieldIndex].type);
1408 currentBlock_->addInstruction(
1409 std::make_unique<LoadInst>(result, fieldAddr));
1410 valueStack_.push(result);
1411 }
1412 return;
1413 }
1414 }
1415 } else if (left->getType()->getKind() == zir::TypeKind::Record) {
1416 auto recordType =
1417 std::static_pointer_cast<zir::RecordType>(left->getType());
1418 int fieldIndex = -1;
1419 const auto &fields = recordType->getFields();
1420 for (size_t i = 0; i < fields.size(); ++i) {
1421 if (fields[i].name == node.member) {
1422 fieldIndex = static_cast<int>(i);
1423 break;
1424 }
1425 }
1426
1427 if (fieldIndex != -1) {
1428 auto fieldAddr = createRegister(
1429 std::make_shared<PointerType>(fields[fieldIndex].type));
1430 currentBlock_->addInstruction(
1431 std::make_unique<GetElementPtrInst>(fieldAddr, left, fieldIndex));
1432 if (evaluateAsAddress_) {
1433 valueStack_.push(fieldAddr);
1434 } else {
1435 auto result = createRegister(fields[fieldIndex].type);
1436 currentBlock_->addInstruction(
1437 std::make_unique<LoadInst>(result, fieldAddr));
1438 valueStack_.push(result);
1439 }
1440 return;
1441 }
1442 }
1443
1444 throw std::runtime_error("IR member access failed: member '" + node.member +
1445 "' not found in type '" +
1446 renderTypeForUser(left->getType()) + "'");
1447}

1449void BoundIRGenerator::visit(sema::BoundStructLiteral &node) {
1450 auto recordType = std::static_pointer_cast<zir::RecordType>(node.type);
1451
1452 // Global initializers are lowered as constants only.
1453 if (!currentFunction_ || !currentBlock_) {
1454 std::vector<AggregateConstant::FieldValue> aggregateFields;
1455 aggregateFields.reserve(node.fields.size());
1456
1457 for (const auto &fieldInit : node.fields) {
1458 fieldInit.second->accept(*this);
1459 auto val = std::move(valueStack_.top());
1460 valueStack_.pop();
1461 aggregateFields.push_back({fieldInit.first, val});
1462 }
1463
1464 valueStack_.push(std::make_shared<AggregateConstant>(
1465 node.type, std::move(aggregateFields)));
1466 return;
1467 }
1468
1469 auto allocaReg = createRegister(std::make_shared<PointerType>(recordType));
1470 currentBlock_->addInstruction(
1471 std::make_unique<AllocaInst>(allocaReg, recordType));
1472
1473 for (const auto &fieldInit : node.fields) {
1474 // Find field index
1475 int fieldIndex = -1;
1476 const auto &fields = recordType->getFields();
1477 for (size_t i = 0; i < fields.size(); ++i) {
1478 if (fields[i].name == fieldInit.first) {
1479 fieldIndex = static_cast<int>(i);
1480 break;
1481 }
1482 }
1483
1484 fieldInit.second->accept(*this);
1485 auto val = std::move(valueStack_.top());
1486 valueStack_.pop();
1487
1488 if (fields[fieldIndex].type &&
1489 fields[fieldIndex].type->getKind() != TypeKind::Void) {
1490 auto fieldAddr = createRegister(
1491 std::make_shared<PointerType>(fields[fieldIndex].type));
1492 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1493 fieldAddr, allocaReg, fieldIndex));
1494 emitInitializationStore(std::move(val), fieldAddr);
1495 }
1496 }
1497
1498 auto result = createRegister(recordType);
1499 currentBlock_->addInstruction(std::make_unique<LoadInst>(result, allocaReg));
1500 valueStack_.push(result);
1501}

1534void BoundIRGenerator::visit(sema::BoundTryExpression &node) {
1535 node.expression->accept(*this);
1536 if (valueStack_.empty()) {
1537 return;
1538 }
1539
1540 auto failableValue = valueStack_.top();
1541 valueStack_.pop();
1542
1543 auto okValue = emitFailableOk(failableValue);
1544 auto successLabel = createBlockLabel("try.ok");
1545 auto failLabel = createBlockLabel("try.fail");
1546 auto mergeLabel = createBlockLabel("try.merge");
1547
1548 currentBlock_->addInstruction(
1549 std::make_unique<CondBranchInst>(okValue, successLabel, failLabel));
1550
1551 auto successBlock = std::make_unique<BasicBlock>(successLabel);
1552 auto *successBlockPtr = successBlock.get();
1553 currentFunction_->addBlock(std::move(successBlock));
1554 currentBlock_ = successBlockPtr;
1555
1556 auto successValue = emitFailableValue(failableValue);
1557 std::string successFrom = currentBlock_->label;
1558 bool resultIsVoid = node.type && node.type->getKind() == TypeKind::Void;
1559 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
1560
1561 auto failBlock = std::make_unique<BasicBlock>(failLabel);
1562 auto *failBlockPtr = failBlock.get();
1563 currentFunction_->addBlock(std::move(failBlock));
1564 currentBlock_ = failBlockPtr;
1565
1566 auto failErr = emitFailableError(failableValue);
1567
1568 auto propagatedType = node.propagatedType;
1569 auto propagatedLayout = getFailableTypeLayout(propagatedType);
1570 auto propagatedValueType =
1571 propagatedLayout ? propagatedLayout->valueType : nullptr;
1572 auto propagatedErrorType =
1573 propagatedLayout ? propagatedLayout->errorType : nullptr;
1574
1575 auto propagatedAlloca =
1576 createRegister(std::make_shared<PointerType>(propagatedType));
1577 currentBlock_->addInstruction(
1578 std::make_unique<AllocaInst>(propagatedAlloca, propagatedType));
1579
1580 auto okAddr = createRegister(std::make_shared<PointerType>(
1581 std::make_shared<PrimitiveType>(TypeKind::Bool)));
1582 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1583 okAddr, propagatedAlloca, FailableTypeLayout::OkField));
1584 currentBlock_->addInstruction(std::make_unique<StoreInst>(
1585 std::make_shared<Constant>(
1586 "false", std::make_shared<PrimitiveType>(TypeKind::Bool)),
1587 okAddr, StoreMode::Assign));
1588
1589 auto valueAddr =
1590 createRegister(std::make_shared<PointerType>(propagatedValueType));
1591 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1592 valueAddr, propagatedAlloca, FailableTypeLayout::ValueField));
1593 if (propagatedValueType && propagatedValueType->getKind() != TypeKind::Void) {
1594 currentBlock_->addInstruction(std::make_unique<StoreInst>(
1595 std::make_shared<Constant>("0", propagatedValueType), valueAddr,
1596 StoreMode::Assign));
1597 }
1598
1599 auto errAddr =
1600 createRegister(std::make_shared<PointerType>(propagatedErrorType));
1601 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1602 errAddr, propagatedAlloca, FailableTypeLayout::ErrorField));
1603 currentBlock_->addInstruction(
1604 std::make_unique<StoreInst>(failErr, errAddr, StoreMode::Assign));
1605
1606 auto propagatedValue = createRegister(propagatedType);
1607 currentBlock_->addInstruction(
1608 std::make_unique<LoadInst>(propagatedValue, propagatedAlloca));
1609 emitReturn(propagatedValue);
1610
1611 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
1612 auto *mergeBlockPtr = mergeBlock.get();
1613 currentFunction_->addBlock(std::move(mergeBlock));
1614 currentBlock_ = mergeBlockPtr;
1615
1616 if (resultIsVoid) {
1617 valueStack_.push(std::make_shared<Constant>("0", node.type));
1618 return;
1619 }
1620
1621 std::vector<std::pair<std::string, std::shared_ptr<Value>>> incoming;
1622 incoming.push_back({successFrom, successValue});
1623 auto result = createRegister(node.type, ownershipForPhi(node.type, incoming));
1624 currentBlock_->addInstruction(std::make_unique<PhiInst>(result, incoming));
1625 valueStack_.push(result);
1626}

1685 node.expression->accept(*this);
1686 if (valueStack_.empty()) {
1687 return;
1688 }
1689
1690 auto failableValue = valueStack_.top();
1691 valueStack_.pop();
1692
1693 auto okValue = emitFailableOk(failableValue);
1694 auto successLabel = createBlockLabel("orerr.ok");
1695 auto handlerLabel = createBlockLabel("orerr.handler");
1696 auto mergeLabel = createBlockLabel("orerr.merge");
1697
1698 currentBlock_->addInstruction(
1699 std::make_unique<CondBranchInst>(okValue, successLabel, handlerLabel));
1700
1701 auto successBlock = std::make_unique<BasicBlock>(successLabel);
1702 auto *successBlockPtr = successBlock.get();
1703 currentFunction_->addBlock(std::move(successBlock));
1704 currentBlock_ = successBlockPtr;
1705 auto successValue = emitFailableValue(failableValue);
1706 successValue = materializeOwnedValue(successValue);
1707 std::string successFrom = currentBlock_->label;
1708 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
1709
1710 auto handlerBlock = std::make_unique<BasicBlock>(handlerLabel);
1711 auto *handlerBlockPtr = handlerBlock.get();
1712 currentFunction_->addBlock(std::move(handlerBlock));
1713 currentBlock_ = handlerBlockPtr;
1714
1715 auto errValue = emitFailableError(failableValue);
1716 if (node.errorSymbol) {
1717 auto errAlloca =
1718 createRegister(std::make_shared<PointerType>(node.errorSymbol->type));
1719 currentBlock_->addInstruction(
1720 std::make_unique<AllocaInst>(errAlloca, node.errorSymbol->type));
1721 currentBlock_->addInstruction(
1722 std::make_unique<StoreInst>(errValue, errAlloca, StoreMode::Assign));
1723 symbolMap_[node.errorSymbol] = errAlloca;
1724 }
1725
1726 if (node.handler) {
1727 node.handler->accept(*this);
1728 }
1729
1730 bool resultIsVoid = node.type && node.type->getKind() == TypeKind::Void;
1731 std::shared_ptr<Value> handlerValue = nullptr;
1732 if (node.handler && node.handler->result) {
1733 handlerValue = valueStack_.top();
1734 valueStack_.pop();
1735 } else {
1736 handlerValue = std::make_shared<Constant>("0", node.type);
1737 }
1738
1739 bool handlerReachesMerge = !isTerminated(currentBlock_);
1740 if (handlerReachesMerge) {
1741 handlerValue = materializeOwnedValue(handlerValue);
1742 }
1743 std::string handlerFrom = currentBlock_->label;
1744 if (handlerReachesMerge) {
1745 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
1746 }
1747
1748 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
1749 auto *mergeBlockPtr = mergeBlock.get();
1750 currentFunction_->addBlock(std::move(mergeBlock));
1751 currentBlock_ = mergeBlockPtr;
1752
1753 if (!handlerReachesMerge) {
1754 valueStack_.push(resultIsVoid ? std::make_shared<Constant>("0", node.type)
1755 : successValue);
1756 return;
1757 }
1758
1759 if (resultIsVoid) {
1760 valueStack_.push(std::make_shared<Constant>("0", node.type));
1761 return;
1762 }
1763
1764 std::vector<std::pair<std::string, std::shared_ptr<Value>>> incoming;
1765 incoming.push_back({successFrom, successValue});
1766 incoming.push_back({handlerFrom, handlerValue});
1767 auto result = createRegister(node.type, ownershipForPhi(node.type, incoming));
1768 currentBlock_->addInstruction(std::make_unique<PhiInst>(result, incoming));
1769 valueStack_.push(result);
1770}

1777void BoundIRGenerator::visit(sema::BoundIfStatement &node) {
1778 auto trueLabel = createBlockLabel("if.then");
1779 auto falseLabel = node.elseBody ? createBlockLabel("if.else") : "";
1780 auto mergeLabel = createBlockLabel("if.merge");
1781
1782 node.condition->accept(*this);
1783 auto cond = valueStack_.top();
1784 valueStack_.pop();
1785
1786 if (node.elseBody) {
1787 currentBlock_->addInstruction(
1788 std::make_unique<CondBranchInst>(cond, trueLabel, falseLabel));
1789 } else {
1790 currentBlock_->addInstruction(
1791 std::make_unique<CondBranchInst>(cond, trueLabel, mergeLabel));
1792 }
1793
1794 auto thenBlock = std::make_unique<BasicBlock>(trueLabel);
1795 auto *thenBlockPtr = thenBlock.get();
1796 currentFunction_->addBlock(std::move(thenBlock));
1797 currentBlock_ = thenBlockPtr;
1798
1799 std::shared_ptr<Value> narrowedAddress;
1800 if (node.narrowedSource && node.narrowedVariable) {
1801 auto sourceIt = symbolMap_.find(node.narrowedSource);
1802 if (sourceIt != symbolMap_.end()) {
1803 narrowedAddress = sourceIt->second;
1804 symbolMap_[node.narrowedVariable] = narrowedAddress;
1805 }
1806 }
1807 if (node.thenBody)
1808 node.thenBody->accept(*this);
1809 if (node.narrowedVariable && narrowedAddress) {
1810 symbolMap_.erase(node.narrowedVariable);
1811 }
1812
1813 std::string actualThenLabel = currentBlock_->label;
1814
1815 if (!isTerminated(currentBlock_)) {
1816 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
1817 }
1818
1819 std::string actualElseLabel = "";
1820 if (node.elseBody) {
1821 auto elseBlock = std::make_unique<BasicBlock>(falseLabel);
1822 auto *elseBlockPtr = elseBlock.get();
1823 currentFunction_->addBlock(std::move(elseBlock));
1824 currentBlock_ = elseBlockPtr;
1825 node.elseBody->accept(*this);
1826
1827 actualElseLabel = currentBlock_->label;
1828
1829 if (!isTerminated(currentBlock_)) {
1830 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
1831 }
1832 }
1833
1834 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
1835 auto *mergeBlockPtr = mergeBlock.get();
1836 currentFunction_->addBlock(std::move(mergeBlock));
1837 currentBlock_ = mergeBlockPtr;
1838}

1840void BoundIRGenerator::emitCaseRecordTest(const sema::BoundCasePattern &record,
1841 const std::shared_ptr<Value> &address,
1842 const std::string &successLabel,
1843 const std::string &failureLabel) {
1844 std::vector<const sema::BoundCaseRecordField *> constrainedFields;
1845 for (const auto &field : record.recordFields) {
1846 if (field.nested) {
1847 constrainedFields.push_back(&field);
1848 }
1849 }
1850 for (size_t fieldIndex = 0; fieldIndex < constrainedFields.size();
1851 ++fieldIndex) {
1852 const auto &field = *constrainedFields[fieldIndex];
1853 const auto nextLabel = fieldIndex + 1 == constrainedFields.size()
1854 ? successLabel
1855 : createBlockLabel("case.record.next");
1856 const auto fieldType = record.recordType->getFields()[field.index].type;
1857 auto fieldAddress =
1858 createRegister(std::make_shared<PointerType>(fieldType));
1859 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1860 fieldAddress, address, field.index));
1861 if (field.nested->kind == sema::BoundCasePatternKind::Record) {
1862 emitCaseRecordTest(*field.nested, fieldAddress, nextLabel, failureLabel);
1863 } else if (field.nested->kind ==
1864 sema::BoundCasePatternKind::TaggedUnionVariant) {
1865 auto tagType = std::make_shared<PrimitiveType>(TypeKind::Int32);
1866 auto tagAddress = createRegister(std::make_shared<PointerType>(tagType));
1867 currentBlock_->addInstruction(
1868 std::make_unique<GetElementPtrInst>(tagAddress, fieldAddress, 0));
1869 auto tag = createRegister(tagType);
1870 currentBlock_->addInstruction(
1871 std::make_unique<LoadInst>(tag, tagAddress));
1872 auto expectedTag = std::make_shared<Constant>(
1873 std::to_string(field.nested->variantTag), tagType);
1874 auto tagMatches =
1875 createRegister(std::make_shared<PrimitiveType>(TypeKind::Bool));
1876 currentBlock_->addInstruction(
1877 std::make_unique<CmpInst>("eq", tagMatches, tag, expectedTag));
1878 if (!field.nested->payloadValue && !field.nested->payloadPattern) {
1879 currentBlock_->addInstruction(std::make_unique<CondBranchInst>(
1880 tagMatches, nextLabel, failureLabel));
1881 } else {
1882 const auto payloadLabel = createBlockLabel("case.payload");
1883 currentBlock_->addInstruction(std::make_unique<CondBranchInst>(
1884 tagMatches, payloadLabel, failureLabel));
1885 auto payloadBlock = std::make_unique<BasicBlock>(payloadLabel);
1886 auto *payloadBlockPtr = payloadBlock.get();
1887 currentFunction_->addBlock(std::move(payloadBlock));
1888 currentBlock_ = payloadBlockPtr;
1889 auto payloadAddress = createRegister(
1890 std::make_shared<PointerType>(field.nested->payloadType));
1891 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
1892 payloadAddress, fieldAddress, 1));
1893 if (field.nested->payloadPattern) {
1894 emitCaseRecordTest(*field.nested->payloadPattern, payloadAddress,
1895 nextLabel, failureLabel);
1896 } else {
1897 auto payload = createRegister(field.nested->payloadType);
1898 currentBlock_->addInstruction(
1899 std::make_unique<LoadInst>(payload, payloadAddress));
1900 if (field.nested->payloadValue->type->getIntrinsicKind() ==
1901 IntrinsicTypeKind::StringView &&
1902 payload->getType()->getIntrinsicKind() ==
1903 IntrinsicTypeKind::String) {
1904 auto payloadView = createRegister(field.nested->payloadValue->type,
1905 ValueOwnership::Borrowed);
1906 currentBlock_->addInstruction(
1907 std::make_unique<BorrowInst>(payloadView, payload));
1908 payload = std::move(payloadView);
1909 }
1910 field.nested->payloadValue->accept(*this);
1911 auto expectedPayload = valueStack_.top();
1912 valueStack_.pop();
1913 auto payloadMatches =
1914 createRegister(std::make_shared<PrimitiveType>(TypeKind::Bool));
1915 currentBlock_->addInstruction(std::make_unique<CmpInst>(
1916 "eq", payloadMatches, payload, expectedPayload));
1917 currentBlock_->addInstruction(std::make_unique<CondBranchInst>(
1918 payloadMatches, nextLabel, failureLabel));
1919 }
1920 }
1921 } else {
1922 auto fieldValue = createRegister(fieldType);
1923 auto fieldMatches =
1924 createRegister(std::make_shared<PrimitiveType>(TypeKind::Bool));
1925 currentBlock_->addInstruction(
1926 std::make_unique<LoadInst>(fieldValue, fieldAddress));
1927 if (field.nested->kind == sema::BoundCasePatternKind::EnumVariant) {
1928 auto expected = std::make_shared<Constant>(
1929 std::to_string(field.nested->variantTag), fieldType);
1930 currentBlock_->addInstruction(std::make_unique<CmpInst>(
1931 "eq", fieldMatches, fieldValue, expected));
1932 } else {
1933 field.nested->value->accept(*this);
1934 auto expected = valueStack_.top();
1935 valueStack_.pop();
1936 currentBlock_->addInstruction(std::make_unique<CmpInst>(
1937 "eq", fieldMatches, fieldValue, expected));
1938 }
1939 currentBlock_->addInstruction(std::make_unique<CondBranchInst>(
1940 fieldMatches, nextLabel, failureLabel));
1941 }
1942 if (fieldIndex + 1 < constrainedFields.size()) {
1943 auto nextBlock = std::make_unique<BasicBlock>(nextLabel);
1944 auto *nextBlockPtr = nextBlock.get();
1945 currentFunction_->addBlock(std::move(nextBlock));
1946 currentBlock_ = nextBlockPtr;
1947 }
1948 }
1949 if (constrainedFields.empty()) {
1950 currentBlock_->addInstruction(std::make_unique<BranchInst>(successLabel));
1951 }
1952}

2205 currentBlock_ = bodyBlockPtr;

2209 arm.patterns.front().kind == sema::BoundCasePatternKind::Record) {
2210 materializeCaseRecordBindings(arm.patterns.front(), recordAddress);
2211 }
2212 if (!arm.patterns.empty() &&
2213 arm.patterns.front().kind ==
2214 sema::BoundCasePatternKind::TaggedUnionVariant &&
2215 arm.patterns.front().payloadPattern) {
2216 const auto &pattern = arm.patterns.front();
2217 auto payloadAddress =
2218 createRegister(std::make_shared<PointerType>(pattern.payloadType));
2219 currentBlock_->addInstruction(std::make_unique<GetElementPtrInst>(
2220 payloadAddress, taggedUnionAddress, 1));
2221 materializeCaseRecordBindings(*pattern.payloadPattern, payloadAddress);
2222 }
2223 materializeCasePayloadBinding(arm, taggedUnionAddress);
2224 if (arm.body) {
2225 arm.body->accept(*this);
2226 }
2227 if (arm.payloadBinding) {
2228 symbolMap_.erase(arm.payloadBinding);
2229 }
2230 for (const auto &binding : arm.recordBindings) {
2231 symbolMap_.erase(binding);
2232 }
2233 if (!isTerminated(currentBlock_)) {
2234 currentBlock_->addInstruction(std::make_unique<BranchInst>(mergeLabel));
2235 }
2236 }
2237
2238 auto mergeBlock = std::make_unique<BasicBlock>(mergeLabel);
2239 auto *mergeBlockPtr = mergeBlock.get();
2240 currentFunction_->addBlock(std::move(mergeBlock));
2241 currentBlock_ = mergeBlockPtr;
2242}
