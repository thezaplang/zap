531 std::string toString() const override {
532 std::string s = "asm \"" + assembly + "\"";
533 auto appendOperands = [&](const char *label,
534 const std::vector<AsmOperand> &ops) {
535 if (ops.empty())
536 return;
537 s += std::string(" ") + label + " ";
538 for (size_t i = 0; i < ops.size(); ++i) {
539 s += "\"" + ops[i].constraint + "\"(" +
540 (ops[i].value ? ops[i].value->getName() : "?") + ")" +
541 (i + 1 < ops.size() ? ", " : "");
542 }
543 };
544 appendOperands("out", outputs);
545 appendOperands("in", inputs);
546 if (!clobbers.empty()) {
547 s += " clobbers ";
548 for (size_t i = 0; i < clobbers.size(); ++i) {
549 s += "\"" + clobbers[i] + "\"" + (i + 1 < clobbers.size() ? ", " : "");
550 }
