108ZirVerifier::verifyOwnershipObligations(const Module &module) const {
109 VerificationResult result;
110 for (const auto &function : module.getFunctions()) {
111 if (!function) {
112 continue;
113 }
114 ControlFlowGraph cfg(*function);
115 OwnershipFlowAnalysis analysis(module, *function, cfg.predecessors(),
116 cfg.successors(), cfg.reachable());
117 for (const auto &plan : analysis.analyzeOwnershipClosurePlans()) {
118 std::string definition = "function argument";
119 if (plan.definition.block) {
120 definition = "block %" + plan.definition.block->label;
121 if (plan.definition.instructionIndex) {
122 definition += " instruction " +
123 std::to_string(*plan.definition.instructionIndex);
124 }
125 }
126 for (const auto &obligation : plan.liveExits) {
127 std::string placement = "no safe destroy placement determined";
128 for (const auto &candidate : plan.destroyPlacements) {
129 if (candidate.destination != obligation.block ||
130 (candidate.kind == OwnershipDestroyPlacementKind::BeforeReturn &&
131 candidate.instructionIndex != obligation.instructionIndex)) {
132 continue;
133 }
134 placement = formatDestroyPlacement(candidate);
135 break;
136 }
137 result.errors_.push_back(
138 {VerificationErrorCode::OwnershipViolation, function->name,
139 obligation.block ? obligation.block->label : std::string{},
140 obligation.instructionIndex,
141 "owned value may remain live at function exit: " +
142 plan.value->getName() + " (defined in " + definition + "; " +
143 "state: " + formatOwnershipFlowState(obligation.state) + "; " +
144 placement + ")"});
145 }
146 }
147 }
