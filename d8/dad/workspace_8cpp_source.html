106AnalysisResult Workspace::watchedFilesChanged(
107 const std::vector<std::filesystem::path> &paths) {
108 std::set<std::string> changedProjects;
109 bool changedUnconfiguredFile = false;
110 for (const auto &path : paths) {
111 const auto canonical = std::filesystem::weakly_canonical(path).string();
112 sourceManager_.invalidatePath(path);
113 invalidateSnapshotsForPath(path);
114 if (path.filename() == "thor.toml") {
115 projectConfigurations_.erase(canonical);
116 changedProjects.insert(canonical);
117 continue;
118 }
119 if (const auto manifest =
120 zap::frontend::findProjectConfigurationManifest(path)) {
121 changedProjects.insert(manifest->string());
122 } else {
123 changedUnconfiguredFile = true;
124 }
125 }
126
127 std::vector<std::string> affectedUris;
128 for (const auto &uri : sourceManager_.openUris()) {
129 const auto *document = this->document(uri);
130 if (!document) {
131 continue;
132 }
133 const auto manifest =
134 zap::frontend::findProjectConfigurationManifest(document->path);
135 if ((manifest && changedProjects.count(manifest->string()) != 0) ||
136 (!manifest && changedUnconfiguredFile)) {
137 invalidateSnapshots(uri);
138 affectedUris.push_back(uri);
139 }
140 }
141
142 AnalysisResult result;
143 for (const auto &uri : affectedUris) {
144 auto project = loadProject(uri);
145 if (!project) {
146 continue;
147 }
148 for (const auto &[diagnosticUri, diagnostics] :
149 project->analysis.diagnosticsByUri) {
150 auto &merged = result.diagnosticsByUri[diagnosticUri];
151 merged.insert(merged.end(), diagnostics.begin(), diagnostics.end());
152 }
153 if (result.diagnosticsByUri.count(uri) == 0) {
154 result.diagnosticsByUri[uri] = {};
155 }
156 }
157 clearStaleDiagnostics(result);
158 return result;
159}

191void Workspace::invalidateSnapshotsForPath(const std::filesystem::path &path) {
192 const auto canonical = std::filesystem::weakly_canonical(path).string();
193 auto invalidate = [&canonical](auto &snapshots) {
194 for (auto it = snapshots.begin(); it != snapshots.end();) {
195 if (it->second->project.dependencyModuleIds.count(canonical) != 0) {
196 it = snapshots.erase(it);
197 } else {
198 ++it;
199 }
200 }
201 };
202 invalidate(strictSnapshots_);
203 invalidate(tolerantSnapshots_);

225std::shared_ptr<const SemanticSnapshot>
226Workspace::buildSnapshot(const SourceSnapshot &document,
227 bool allowEntryErrors) {
228 auto snapshot = std::make_shared<SemanticSnapshot>();
229 snapshot->documentVersion = document.version;
230 const auto configuration = projectConfigurationFor(document.path);
231 const auto importMap = configuration && configuration->configuration
232 ? configuration->configuration->importMap
233 : zap::frontend::ImportMap{};
234 if (configuration) {
235 appendConfigurationDiagnostics(snapshot->project.analysis, document,
236 configuration->errors);
237 }
238
239 zap::frontend::FrontendSession session(
240 {runtimePaths_, importMap, true, allowEntryErrors},
241 [this](const std::filesystem::path &path) -> std::optional<std::string> {
242 auto source = sourceManager_.sourceForPath(path);
243 return source ? std::optional<std::string>((*source)->text)
244 : std::nullopt;
245 });
246 auto project = session.load(document.path);
247 snapshot->project.dependencyModuleIds = std::move(project.visitedModuleIds);
248 if (!project.modules.empty()) {
249 session.bind(project);
250 }
251 appendDiagnostics(snapshot->project.analysis, project.diagnostics,
252 sourceManager_.uriForPath(document.path));
253 snapshot->project.boundRoot = std::move(project.boundRoot);
254 snapshot->project.semanticInfo = std::move(project.semanticInfo);
255 snapshot->project.moduleMap = std::move(project.modules);
256 for (const auto &[moduleId, _] : snapshot->project.moduleMap) {
257 snapshot->project.uriByModuleId[moduleId] = sourceManager_.uriForPath(moduleId);
258 }
259 return snapshot;
260}

262std::shared_ptr<const ProjectState>
263Workspace::loadProject(const std::string &uri, bool allowEntryErrors) {
264 const auto *document = this->document(uri);
265 if (!document) {
266 return nullptr;
267 }
268 auto &snapshots = allowEntryErrors ? tolerantSnapshots_ : strictSnapshots_;
269 auto snapshot = snapshots.find(uri);
270 if (snapshot != snapshots.end() &&
271 snapshot->second->documentVersion == document->version) {
272 return std::shared_ptr<const ProjectState>(snapshot->second,
273 &snapshot->second->project);
274 }
275 auto built = buildSnapshot(*document, allowEntryErrors);
276 snapshots[uri] = std::move(built);
277 return std::shared_ptr<const ProjectState>(snapshots[uri],
278 &snapshots[uri]->project);
279}
