104void Workspace::invalidateSnapshotsForPath(const std::filesystem::path &path) {
105 const auto canonical = std::filesystem::weakly_canonical(path).string();
106 auto invalidate = [&canonical](auto &snapshots) {
107 for (auto it = snapshots.begin(); it != snapshots.end();) {
108 if (it->second->project.dependencyModuleIds.count(canonical) != 0) {
109 it = snapshots.erase(it);

115 invalidate(strictSnapshots_);

119std::shared_ptr<const SemanticSnapshot>
120Workspace::buildSnapshot(const SourceSnapshot &document,
121 bool allowEntryErrors) {
122 auto snapshot = std::make_shared<SemanticSnapshot>();
123 snapshot->documentVersion = document.version;
124 auto flags = findAndReadFlags(document.path);
125
126 zap::frontend::FrontendSession session(
127 {runtimePaths_, flags.importMap, true, allowEntryErrors},
128 [this](const std::filesystem::path &path) -> std::optional<std::string> {
129 auto source = sourceManager_.sourceForPath(path);
130 return source ? std::optional<std::string>((*source)->text)
131 : std::nullopt;
132 });
133 auto project = session.load(document.path);
134 snapshot->project.dependencyModuleIds = std::move(project.visitedModuleIds);
135 if (!project.loaded) {
136 appendDiagnostics(snapshot->project.analysis, project.diagnostics,
137 sourceManager_.uriForPath(document.path));
138 return snapshot;
139 }
140 session.bind(project);
141 appendDiagnostics(snapshot->project.analysis, project.diagnostics,
142 sourceManager_.uriForPath(document.path));
143 snapshot->project.boundRoot = std::move(project.boundRoot);
144 snapshot->project.semanticInfo = std::move(project.semanticInfo);
145 snapshot->project.moduleMap = std::move(project.modules);
146 for (const auto &[moduleId, _] : snapshot->project.moduleMap) {
147 snapshot->project.uriByModuleId[moduleId] = sourceManager_.uriForPath(moduleId);
148 }
149 return snapshot;
150}

152std::shared_ptr<const ProjectState>
153Workspace::loadProject(const std::string &uri, bool allowEntryErrors) {
154 const auto *document = this->document(uri);
155 if (!document) {
156 return nullptr;
157 }
158 auto &snapshots = allowEntryErrors ? tolerantSnapshots_ : strictSnapshots_;
159 auto snapshot = snapshots.find(uri);
160 if (snapshot != snapshots.end() &&
161 snapshot->second->documentVersion == document->version) {
162 return std::shared_ptr<const ProjectState>(snapshot->second,
163 &snapshot->second->project);
164 }
165 auto built = buildSnapshot(*document, allowEntryErrors);
166 snapshots[uri] = std::move(built);
167 return std::shared_ptr<const ProjectState>(snapshots[uri],
168 &snapshots[uri]->project);
169}

181AnalysisResult Workspace::analyze(const std::string &uri) {
182 AnalysisResult result;
183 const auto *document = this->document(uri);
184 if (!document) {
185 return result;
186 }
187
188 auto project = loadProject(uri);
189 if (!project) {
190 return result;
191 }
192 result = project->analysis;
193 if (result.diagnosticsByUri.find(uri) == result.diagnosticsByUri.end()) {
194 result.diagnosticsByUri[uri] = {};
195 }
196 clearStaleDiagnostics(result);
197 return result;
