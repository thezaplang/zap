47void LLVMCodeGen::emitManagedRetain(
48 llvm::Value *value, const std::shared_ptr<zir::Type> &type) {
49 if (!value || !type) {
50 return;
51 }
52 if (isOwnedStringType(type)) {
53 (void)emitStringRetainIfNeeded(value, type);
54 return;
55 }
56 if (isWeakClassType(type)) {
57 emitRetainWeakIfNeeded(value, type);
58 return;
59 }
60 if (isClassType(type)) {
61 emitRetainIfNeeded(value, type);
62 return;
63 }
64 if (type->getKind() == zir::TypeKind::Record) {
65 const auto &record = static_cast<const zir::RecordType &>(*type);
66 for (size_t i = 0; i < record.getFields().size(); ++i) {
67 const auto &field = record.getFields()[i];
68 if (containsManagedValues(field.type)) {
69 emitManagedRetain(
70 builder_.CreateExtractValue(value, {static_cast<unsigned>(i)}),
71 field.type);
72 }
73 }
74 } else if (type->getKind() == zir::TypeKind::Array) {
75 const auto &array = static_cast<const zir::ArrayType &>(*type);
76 if (!containsManagedValues(array.getBaseType())) {
77 return;
78 }
79 for (size_t i = 0; i < array.getSize(); ++i) {
80 emitManagedRetain(builder_.CreateExtractValue(
81 value, {static_cast<unsigned>(i)}),
82 array.getBaseType());
83 }
84 } else if (type->getKind() == zir::TypeKind::TaggedUnion) {
85 emitManagedForActiveTaggedUnion(
86 value, std::static_pointer_cast<zir::TaggedUnionType>(type),
87 &LLVMCodeGen::emitManagedRetain);
88 }
89}

91void LLVMCodeGen::emitManagedRelease(
92 llvm::Value *value, const std::shared_ptr<zir::Type> &type) {
93 if (!value || !type) {
94 return;
95 }
96 if (isOwnedStringType(type)) {
97 emitStringReleaseIfNeeded(value, type);
98 return;
99 }
100 if (isWeakClassType(type)) {
101 emitReleaseWeakIfNeeded(value, type);
102 return;
103 }
104 if (isClassType(type)) {
105 emitReleaseIfNeeded(value, type);
106 return;
107 }
108 if (type->getKind() == zir::TypeKind::Record) {
109 const auto &record = static_cast<const zir::RecordType &>(*type);
110 for (size_t i = record.getFields().size(); i > 0; --i) {
111 const auto &field = record.getFields()[i - 1];
112 if (containsManagedValues(field.type)) {
113 emitManagedRelease(builder_.CreateExtractValue(
114 value, {static_cast<unsigned>(i - 1)}),
115 field.type);
116 }
117 }
118 } else if (type->getKind() == zir::TypeKind::Array) {
119 const auto &array = static_cast<const zir::ArrayType &>(*type);
120 if (!containsManagedValues(array.getBaseType())) {
121 return;
122 }
123 for (size_t i = array.getSize(); i > 0; --i) {
124 emitManagedRelease(builder_.CreateExtractValue(
125 value, {static_cast<unsigned>(i - 1)}),
126 array.getBaseType());
127 }
128 } else if (type->getKind() == zir::TypeKind::TaggedUnion) {
129 emitManagedForActiveTaggedUnion(
130 value, std::static_pointer_cast<zir::TaggedUnionType>(type),
131 &LLVMCodeGen::emitManagedRelease);
132 }

135void LLVMCodeGen::emitManagedForActiveTaggedUnion(
136 llvm::Value *value, const std::shared_ptr<zir::TaggedUnionType> &type,
137 void (LLVMCodeGen::*operation)(llvm::Value *,
138 const std::shared_ptr<zir::Type> &)) {
139 if (!value || !type || !containsManagedValues(type)) {
140 return;
141 }
142
143 auto *unionTy = llvm::cast<llvm::StructType>(toLLVMType(*type));
144 auto *storage = createEntryAlloca(currentFn_, "arc.union.addr", unionTy);
145 builder_.CreateStore(value, storage);
146 auto *tagAddr =
147 builder_.CreateStructGEP(unionTy, storage, 0, "arc.union.tag.addr");
148 auto *tag = builder_.CreateLoad(llvm::Type::getInt32Ty(ctx_), tagAddr,
149 "arc.union.tag");
150 auto *done = llvm::BasicBlock::Create(ctx_, "arc.union.done", currentFn_);
151
152 for (const auto &variant : type->getVariants()) {
153 if (!containsManagedValues(variant.payloadType)) {
154 continue;
155 }
156
157 auto *active =
158 llvm::BasicBlock::Create(ctx_, "arc.union.active", currentFn_);
159 auto *next = llvm::BasicBlock::Create(ctx_, "arc.union.next", currentFn_);
160 auto *isActive = builder_.CreateICmpEQ(
161 tag, llvm::ConstantInt::getSigned(llvm::Type::getInt32Ty(ctx_),
162 variant.tag),
163 "arc.union.is_active");
164 builder_.CreateCondBr(isActive, active, next);
165
166 builder_.SetInsertPoint(active);
167 auto *payloadAddr = builder_.CreateStructGEP(
168 unionTy, storage, 1, "arc.union.payload.addr");
169 auto *payload = builder_.CreateLoad(toLLVMType(*variant.payloadType),
170 payloadAddr, "arc.union.payload");
171 (this->*operation)(payload, variant.payloadType);
172 builder_.CreateBr(done);
173
174 builder_.SetInsertPoint(next);
175 }
176
177 builder_.CreateBr(done);
178 builder_.SetInsertPoint(done);
179}

195void LLVMCodeGen::emitArcCollectionSafePoint() {
196 // ZIR lowering calls this only after local releases on a non-destructor
197 // return, so collection cannot reenter through an active destructor.
198 auto *rawPtrTy = llvm::PointerType::getUnqual(ctx_);
199 auto contextIt = functionMap_.find("zap_arc_default_context");
200 if (contextIt == functionMap_.end()) {
201 auto *contextTy = llvm::FunctionType::get(rawPtrTy, {}, false);
202 auto *contextFn = llvm::Function::Create(
203 contextTy, llvm::Function::ExternalLinkage, "zap_arc_default_context",
204 *module_);
205 contextIt = functionMap_.emplace("zap_arc_default_context", contextFn)
206 .first;
207 }
208 auto it = functionMap_.find("zap_arc_collect_at_safepoint");
209 if (it == functionMap_.end()) {
210 auto *safePointTy = llvm::FunctionType::get(
211 llvm::Type::getVoidTy(ctx_), {rawPtrTy}, false);
212 auto *safePointFn = llvm::Function::Create(
213 safePointTy, llvm::Function::ExternalLinkage,
214 "zap_arc_collect_at_safepoint", *module_);
215 it = functionMap_.emplace("zap_arc_collect_at_safepoint", safePointFn)
216 .first;
217 }
218 auto *context = builder_.CreateCall(contextIt->second, {}, "arc.context");
219 builder_.CreateCall(it->second, {context});

241LLVMCodeGen::emitStringRetainIfNeeded(llvm::Value *value,
242 const std::shared_ptr<zir::Type> &type) {
243 if (!isOwnedStringType(type)) {
244 return value;
245 }
246 auto *stringTy = toLLVMType(*type);
247 if (value->getType() != stringTy && isStringLikeStruct(value->getType()) &&
248 isStringLikeStruct(stringTy)) {
249 auto *ptr = builder_.CreateExtractValue(value, {0}, "str.cvt.ptr");
250 auto *len = builder_.CreateExtractValue(value, {1}, "str.cvt.len");
251 llvm::Value *converted = llvm::UndefValue::get(stringTy);
252 converted =
253 builder_.CreateInsertValue(converted, ptr, {0}, "str.cvt.ptr.i");
254 converted =
255 builder_.CreateInsertValue(converted, len, {1}, "str.cvt.len.i");
256 value = converted;
257 }
258 auto *fnTy = llvm::FunctionType::get(stringTy, {stringTy}, false);
259 auto callee = module_->getOrInsertFunction("zap_string_retain", fnTy);
260 return builder_.CreateCall(fnTy, callee.getCallee(), {value}, "str.retain");
261}
