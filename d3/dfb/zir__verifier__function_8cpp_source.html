35 void verify() {
36 verifySignature();
37 if (function_.getBlocks().empty()) {
38 error(VerificationErrorCode::MissingBody, nullptr, std::nullopt,
39 "defined function has no basic blocks");

75 void verifySignature() {
76 if (!function_.getReturnType()) {
77 error(VerificationErrorCode::NullNode, nullptr, std::nullopt,
78 "function has no return type");
79 }
80 for (const auto &argument : function_.getArguments()) {
81 if (!argument || !argument->getType()) {
82 error(VerificationErrorCode::NullNode, nullptr, std::nullopt,
83 "function has a null argument or argument type");
84 continue;
85 }
86 if (!valueNames_.emplace(argument->getName(), argument.get()).second) {
87 error(VerificationErrorCode::DuplicateValue, nullptr, std::nullopt,
88 "duplicate argument name " + argument->getName());
89 }
90 if (argument->getParameterEscape() == ParameterEscape::NoEscape) {
91 if (argument->getParameterOwnership() != ParameterOwnership::Borrow) {
92 error(VerificationErrorCode::InvalidOperand, nullptr, std::nullopt,
93 "noescape parameter must use borrowed ownership: " +
94 argument->getName());
95 }
96 if (argument->getType()->getIntrinsicKind() !=
97 IntrinsicTypeKind::StringView) {
98 error(VerificationErrorCode::InvalidOperand, nullptr, std::nullopt,
99 "noescape currently requires a StringView parameter: " +
100 argument->getName());
101 }
102 }
103 }

107 for (const auto &block : function_.getBlocks()) {

120 void collectDefinitionsAndEdges() {
121 for (const auto &blockOwner : function_.getBlocks()) {
122 if (!blockOwner) {
123 continue;
124 }
125 const auto &block = *blockOwner;
126 bool sawTerminator = false;
127 for (size_t i = 0; i < block.getInstructions().size(); ++i) {
128 const auto &instruction = block.getInstructions()[i];
129 if (!instruction) {
130 error(VerificationErrorCode::NullNode, &block, i,
131 "basic block contains a null instruction");
132 continue;
133 }
134 if (sawTerminator) {
135 error(VerificationErrorCode::InstructionAfterTerminator, &block, i,
136 "instruction appears after a terminator");
137 }
138 sawTerminator = sawTerminator || isTerminator(instruction->getOpCode());
139
140 if (auto result = instructionResult(*instruction)) {
141 if (!result->getType()) {
142 error(VerificationErrorCode::NullNode, &block, i,
143 "instruction result has no type");
144 }
145 if (!definitions_.emplace(result.get(), DefinitionSite{&block, i})
146 .second) {
147 error(VerificationErrorCode::DuplicateValue, &block, i,
148 "value is defined more than once: " + result->getName());
149 }
150 auto [_, inserted] =
151 valueNames_.emplace(result->getName(), result.get());
152 if (!inserted) {
153 error(VerificationErrorCode::DuplicateValue, &block, i,
154 "duplicate value name " + result->getName());
155 }
156 }
157
158 if (instruction->getOpCode() == OpCode::Br) {
159 const auto &branch = static_cast<const BranchInst &>(*instruction);
160 if (!cfg_.findBlock(branch.getTarget())) {
161 error(VerificationErrorCode::InvalidBranchTarget, &block, i,
162 "branch targets unknown block " + branch.getTarget());
163 }
164 } else if (instruction->getOpCode() == OpCode::CondBr) {
165 const auto &branch =
166 static_cast<const CondBranchInst &>(*instruction);
167 for (const auto &label :
168 {branch.getTrueLabel(), branch.getFalseLabel()}) {
169 if (!cfg_.findBlock(label)) {
170 error(VerificationErrorCode::InvalidBranchTarget, &block, i,
171 "branch targets unknown block " + label);
172 }
173 }
174 }
175 }
176 if (!sawTerminator) {
177 error(VerificationErrorCode::MissingTerminator, &block, std::nullopt,
178 "basic block has no terminator");
179 }
180 }

183 bool verifyValue(const std::shared_ptr<Value> &value,
184 const BasicBlock &useBlock, size_t useIndex,
185 const BasicBlock *edgePredecessor = nullptr) {
186 if (!value || !value->getType()) {
187 error(VerificationErrorCode::NullNode, &useBlock, useIndex,
188 "instruction contains a null value or value type");
189 return false;
190 }
191 if (value->getKind() != ValueKind::Register) {
192 return true;
193 }
194
195 auto definition = definitions_.find(value.get());
196 if (definition == definitions_.end()) {
197 error(VerificationErrorCode::UndefinedValue, &useBlock, useIndex,
198 "use of undefined value " + value->getName());
199 return false;
200 }
201
202 const auto *effectiveBlock = edgePredecessor ? edgePredecessor : &useBlock;
203 if (definition->second.block == effectiveBlock) {
204 const size_t effectiveIndex =
205 edgePredecessor ? effectiveBlock->getInstructions().size() : useIndex;
206 if (definition->second.instructionIndex >= effectiveIndex) {
207 error(VerificationErrorCode::UseBeforeDefinition, &useBlock, useIndex,
208 "value is used before its definition: " + value->getName());
209 return false;
210 }
211 return true;
212 }
213
214 if (cfg_.isReachable(*effectiveBlock) &&
215 !cfg_.dominates(*definition->second.block, *effectiveBlock)) {
216 error(VerificationErrorCode::DominanceViolation, &useBlock, useIndex,
217 "definition of " + value->getName() + " does not dominate its use");
218 return false;
219 }
220 return true;
221 }

275 void verifyOwnershipTransfers() {
276 OwnershipFlowAnalysis analysis(module_, function_, cfg_.predecessors(),
277 cfg_.successors(), cfg_.reachable());
278 for (const auto &violation : analysis.analyze()) {
279 error(VerificationErrorCode::OwnershipViolation, violation.block,
280 violation.instructionIndex,
281 "owned value is not live for " + violation.operation + ": " +
282 violation.value->getName() + " (state: " +
283 formatOwnershipFlowState(violation.priorState) + ")");
284 }
285 if (includeOwnershipObligations_) {
286 verifyOwnershipClosurePlans(analysis);
287 }
288
289 const auto liveness = analyzeOwnershipLiveness(module_, function_, cfg_);
290 for (const auto &blockOwner : function_.getBlocks()) {
291 if (!blockOwner) {
292 continue;
293 }
294 const auto &block = *blockOwner;
295 for (size_t i = 0; i < block.getInstructions().size(); ++i) {
296 const auto &instruction = block.getInstructions()[i];
297 if (!instruction || instruction->getOpCode() != OpCode::Destroy) {
298 continue;
299 }
300 const auto &value =
301 static_cast<const DestroyInst &>(*instruction).getValue();
302 if (liveness.isLiveAfter(block, i, value)) {
303 error(VerificationErrorCode::OwnershipViolation, &block, i,
304 "cannot destroy a String owner while one of its borrowed "
305 "StringViews is still live");
306 }
307 }
308 }

311 void verifyOwnershipClosurePlans(OwnershipFlowAnalysis &analysis) {
312 for (const auto &plan : analysis.analyzeOwnershipClosurePlans()) {
313 std::string definition = "function argument";
314 if (plan.definition.block) {
315 definition = "block %" + plan.definition.block->label;
316 if (plan.definition.instructionIndex) {
317 definition += " instruction " +
318 std::to_string(*plan.definition.instructionIndex);
319 }
320 }
321 for (const auto &obligation : plan.liveExits) {
322 std::string placement = "no safe destroy placement determined";
323 for (const auto &candidate : plan.destroyPlacements) {
324 if (candidate.destination != obligation.block ||
325 (candidate.kind == OwnershipDestroyPlacementKind::BeforeReturn &&
326 candidate.instructionIndex != obligation.instructionIndex)) {
327 continue;
328 }
329 if (candidate.kind == OwnershipDestroyPlacementKind::BeforeReturn) {
330 placement = "suggested destroy before return";
331 } else {
332 placement = "suggested destroy on edge %" +
333 candidate.source->label + " -> %" +
334 candidate.destination->label;
335 if (candidate.requiresEdgeSplit) {
336 placement += " (split critical edge)";
337 }
338 }
339 break;
340 }
341 error(VerificationErrorCode::OwnershipViolation, obligation.block,
342 obligation.instructionIndex,
343 "owned value may remain live at function exit: " +
344 plan.value->getName() + " (defined in " + definition +
345 "; state: " + formatOwnershipFlowState(obligation.state) +
346 "; " + placement + ")");
347 }
348 }

351 void verifyInstruction(const Instruction &instruction,
352 const BasicBlock &block, size_t index) {
353 switch (instruction.getOpCode()) {
354 case OpCode::Alloca: {
355 const auto &alloca = static_cast<const AllocaInst &>(instruction);
356 if (!alloca.getResult() || !alloca.getAllocatedType()) {
357 error(VerificationErrorCode::InvalidResult, &block, index,
358 "alloca requires a result and allocated type");
359 return;
360 }
361 auto pointer =
362 std::dynamic_pointer_cast<PointerType>(alloca.getResult()->getType());
363 if (!pointer || !typeInterner_.same(pointer->getBaseType(),
364 alloca.getAllocatedType())) {
365 error(VerificationErrorCode::TypeMismatch, &block, index,
366 "alloca result must point to the allocated type");
367 }
368 return;
369 }
370 case OpCode::Load: {
371 const auto &load = static_cast<const LoadInst &>(instruction);
372 verifyValue(load.getSource(), block, index);
373 auto pointer = load.getSource() ? std::dynamic_pointer_cast<PointerType>(
374 load.getSource()->getType())
375 : nullptr;
376 if (!pointer || !load.getResult()) {
377 error(VerificationErrorCode::InvalidOperand, &block, index,
378 "load requires a pointer source and result");
379 } else {
380 expectSameType(load.getResult()->getType(), pointer->getBaseType(),
381 block, index, "load result type");
382 }
383 return;
384 }
385 case OpCode::Store: {
386 const auto &store = static_cast<const StoreInst &>(instruction);
387 verifyValue(store.getSource(), block, index);
388 verifyValue(store.getDestination(), block, index);
389 auto pointer = store.getDestination()
390 ? std::dynamic_pointer_cast<PointerType>(
391 store.getDestination()->getType())
392 : nullptr;
393 if (!pointer || !store.getSource()) {
394 error(VerificationErrorCode::InvalidOperand, &block, index,
395 "store requires a value and pointer destination");
396 } else {
397 expectAssignable(store.getSource()->getType(), pointer->getBaseType(),
398 block, index, "stored value type");
399 if (store.getMode() == StoreMode::Initialize &&
400 containsManagedValues(store.getSource()->getType()) &&
401 !isOwned(store.getSource()->getOwnership())) {
402 error(VerificationErrorCode::InvalidOperand, &block, index,
403 "managed initialization must transfer ownership");
404 }
405 }
406 return;
407 }
408 case OpCode::Add:
409 case OpCode::Sub:
410 case OpCode::Mul:
411 case OpCode::SDiv:
412 case OpCode::UDiv:
413 case OpCode::SRem:
414 case OpCode::URem:
415 case OpCode::Shl:
416 case OpCode::LShr:
417 case OpCode::AShr:
418 case OpCode::BitAnd:
419 case OpCode::BitOr:
420 case OpCode::BitXor: {
421 const auto &binary = static_cast<const BinaryInst &>(instruction);
422 verifyValue(binary.getLhs(), block, index);
423 verifyValue(binary.getRhs(), block, index);
424 if (!binary.getResult() || !binary.getLhs() || !binary.getRhs()) {
425 error(VerificationErrorCode::InvalidOperand, &block, index,
426 "binary instruction requires two operands and a result");
427 return;
428 }
429
430 const auto lhsType = binary.getLhs()->getType();
431 const auto rhsType = binary.getRhs()->getType();
432 const bool isAdd = instruction.getOpCode() == OpCode::Add;
433 const bool isSubtract = instruction.getOpCode() == OpCode::Sub;
434 const bool isStringConcat =
435 isAdd && (isStringType(lhsType) || isStringType(rhsType) ||
436 lhsType->getKind() == TypeKind::Char ||
437 rhsType->getKind() == TypeKind::Char);
438 const bool lhsPointer = lhsType->getKind() == TypeKind::Pointer;
439 const bool rhsPointer = rhsType->getKind() == TypeKind::Pointer;
440
441 if (isStringConcat) {
442 if ((!isStringType(lhsType) && lhsType->getKind() != TypeKind::Char) ||
443 (!isStringType(rhsType) && rhsType->getKind() != TypeKind::Char) ||
444 binary.getResult()->getType()->getIntrinsicKind() !=
445 IntrinsicTypeKind::String) {
446 error(VerificationErrorCode::TypeMismatch, &block, index,
447 "string concatenation requires String/StringView/Char "
448 "operands and a String result");
449 }
450 } else if (isAdd && lhsPointer != rhsPointer) {
451 const auto offsetType = lhsPointer ? rhsType : lhsType;
452 const auto pointerType = lhsPointer ? lhsType : rhsType;
453 if (!offsetType->isInteger()) {
454 error(VerificationErrorCode::TypeMismatch, &block, index,
455 "pointer offset must be an integer");
456 }
457 expectSameType(binary.getResult()->getType(), pointerType, block, index,
458 "pointer addition result type");
459 } else if (isSubtract && lhsPointer && rhsPointer) {
460 expectSameType(lhsType, rhsType, block, index,
461 "pointer subtraction operand type");
462 if (!binary.getResult()->getType()->isInteger()) {
463 error(VerificationErrorCode::TypeMismatch, &block, index,
464 "pointer subtraction result must be an integer");
465 }
466 } else if (isSubtract && lhsPointer && !rhsPointer) {
467 if (!rhsType->isInteger()) {
468 error(VerificationErrorCode::TypeMismatch, &block, index,
469 "pointer offset must be an integer");
470 }
471 expectSameType(binary.getResult()->getType(), lhsType, block, index,
472 "pointer subtraction result type");
473 } else {
474 expectSameType(rhsType, lhsType, block, index, "binary operand type");
475 expectSameType(binary.getResult()->getType(), lhsType, block, index,
476 "binary result type");
477 }
478 return;
479 }
480 case OpCode::Cmp: {
481 const auto &compare = static_cast<const CmpInst &>(instruction);
482 verifyValue(compare.getLhs(), block, index);
483 verifyValue(compare.getRhs(), block, index);
484 if (compare.getLhs() && compare.getRhs()) {
485 expectSameType(compare.getRhs()->getType(), compare.getLhs()->getType(),
486 block, index, "comparison operand type");
487 }
488 if (!compare.getResult() ||
489 compare.getResult()->getType()->getKind() != TypeKind::Bool) {
490 error(VerificationErrorCode::InvalidResult, &block, index,
491 "comparison result must be Bool");
492 }
493 return;
494 }
495 case OpCode::Br:
496 return;
497 case OpCode::CondBr: {
498 const auto &branch = static_cast<const CondBranchInst &>(instruction);
499 verifyValue(branch.getCondition(), block, index);
500 if (!branch.getCondition() ||
501 branch.getCondition()->getType()->getKind() != TypeKind::Bool) {
502 error(VerificationErrorCode::InvalidConditionType, &block, index,
503 "conditional branch requires a Bool condition");
504 }
505 return;
506 }
507 case OpCode::Ret: {
508 const auto &returnInstruction =
509 static_cast<const ReturnInst &>(instruction);
510 if (returnInstruction.getValue()) {
511 verifyValue(returnInstruction.getValue(), block, index);
512 }
513 const bool returnsVoid =
514 function_.getReturnType() &&
515 function_.getReturnType()->getKind() == TypeKind::Void;
516 if (returnsVoid && returnInstruction.getValue()) {
517 error(VerificationErrorCode::InvalidReturn, &block, index,
518 "void function returns a value");
519 } else if (!returnsVoid && !returnInstruction.getValue()) {
520 error(VerificationErrorCode::InvalidReturn, &block, index,
521 "non-void function returns no value");
522 } else if (returnInstruction.getValue()) {
523 auto expectedReturnType = function_.getReturnType();
524 if (function_.returnsRef) {
525 expectedReturnType =
526 std::make_shared<PointerType>(expectedReturnType);
527 }
528 expectAssignable(returnInstruction.getValue()->getType(),
529 expectedReturnType, block, index, "return value type");
530 if (function_.returnsRef &&
531 isOwned(returnInstruction.getValue()->getOwnership())) {
532 error(VerificationErrorCode::InvalidReturn, &block, index,
533 "ref return cannot transfer ownership");
534 }
535 }
536 return;
537 }
538 case OpCode::Call:
539 verifyCall(static_cast<const CallInst &>(instruction), block, index);
540 return;
541 case OpCode::Copy: {
542 const auto &copy = static_cast<const CopyInst &>(instruction);
543 verifyValue(copy.getSource(), block, index);
544 if (!copy.getResult() || !copy.getSource() ||
545 !containsManagedValues(copy.getSource()->getType()) ||
546 copy.getResult() == copy.getSource() ||
547 copy.getResult()->getOwnership() !=
548 ownedForType(copy.getSource()->getType())) {
549 error(VerificationErrorCode::InvalidOperand, &block, index,
550 "copy requires a distinct owned result and managed source");
551 } else {
552 expectSameType(copy.getResult()->getType(), copy.getSource()->getType(),
553 block, index, "copy result type");
554 }
555 return;
556 }
557 case OpCode::Move: {
558 const auto &move = static_cast<const MoveInst &>(instruction);
559 verifyValue(move.getSource(), block, index);
560 if (!move.getResult() || !ownsManagedValue(move.getSource()) ||
561 move.getResult() == move.getSource() ||
562 move.getResult()->getOwnership() !=
563 move.getSource()->getOwnership()) {
564 error(VerificationErrorCode::InvalidOperand, &block, index,
565 "move requires a distinct owned result and owned managed source");
566 } else {
567 expectSameType(move.getResult()->getType(), move.getSource()->getType(),
568 block, index, "move result type");
569 }
570 return;
571 }
572 case OpCode::Borrow: {
573 const auto &borrow = static_cast<const BorrowInst &>(instruction);
574 verifyValue(borrow.getOwner(), block, index);
575 if (!borrow.getResult() || !borrow.getOwner() ||
576 borrow.getOwner()->getType()->getIntrinsicKind() !=
577 IntrinsicTypeKind::String ||
578 borrow.getResult()->getType()->getIntrinsicKind() !=
579 IntrinsicTypeKind::StringView ||
580 borrow.getResult()->getOwnership() != ValueOwnership::Borrowed) {
581 error(
582 VerificationErrorCode::InvalidOperand, &block, index,
583 "borrow requires a String owner and a borrowed StringView result");
584 }
585 return;
586 }
587 case OpCode::Destroy: {
588 const auto &destroy = static_cast<const DestroyInst &>(instruction);
589 verifyValue(destroy.getValue(), block, index);
590 if (!ownsManagedValue(destroy.getValue())) {
591 error(VerificationErrorCode::InvalidOperand, &block, index,
592 "destroy requires an owned managed value");
593 }
594 return;
595 }
596 case OpCode::Alloc: {
597 const auto &alloc = static_cast<const AllocInst &>(instruction);
598 if (!alloc.getResult() || !alloc.getAllocatedType()) {
599 error(VerificationErrorCode::InvalidResult, &block, index,
600 "alloc requires a result and allocated type");
601 } else {
602 expectSameType(alloc.getResult()->getType(), alloc.getAllocatedType(),
603 block, index, "alloc result type");
604 if (alloc.getResult()->getOwnership() != ValueOwnership::OwnedStrong) {
605 error(VerificationErrorCode::InvalidResult, &block, index,
606 "alloc result must be owned");
607 }
608 }
609 return;
610 }
611 case OpCode::GetElementPtr:
612 verifyGetElementPtr(static_cast<const GetElementPtrInst &>(instruction),
613 block, index);
614 return;
615 case OpCode::Phi:
616 verifyPhi(static_cast<const PhiInst &>(instruction), block, index);
617 return;
618 case OpCode::Cast: {
619 const auto &cast = static_cast<const CastInst &>(instruction);
620 verifyValue(cast.getSource(), block, index);
621 if (!cast.getResult() || !cast.getTargetType() || !cast.getSource()) {
622 error(VerificationErrorCode::InvalidResult, &block, index,
623 "cast requires a source, result, and target type");
624 } else {
625 expectSameType(cast.getResult()->getType(), cast.getTargetType(), block,
626 index, "cast result type");
627 const auto expectedOwnership =
628 cast.getTargetType()->getIntrinsicKind() ==
629 IntrinsicTypeKind::String
630 ? ValueOwnership::OwnedStrong
631 : containsManagedValues(cast.getTargetType()) &&
632 isOwned(cast.getSource()->getOwnership())
633 ? cast.getSource()->getOwnership()
634 : ValueOwnership::Borrowed;
635 if (cast.getResult()->getOwnership() != expectedOwnership) {
636 error(VerificationErrorCode::InvalidResult, &block, index,
637 "cast result ownership does not match source and target");
638 }
639 }
640 return;
641 }
642 case OpCode::WeakLock: {
643 const auto &weakLock = static_cast<const WeakLockInst &>(instruction);
644 verifyValue(weakLock.getWeakValue(), block, index);
645 if (!weakLock.getResult() || !weakLock.getWeakValue() ||
646 weakLock.getResult()->getType()->getKind() != TypeKind::Class ||
647 weakLock.getWeakValue()->getType()->getKind() != TypeKind::Class) {
648 error(VerificationErrorCode::InvalidOperand, &block, index,
649 "weak.lock requires class operands");
650 } else {
651 const auto weakType = std::static_pointer_cast<ClassType>(
652 weakLock.getWeakValue()->getType());
653 const auto resultType = std::static_pointer_cast<ClassType>(
654 weakLock.getResult()->getType());
655 if (!weakType->isWeak() || resultType->isWeak()) {
656 error(VerificationErrorCode::InvalidOperand, &block, index,
657 "weak.lock requires a weak source and strong result");
658 } else {
659 auto expectedResultType = std::make_shared<ClassType>(*weakType);
660 expectedResultType->setWeak(false);
661 expectSameType(weakLock.getResult()->getType(), expectedResultType,
662 block, index, "weak.lock result type");
663 }
664 if (weakLock.getResult()->getOwnership() !=
665 ValueOwnership::OwnedStrong) {
666 error(VerificationErrorCode::InvalidResult, &block, index,
667 "weak.lock result must be owned");
668 }
669 }
670 return;
671 }
672 case OpCode::WeakAlive: {
673 const auto &weakAlive = static_cast<const WeakAliveInst &>(instruction);
674 verifyValue(weakAlive.getWeakValue(), block, index);
675 if (!weakAlive.getResult() ||
676 weakAlive.getResult()->getType()->getKind() != TypeKind::Bool) {
677 error(VerificationErrorCode::InvalidResult, &block, index,
678 "weak.alive result must be Bool");
679 }
680 return;
681 }
682 case OpCode::ClassIs: {
683 const auto &classIs = static_cast<const ClassIsInst &>(instruction);
684 verifyValue(classIs.getObject(), block, index);
685 if (!classIs.getResult() || !classIs.getObject() ||
686 !classIs.getTargetType() ||
687 classIs.getResult()->getType()->getKind() != TypeKind::Bool ||
688 classIs.getObject()->getType()->getKind() != TypeKind::Class) {
689 error(VerificationErrorCode::InvalidOperand, &block, index,
690 "classis requires a class operand and Bool result");
691 }
692 return;
693 }
694 case OpCode::InlineAsm: {
695 const auto &assembly = static_cast<const InlineAsmInst &>(instruction);
696 for (const auto &operand : assembly.getOutputs()) {
697 verifyValue(operand.value, block, index);
698 }
699 for (const auto &operand : assembly.getInputs()) {
700 verifyValue(operand.value, block, index);
701 }
702 return;
703 }
704 }
705 }

707 void verifyCall(const CallInst &call, const BasicBlock &block, size_t index) {
708 for (const auto &argument : call.getArguments()) {
709 verifyValue(argument, block, index);
710 }
711 if (call.getVariadicPack()) {
712 verifyValue(call.getVariadicPack(), block, index);
713 }
714
715 std::vector<std::shared_ptr<Type>> parameterTypes;
716 std::vector<ParameterOwnership> parameterOwnership;
717 ResultBorrowContract resultBorrow;
718 std::shared_ptr<Type> returnType;
719 const bool returnsRef = callReturnsRef(module_, call);
720 bool variadic = false;
721 if (call.isIndirect()) {
722 verifyValue(call.getCalleeValue(), block, index);
723 auto functionType = call.getCalleeValue()
724 ? std::dynamic_pointer_cast<FunctionPointerType>(
725 call.getCalleeValue()->getType())
726 : nullptr;
727 if (!functionType) {
728 error(VerificationErrorCode::InvalidCall, &block, index,
729 "indirect call requires a function pointer");
730 return;
731 }
732 parameterTypes = functionType->getParams();
733 parameterOwnership = functionType->getParameterOwnership();
734 resultBorrow = functionType->getResultBorrow();
735 returnType = functionType->getReturnType();
736 } else {
737 const auto *callee = module_.findFunction(call.getFunctionName());
738 if (!callee) {
739 error(VerificationErrorCode::InvalidCall, &block, index,
740 "call targets unknown function " + call.getFunctionName());
741 return;
742 }
743 for (const auto &argument : callee->getArguments()) {
744 if (argument && !argument->isVariadicPack()) {
745 parameterTypes.push_back(argument->getType());
746 parameterOwnership.push_back(argument->getParameterOwnership());
747 }
748 }
749 returnType = callee->getReturnType();
750 resultBorrow = callee->resultBorrow;
751 variadic = callee->isCVariadic ||
752 (!callee->getArguments().empty() &&
753 callee->getArguments().back()->isVariadicPack());
754 }
755
756 if ((!variadic && call.getArguments().size() != parameterTypes.size()) ||
757 (variadic && call.getArguments().size() < parameterTypes.size())) {
758 error(VerificationErrorCode::InvalidCall, &block, index,
759 "call argument count does not match the callee signature");
760 }
761 const size_t checkedArguments =
762 std::min(call.getArguments().size(), parameterTypes.size());
763 for (size_t i = 0; i < checkedArguments; ++i) {
764 if (call.getArguments()[i]) {
765 expectAssignable(call.getArguments()[i]->getType(), parameterTypes[i],
766 block, index,
767 "call argument " + std::to_string(i) + " type");
768 }
769 }
770 if (call.getResult() && returnType) {
771 auto expectedReturnType = returnType;
772 if (returnsRef) {
773 expectedReturnType = std::make_shared<PointerType>(returnType);
774 }
775 expectSameType(call.getResult()->getType(), expectedReturnType, block,
776 index, "call result type");
777 const auto expectedOwnership =
778 !returnsRef && containsManagedValues(returnType)
779 ? ownedForType(returnType)
780 : ValueOwnership::Borrowed;
781 if (call.getResult()->getOwnership() != expectedOwnership) {
782 error(VerificationErrorCode::InvalidResult, &block, index,
783 "call result ownership does not match its return type");
784 }
785 }
786 if (!call.getArgumentIsRef().empty() &&
787 call.getArgumentIsRef().size() != call.getArguments().size()) {
788 error(VerificationErrorCode::InvalidCall, &block, index,
789 "call ref-argument metadata has the wrong size");
790 }
791 if (resultBorrow.hasSource() &&
792 *resultBorrow.sourceParameter() >= call.getArguments().size()) {
793 error(VerificationErrorCode::InvalidCall, &block, index,
794 "call result borrow source argument is out of range");
795 }
796 for (size_t i = 0; i < call.getArguments().size(); ++i) {
797 const auto &argument = call.getArguments()[i];
798 const bool transfers = i < parameterOwnership.size() &&
799 transfersOwnership(parameterOwnership[i]);
800 if (transfers && !ownsManagedValue(argument)) {
801 error(VerificationErrorCode::InvalidCall, &block, index,
802 "call transfer argument must be owned: " + std::to_string(i));
803 }
804 if (i < call.getArgumentIsRef().size() && call.getArgumentIsRef()[i] &&
805 transfers) {
806 error(VerificationErrorCode::InvalidCall, &block, index,
807 "ref call argument cannot transfer ownership");
808 }
809 if (transfers && isBorrowedMethodSelf(call, i)) {
810 error(VerificationErrorCode::InvalidCall, &block, index,
811 "method self argument cannot transfer ownership");
812 }
813 }
814 }

817 const BasicBlock &block, size_t index) {

819 if (gep.getIndexValue()) {
820 verifyValue(gep.getIndexValue(), block, index);
821 if (!gep.getIndexValue()->getType()->isInteger()) {
822 error(VerificationErrorCode::InvalidOperand, &block, index,
823 "dynamic getelementptr index must be an integer");
824 }

827 error(VerificationErrorCode::InvalidResult, &block, index,
828 "getelementptr requires pointer and result operands");
829 return;
830 }

832 error(VerificationErrorCode::TypeMismatch, &block, index,
833 "getelementptr result must be a pointer");

837 void verifyPhi(const PhiInst &phi, const BasicBlock &block, size_t index) {
838 if (!phi.getResult()) {
839 error(VerificationErrorCode::InvalidResult, &block, index,
840 "phi requires a result");
841 return;
842 }
843 std::unordered_set<const BasicBlock *> incomingBlocks;
844 for (const auto &[label, value] : phi.getIncoming()) {
845 const auto *incomingBlock = cfg_.findBlock(label);
846 const auto &predecessors = cfg_.predecessors().at(&block);
847 if (!incomingBlock || std::find(predecessors.begin(), predecessors.end(),
848 incomingBlock) == predecessors.end()) {
849 error(VerificationErrorCode::InvalidPhi, &block, index,
850 "phi incoming label is not a predecessor: " + label);
851 continue;
852 }
853 if (!incomingBlocks.insert(incomingBlock).second) {
854 error(VerificationErrorCode::InvalidPhi, &block, index,
855 "phi contains duplicate incoming block " + label);
856 }
857 if (value) {
858 verifyValue(value, block, index, incomingBlock);
859 expectSameType(value->getType(), phi.getResult()->getType(), block,
860 index, "phi incoming value type");
861 }
862 }
863 std::unordered_set<const BasicBlock *> expectedPredecessors(
864 cfg_.predecessors().at(&block).begin(),
865 cfg_.predecessors().at(&block).end());
866 if (incomingBlocks != expectedPredecessors) {
867 error(VerificationErrorCode::InvalidPhi, &block, index,
868 "phi must have one incoming value for every predecessor");
869 }
870
871 ValueOwnership expectedOwnership = ValueOwnership::Borrowed;
872 if (containsManagedValues(phi.getResult()->getType()) &&
873 !phi.getIncoming().empty() && phi.getIncoming().front().second) {
874 const auto incomingOwnership =
875 phi.getIncoming().front().second->getOwnership();
876 const bool allIncomingHaveSameOwnership = std::all_of(
877 phi.getIncoming().begin(), phi.getIncoming().end(),
878 [incomingOwnership](const auto &incoming) {
879 return incoming.second &&
880 incoming.second->getOwnership() == incomingOwnership;
881 });
882 if (!allIncomingHaveSameOwnership) {
883 error(VerificationErrorCode::InvalidResult, &block, index,
884 "phi incoming values must have matching ownership");
885 return;
886 }
887 expectedOwnership = incomingOwnership;
888 }
889 if (phi.getResult()->getOwnership() != expectedOwnership) {
890 error(VerificationErrorCode::InvalidResult, &block, index,
891 "phi result ownership does not match incoming values");
892 }
893 }
