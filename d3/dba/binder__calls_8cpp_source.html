748 if (!resolvedSymbol->parameters[i]->is_ref) {
749 if (auto conversion =
750 conversions_.classifyImplicit(argClone->type, expected)) {
751 argClone = applyConversion(std::move(argClone), *conversion);
752 }
753 }
754 }
755 remappedArgs.push_back(std::move(argClone));
756 remappedRef.push_back(
757 i < match.argumentIsRef.size() ? match.argumentIsRef[i] : false);
758 }
759
760 match.arguments = std::move(remappedArgs);
761 match.argumentIsRef = std::move(remappedRef);
762 match.symbol = resolvedSymbol;
763 }
764
765 if (resolvedSymbol->isUnsafe && !isUnsafeActive()) {
766 blockedUnsafeMatch = resolvedSymbol;
767 rejectionNotes.push_back("'" + renderFunctionSignature(*resolvedSymbol) +
768 "': requires unsafe context");
769 continue;
770 }
771 matches.push_back(std::move(match));
772 }
773
774 if (matches.empty()) {
775 if (blockedUnsafeMatch) {
776 requireUnsafeContext(node.span, "unsafe function calls");
777 return;
778 }
779
780 error(
781 node.callee_->span,
782 "No matching overload for function '" + calleeParts.back() + "'. " +
783 (rejectionNotes.empty() ? std::string() : ("Candidates: " + [&]() {
784 std::string details;
785 for (size_t i = 0; i < rejectionNotes.size(); ++i) {
786 if (i != 0) {
787 details += "; ";
788 }
789 details += rejectionNotes[i];
790 }
791 return details;
792 }())));
793 return;
794 }
795
796 size_t bestIndex = 0;
797 for (size_t i = 1; i < matches.size(); ++i) {
798 if (compareCost(matches[i], matches[bestIndex])) {
799 bestIndex = i;
800 }
801 }
802
803 std::vector<size_t> ambiguous = {bestIndex};
804 for (size_t i = 0; i < matches.size(); ++i) {
805 if (i == bestIndex) {
806 continue;
807 }
808 if (!compareCost(matches[i], matches[bestIndex]) &&
809 !compareCost(matches[bestIndex], matches[i])) {
810 ambiguous.push_back(i);
811 }
812 }
813
814 if (ambiguous.size() > 1) {
815 std::string message =
816 "Call to function '" + calleeParts.back() + "' is ambiguous between ";
817 for (size_t i = 0; i < ambiguous.size(); ++i) {
818 if (i != 0) {
819 message += i + 1 == ambiguous.size() ? " and " : ", ";
820 }
821 message +=
822 "'" + renderFunctionSignature(*matches[ambiguous[i]].symbol) + "'";
823 }
824 message += ".";
825 if (expectedReturnType) {
826 message +=
827 " Expected result type: '" + expectedReturnType->toString() + "'.";
828 }
829 message += " Candidate details: ";
830 for (size_t i = 0; i < ambiguous.size(); ++i) {
831 if (i != 0) {
832 message += "; ";
833 }
834 message +=
835 "'" + renderFunctionSignature(*matches[ambiguous[i]].symbol) + "' [";
836 for (size_t j = 0; j < matches[ambiguous[i]].notes.size(); ++j) {
837 if (j != 0) {
838 message += ", ";
839 }
840 message += matches[ambiguous[i]].notes[j];
841 }
842 message += "]";
843 }
844 error(node.callee_->span, message);
845 return;
846 }
847
848 auto &best = matches[bestIndex];
849 expressionStack_.push(std::make_unique<BoundFunctionCall>(
850 best.symbol, std::move(best.arguments), std::move(best.argumentIsRef),
851 std::move(best.variadicPack)));
852 if (semanticInfo_) {
853 semanticInfo_->recordCall(currentModuleId_, node.span, best.symbol);
854 }
855}

882bool Binder::bindWeakBuiltinCall(FunCall &node) {
883 auto *calleeId = dynamic_cast<ConstId *>(node.callee_.get());
884 if (!calleeId) {
885 return false;
886 }
887
888 const bool isLock = calleeId->value_ == "lock";
889 const bool isAlive = calleeId->value_ == "alive";
890 if (!isLock && !isAlive) {
891 return false;
892 }
893
894 if (node.params_.size() != 1 || node.params_[0]->isRef ||
895 node.params_[0]->isSpread || !node.params_[0]->name.empty()) {
896 error(node.span, "'" + calleeId->value_ +
897 "' expects exactly one positional argument.");
898 return true;
899 }
900
901 auto weakExpr =
902 bindExpressionWithExpected(node.params_[0]->value.get(), nullptr);
903 if (!weakExpr) {
904 return true;
905 }
906
907 if (weakExpr->type->getKind() != zir::TypeKind::Class) {
908 error(node.params_[0]->value->span,
909 "'" + calleeId->value_ + "' expects a weak class reference.");
910 return true;
911 }
912
913 auto weakClassType = std::static_pointer_cast<zir::ClassType>(weakExpr->type);
914 if (!weakClassType->isWeak()) {
915 error(node.params_[0]->value->span,
916 "'" + calleeId->value_ + "' expects a weak class reference.");
917 return true;
918 }
919
920 if (isAlive) {
921 expressionStack_.push(
922 std::make_unique<BoundWeakAliveExpression>(std::move(weakExpr)));
923 return true;
924 }
925
926 auto strongType = std::make_shared<zir::ClassType>(*weakClassType);
927 strongType->setWeak(false);
928 expressionStack_.push(std::make_unique<BoundWeakLockExpression>(
929 std::move(weakExpr), strongType));
930 return true;
931}
