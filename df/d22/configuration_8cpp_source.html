69 RuntimePathConfiguration result;

71 std::optional<std::string> configCorePath;
72 std::optional<std::string> configStdlibPath;
73 std::filesystem::path configBase = workspaceRoot;

76 if (std::filesystem::is_regular_file(configPath, ec) && !ec) {
77 std::ifstream input(configPath, std::ios::binary);
78 std::ostringstream contents;
79 contents << input.rdbuf();
80 JsonObject config = JsonParser::parse(contents.str());
81 if (!input.good() && !input.eof()) {
82 result.errors.push_back("Could not read " + configPath.string());
83 } else if (!config.isObject()) {
84 result.errors.push_back(configPath.string() +
85 " must contain a JSON object");
86 } else {
87 for (const auto &[property, _] : config.getAsObject()) {
88 if (!isKnownProperty(property)) {
89 result.errors.push_back("Unknown property '" + property +
90 "' in zaplsp.json");
91 }
92 }
93 auto zapRoot = optionalStringProperty(config, "zapRoot", result.errors);
94 if (zapRoot) {
95 configBase = resolvePath(workspaceRoot, *zapRoot);
96 }
97 configCorePath =
98 optionalStringProperty(config, "corePath", result.errors);
99 configStdlibPath =
100 optionalStringProperty(config, "stdlibPath", result.errors);
101 if (zapRoot) {
102 configCorePath = configCorePath.value_or("core");
103 configStdlibPath = configStdlibPath.value_or("std");
104 } else if (!configCorePath || !configStdlibPath) {
105 result.errors.push_back(
106 "zaplsp.json must define zapRoot or both corePath and stdlibPath");
107 }
108 }
109 }
110
111 const auto corePath =
112 initializationCorePath ? initializationCorePath : configCorePath;
113 const auto stdlibPath =
114 initializationStdlibPath ? initializationStdlibPath : configStdlibPath;
115
116 if (corePath) {
117 const auto base = initializationCorePath ? workspaceRoot : configBase;
118 result.coreDir = resolvePath(base, *corePath);
119 validatePath("corePath",
120 initializationCorePath ? "initializationOptions"
121 : "zaplsp.json",
122 *result.coreDir, "core.zp", result.errors);
123 }
124 if (stdlibPath) {
125 const auto base = initializationStdlibPath ? workspaceRoot : configBase;
126 result.stdlibDir = resolvePath(base, *stdlibPath);
127 validatePath("stdlibPath",
128 initializationStdlibPath ? "initializationOptions"
129 : "zaplsp.json",
130 *result.stdlibDir, "prelude.zp", result.errors);
131 }
