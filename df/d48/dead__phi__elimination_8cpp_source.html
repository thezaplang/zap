129std::unordered_set<const Value *> collectUsedValues(const Function &function) {
130 std::unordered_set<const Value *> phiResults;
131 for (const auto &block : function.getBlocks()) {
132 if (!block) {
133 continue;
134 }
135 for (const auto &instruction : block->getInstructions()) {
136 if (!instruction || instruction->getOpCode() != OpCode::Phi) {
137 continue;
138 }
139 const auto &result =
140 static_cast<const PhiInst &>(*instruction).getResult();
141 if (result) {
142 phiResults.insert(result.get());
143 }
144 }
145 }

148 for (const auto &block : function.getBlocks()) {
149 if (!block) {
150 continue;
151 }

153 if (!candidate) {
154 continue;

169void removeDeadPhiInstructions(Function &function) {
170 bool changed = true;
171 while (changed) {
172 changed = false;
173 const auto used = collectUsedValues(function);
174 for (const auto &block : function.getBlocks()) {
175 if (!block) {
176 continue;
177 }
178 auto &instructions = block->instructions;
179 const auto newEnd = std::remove_if(
180 instructions.begin(), instructions.end(),
181 [&](const std::unique_ptr<Instruction> &instruction) {
182 if (!instruction || instruction->getOpCode() != OpCode::Phi) {
183 return false;
184 }

187 const bool dead = result && used.count(result.get()) == 0;
188 changed = changed || dead;
189 return dead;
190 });
191 instructions.erase(newEnd, instructions.end());
192 }
