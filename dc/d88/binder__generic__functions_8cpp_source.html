266std::shared_ptr<zir::Type> Binder::substituteGenericType(
267 std::shared_ptr<zir::Type> type,
268 const std::unordered_map<std::string, std::shared_ptr<zir::Type>>
269 &genericBindings) const {
270 if (!type) {
271 return nullptr;
272 }
273
274 if (type->getKind() == zir::TypeKind::Record) {
275 auto record = std::static_pointer_cast<zir::RecordType>(type);
276 if (record->getRole() == zir::RecordRole::GenericParameter) {
277 auto it = genericBindings.find(record->getName());
278 return it != genericBindings.end() ? it->second : type;
279 }
280 if (record->getRole() == zir::RecordRole::VariadicView) {
281 const auto &fields = record->getFields();
282 if (fields.size() >= 2 &&
283 fields[0].type->getKind() == zir::TypeKind::Pointer) {
284 auto dataPtr =
285 std::static_pointer_cast<zir::PointerType>(fields[0].type);
286 return makeVariadicViewType(
287 substituteGenericType(dataPtr->getBaseType(), genericBindings));
288 }
289 return type;
290 }
291 if (record->isGenericInstance()) {
292 std::vector<std::shared_ptr<zir::Type>> substitutedArgs;
293 substitutedArgs.reserve(record->getGenericArguments().size());
294 for (const auto &arg : record->getGenericArguments()) {
295 substitutedArgs.push_back(substituteGenericType(arg, genericBindings));
296 }
297
298 auto substituted = std::make_shared<zir::RecordType>(
299 renderGenericTypeName(record->getGenericBaseName(), substitutedArgs),
300 renderGenericCodegenName(record->getGenericCodegenBaseName(),
301 substitutedArgs));
302 substituted->setGenericInstance(record->getGenericBaseName(),
303 record->getGenericCodegenBaseName(),
304 substitutedArgs);
305 for (const auto &field : record->getFields()) {
306 substituted->addField(
307 field.name, substituteGenericType(field.type, genericBindings),
308 field.visibility);
309 }
310 return substituted;
311 }
312
313 return type;
314 }
315
316 if (type->getKind() == zir::TypeKind::Class) {
317 auto classType = std::static_pointer_cast<zir::ClassType>(type);
318 auto it = genericBindings.find(classType->getName());
319 if (it != genericBindings.end()) {
320 return it->second;
321 }
322 if (classType->isGenericInstance()) {
323 std::vector<std::shared_ptr<zir::Type>> substitutedArgs;
324 substitutedArgs.reserve(classType->getGenericArguments().size());
325 for (const auto &arg : classType->getGenericArguments()) {
326 substitutedArgs.push_back(substituteGenericType(arg, genericBindings));
327 }
328
329 auto substituted = std::make_shared<zir::ClassType>(
330 renderGenericTypeName(classType->getGenericBaseName(),
331 substitutedArgs),
332 renderGenericCodegenName(classType->getGenericCodegenBaseName(),
333 substitutedArgs));
334 substituted->setGenericInstance(classType->getGenericBaseName(),
335 classType->getGenericCodegenBaseName(),
336 substitutedArgs);
337 substituted->setWeak(classType->isWeak());
338 if (auto base = classType->getBase()) {
339 auto substitutedBase = substituteGenericType(base, genericBindings);
340 if (substitutedBase &&
341 substitutedBase->getKind() == zir::TypeKind::Class) {
342 substituted->setBase(
343 std::static_pointer_cast<zir::ClassType>(substitutedBase));
344 }
345 }
346 for (const auto &field : classType->getFields()) {
347 substituted->addField(
348 field.name, substituteGenericType(field.type, genericBindings),
349 field.visibility);
350 }
351 return substituted;
352 }
353 return type;
354 }
355
356 if (type->getKind() == zir::TypeKind::Pointer) {
357 auto ptr = std::static_pointer_cast<zir::PointerType>(type);
358 auto base = substituteGenericType(ptr->getBaseType(), genericBindings);
359 return std::make_shared<zir::PointerType>(base);
360 }
361
362 if (type->getKind() == zir::TypeKind::Array) {
363 auto arr = std::static_pointer_cast<zir::ArrayType>(type);
364 auto base = substituteGenericType(arr->getBaseType(), genericBindings);
365 return std::make_shared<zir::ArrayType>(base, arr->getSize());
366 }

368 return type;

371bool Binder::validateGenericConstraints(
372 const std::vector<GenericConstraint> &constraints,
373 std::unordered_map<std::string, std::shared_ptr<zir::Type>> &bindings,
374 std::string *failureReason) {
375 for (const auto &constraint : constraints) {
376 auto boundIt = bindings.find(constraint.parameterName);
377 if (boundIt == bindings.end() || !boundIt->second) {
378 if (failureReason) {
379 *failureReason = "missing binding for constrained type parameter '" +
380 constraint.parameterName + "'";
381 }
382 return false;
383 }
384 if (!constraint.boundType) {
385 continue;
386 }
387
388 activeGenericBindingsStack_.push_back(bindings);
389 auto requiredType = mapType(*constraint.boundType);
390 activeGenericBindingsStack_.pop_back();
391 if (!requiredType) {
392 if (failureReason) {
393 *failureReason =
394 "unknown constraint type for '" + constraint.parameterName + "'";
395 }
396 return false;
397 }
398 if (!conversions_.isSubtype(boundIt->second, requiredType)) {
399 if (failureReason) {
400 *failureReason =
401 "type parameter '" + constraint.parameterName + "' with type '" +
402 boundIt->second->toString() + "' does not satisfy constraint '" +
403 constraint.parameterName + ": " + requiredType->toString() + "'";
404 }
405 return false;
406 }
407 }
408 return true;
409}

412Binder::orderedGenericBindings(
413 const std::unordered_map<std::string, std::shared_ptr<zir::Type>>
414 &genericBindings) const {
415 std::vector<std::pair<std::string, std::shared_ptr<zir::Type>>> ordered;
416 ordered.reserve(genericBindings.size());
417 for (const auto &[name, type] : genericBindings) {
418 ordered.emplace_back(name, type);
419 }
420 std::sort(
421 ordered.begin(), ordered.end(),
422 [](const auto &lhs, const auto &rhs) { return lhs.first < rhs.first; });
423 return ordered;
424}
