76ProjectConfigurationResult
77loadProjectConfiguration(const std::filesystem::path &manifestPath) {
78 ProjectConfigurationResult result;
79 ParsedToml document(manifestPath);
80 if (!document.ok()) {
81 result.errors.push_back("Could not parse " + manifestPath.string() +
82 ": " + std::string(document.error()));
83 return result;
84 }
85 if (document.root().type != TOML_TABLE) {
86 result.errors.push_back(manifestPath.string() +
87 " must contain a TOML table");
88 return result;
89 }
90
91 ProjectConfiguration configuration;
92 configuration.manifestPath = manifestPath;
93 configuration.rootDirectory = manifestPath.parent_path();
94
95 const toml_datum_t entry = toml_get(document.root(), "entry");
96 if (entry.type != TOML_UNKNOWN) {
97 if (entry.type != TOML_STRING || entry.u.str.len == 0) {
98 result.errors.push_back("Property 'entry' in " + manifestPath.string() +
99 datumLocation(entry) +
100 " must be a non-empty string");
101 } else {
102 std::filesystem::path entryPath(stringValue(entry));
103 if (entryPath.is_relative()) {
104 entryPath = configuration.rootDirectory / entryPath;
105 }
106 configuration.entryPath = canonicalPath(entryPath);
107 }
108 }
109
110 const toml_datum_t imports = toml_get(document.root(), "imports");
111 if (imports.type != TOML_UNKNOWN) {
112 if (imports.type != TOML_TABLE) {
113 result.errors.push_back("Property 'imports' in " +
114 manifestPath.string() + datumLocation(imports) +
115 " must be a table");
116 } else {
117 for (int32_t i = 0; i < imports.u.tab.size; ++i) {
118 const std::string alias(imports.u.tab.key[i],
119 static_cast<size_t>(imports.u.tab.len[i]));
120 const toml_datum_t &target = imports.u.tab.value[i];
121 if (target.type != TOML_STRING || target.u.str.len == 0) {
122 result.errors.push_back("Import '" + alias + "' in " +
123 manifestPath.string() + datumLocation(target) +
124 " must have a non-empty string path");
125 continue;
126 }
127 std::filesystem::path targetPath(stringValue(target));
128 if (targetPath.is_relative()) {
129 targetPath = configuration.rootDirectory / targetPath;
130 }
131 configuration.importMap.emplace(alias,
132 canonicalPath(targetPath).string());
133 }
134 }
135 }
136
137 if (result.errors.empty()) {
138 result.configuration = std::move(configuration);
139 }
140 return result;
141}
