83SourceManager::sourceForPath(const std::filesystem::path &path) {

87 if (openUri != uriByCanonicalPath_.end()) {
88 auto source = sourcesByUri_.find(openUri->second);
89 if (source != sourcesByUri_.end()) {
90 return source->second;
91 }
92 }

94 std::error_code ec;
95 const auto writeTime = std::filesystem::last_write_time(canonical, ec);
96 if (!ec) {
97 auto cached = fileContentCache_.find(pathKey);
98 if (cached != fileContentCache_.end() &&
99 cached->second.lastWriteTime == writeTime) {
100 return cached->second.snapshot;
101 }
102 }
103
104 std::string content;
105 if (!readSourceFile(canonical, content)) {
106 return std::nullopt;

109 makeSnapshot(pathToUri(canonical), canonical, std::move(content), 0);
110 if (!ec) {
111 fileContentCache_[pathKey] = CachedFile{writeTime, snapshot};
