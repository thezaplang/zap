67void verifyEscapes(const Module &module, const Function &function,
68 const ControlFlowGraph &cfg,
69 std::vector<VerificationError> &errors) {
70 const auto provenance = analyzeBorrowProvenance(module, function, cfg);
71 for (const auto &blockOwner : function.getBlocks()) {
72 if (!blockOwner) {
73 continue;
74 }
75 const auto &block = *blockOwner;
76 for (size_t i = 0; i < block.getInstructions().size(); ++i) {
77 const auto &instruction = block.getInstructions()[i];
78 if (!instruction) {
79 continue;
80 }
81 if (instruction->getOpCode() == OpCode::Ret) {
82 const auto &ret = static_cast<const ReturnInst &>(*instruction);
83 if (!isBorrowTrackedValue(ret.getValue())) {
84 continue;
85 }
86 const auto sources = provenance.ownersAtDefinition(ret.getValue());
87 if (hasDisallowedReturnSource(function, sources)) {
88 addError(errors, function, VerificationErrorCode::InvalidReturn,
89 &block, i,
90 "cannot return " + ret.getValue()->getName() +
91 " backed by non-escaping borrow source " +
92 formatSources(sources));
93 }
94 } else if (instruction->getOpCode() == OpCode::Store) {
95 const auto &store = static_cast<const StoreInst &>(*instruction);
96 if (!isBorrowTrackedValue(store.getSource())) {
97 continue;
98 }
99 const auto sources =
100 provenance.ownersAtDefinition(store.getSource());
101 if (!sources.empty() &&
102 !provenance.isLocalStorage(store.getDestination())) {
103 addError(errors, function, VerificationErrorCode::InvalidOperand,
104 &block, i,
105 "cannot store " + store.getSource()->getName() +
106 " backed by non-escaping borrow source " +
107 formatSources(sources) + " outside local storage");
108 }
109 } else if (instruction->getOpCode() == OpCode::Call) {
110 const auto &call = static_cast<const CallInst &>(*instruction);
111 for (size_t argumentIndex = 0;
112 argumentIndex < call.getArguments().size(); ++argumentIndex) {
113 if (!isBorrowTrackedValue(call.getArguments()[argumentIndex])) {
114 continue;
115 }
116 const auto sources = provenance.ownersAtDefinition(
117 call.getArguments()[argumentIndex]);
118 const auto resultBorrow =
119 resolveCallResultBorrowContract(module, call);
120 const bool mayBackResult =
121 resultBorrow.hasSource() &&
122 *resultBorrow.sourceParameter() == argumentIndex;
123 const auto contract =
124 resolveCallParameterContract(module, call, argumentIndex);
125 if (!sources.empty() &&
126 (!contract || contract->escape != ParameterEscape::NoEscape) &&
127 !mayBackResult) {
128 addError(
129 errors, function, VerificationErrorCode::InvalidCall, &block,
130 i,
131 "cannot pass " +
132 call.getArguments()[argumentIndex]->getName() +
133 " backed by tracked borrow source " +
134 formatSources(sources) +
135 " to a parameter with an unspecified escape contract");
136 }
137 }

143void verifyResultContract(const Function &function,
144 std::vector<VerificationError> &errors) {
145 if (!function.resultBorrow.hasSource()) {
146 return;
147 }
148
149 const size_t sourceIndex = *function.resultBorrow.sourceParameter();
150 if (sourceIndex >= function.getArguments().size()) {
151 addError(errors, function, VerificationErrorCode::InvalidReturn, nullptr,
152 std::nullopt, "result borrow source parameter is out of range");
153 } else {
154 const auto &source = function.getArguments()[sourceIndex];
155 if (!source ||
156 source->getParameterOwnership() != ParameterOwnership::Borrow) {
157 addError(errors, function, VerificationErrorCode::InvalidReturn, nullptr,
158 std::nullopt,
159 "result borrow source parameter must be borrowed");
160 } else if (source->getParameterEscape() ==
161 ParameterEscape::NoEscape) {
162 addError(errors, function, VerificationErrorCode::InvalidReturn, nullptr,
163 std::nullopt,
164 "noescape parameter cannot back the function result");
165 }
166 }
167 if (function.returnsRef || !function.getReturnType() ||
168 function.getReturnType()->getIntrinsicKind() !=
169 IntrinsicTypeKind::StringView) {
170 addError(errors, function, VerificationErrorCode::InvalidReturn, nullptr,
171 std::nullopt,
172 "result borrow contract requires a by-value StringView result");
173 }
174}
