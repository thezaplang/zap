321static int zap_arc_ptrmap_get(const zap_arc_ptrmap_t *m, void *key,
322 uint32_t *out_index) {
323 size_t mask = m->cap - 1;
324 size_t i = zap_arc_hash_ptr(key) & mask;
325 while (m->keys[i]) {
326 if (m->keys[i] == key) {
327 *out_index = m->vals[i];
328 return 1;
329 }
330 i = (i + 1) & mask;
331 }
332 return 0;

335static void zap_arc_ptrmap_grow(zap_arc_ptrmap_t *m) {
336 size_t ncap = zap_runtime_next_capacity(m->cap, 16);
337 void **nkeys = (void **)zap_runtime_calloc_array(ncap, sizeof(void *));
338 uint32_t *nvals =
339 (uint32_t *)zap_runtime_calloc_array(ncap, sizeof(uint32_t));
340 size_t nmask = ncap - 1;
341 for (size_t i = 0; i < m->cap; ++i) {
342 if (!m->keys[i]) {
343 continue;
344 }
345 size_t j = zap_arc_hash_ptr(m->keys[i]) & nmask;
346 while (nkeys[j]) {
347 j = (j + 1) & nmask;
348 }
349 nkeys[j] = m->keys[i];
350 nvals[j] = m->vals[i];
351 }
352 free(m->keys);
353 free(m->vals);
354 m->keys = nkeys;
355 m->vals = nvals;
356 m->cap = ncap;

359static void zap_arc_ptrmap_put(zap_arc_ptrmap_t *m, void *key, uint32_t value) {
360 if ((m->len + 1) * 2 >= m->cap) {
361 zap_arc_ptrmap_grow(m);
362 }
363 size_t mask = m->cap - 1;
364 size_t i = zap_arc_hash_ptr(key) & mask;
365 while (m->keys[i]) {

367 m->vals[i] = value;

370 i = (i + 1) & mask;
371 }

396static void zap_arc_ensure_scratch(zap_arc_runtime_context_t *context,
397 size_t n) {
398 if (n <= context->scratch_cap) {
399 return;
400 }
401#if defined(ZAP_RUNTIME_INSTRUMENTATION)

405 }

408 while (ncap < n) {

416 sizeof(uint32_t));
417 context->incoming = ni;
418 context->reachable = nr;

420 context->scratch_cap = ncap;

468void zap_arc_cycle_collect(zap_arc_runtime_context_t *context) {
469 if (!context || context->collecting || context->root_count == 0) {
470 return;
471 }
472 context->collecting = 1;
473#if defined(ZAP_RUNTIME_INSTRUMENTATION)
474 ++zap_runtime_ownership_counters.collection_runs;
475#endif
476
477 if (context->root_count > context->snap_cap) {
478 void **next = (void **)zap_runtime_realloc_array(
479 context->snap, context->root_count, sizeof(void *));
480 context->snap = next;
481 context->snap_cap = context->root_count;
482 }
483 void **roots = context->snap;
484 size_t root_count = 0;
485 for (size_t i = 0; i < context->root_count; ++i) {
486 void *object = context->roots[i];
487 if (!object) {
488 continue;
489 }
490 zap_arc_header_t *header = (zap_arc_header_t *)object;
491 header->gc_mark &= (uint8_t)~ZAP_ARC_GC_BUFFERED;
492 roots[root_count++] = object;
493 }
494 context->root_count = 0;
495 context->collection_pending = 0;
496 if (root_count == 0) {
497 context->collecting = 0;
498 return;
499 }
500
501 if (context->map.cap == 0) {
502 zap_arc_ptrmap_init(&context->map, 64);
503 }
504 zap_arc_ptrmap_clear(&context->map);
505
506 size_t ws_count = 0;
507 for (size_t i = 0; i < root_count; ++i) {
508 zap_arc_ws_push(&context->worklist, &ws_count, &context->worklist_cap,
509 &context->map, roots[i]);
510 }
511 for (size_t cursor = 0; cursor < ws_count; ++cursor) {
512 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[cursor];
513 if (!header->alive || !header->metadata) {
514 continue;
515 }
516 if (header->metadata->trace_fn) {
517 zap_arc_discover_context_t discovery = {
518 &context->worklist, &ws_count, &context->worklist_cap, &context->map};
519 header->metadata->trace_fn(context->worklist[cursor],
520 zap_arc_discover_child, &discovery);
521 }
522 }
523
524#if defined(ZAP_RUNTIME_INSTRUMENTATION)
525 zap_runtime_ownership_counters.visited_objects += ws_count;
526#endif
527
528 zap_arc_ensure_scratch(context, ws_count);
529 int *incoming = context->incoming;
530 uint8_t *reachable = context->reachable;
531 uint32_t *stack = context->stack;
532 memset(incoming, 0, ws_count * sizeof(int));
533 memset(reachable, 0, ws_count * sizeof(uint8_t));
534
535 for (size_t i = 0; i < ws_count; ++i) {
536 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
537 if (!header->alive || !header->metadata) {
538 continue;
539 }
540 if (header->metadata->trace_fn) {
541 zap_arc_incoming_context_t incoming_context = {&context->map, incoming};
542 header->metadata->trace_fn(context->worklist[i], zap_arc_count_incoming,
543 &incoming_context);
544 }
545 }
546
547 size_t sp = 0;
548 for (size_t i = 0; i < ws_count; ++i) {
549 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
550 if (header->alive && header->strong_count > incoming[i] && !reachable[i]) {
551 reachable[i] = 1;
552 stack[sp++] = (uint32_t)i;
553 }
554 }
555 while (sp) {
556 uint32_t idx = stack[--sp];
557 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[idx];
558 if (!header->alive || !header->metadata) {
559 continue;
560 }
561 if (header->metadata->trace_fn) {
562 zap_arc_reachable_context_t reachable_context = {&context->map, reachable,
563 stack, &sp};
564 header->metadata->trace_fn(context->worklist[idx], zap_arc_mark_reachable,
565 &reachable_context);
566 }
567 }
568
569 for (size_t i = 0; i < ws_count; ++i) {
570 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
571 if (!header->alive || reachable[i]) {
572 continue;
573 }
574 header->gc_mark = ZAP_ARC_GC_GARBAGE;
575 header->alive = 0;
576#if defined(ZAP_RUNTIME_INSTRUMENTATION)
577 ++zap_runtime_ownership_counters.reclaimed_objects;
578#endif
579 }
580 for (size_t i = 0; i < ws_count; ++i) {
581 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
582 if ((header->gc_mark & ZAP_ARC_GC_GARBAGE) && header->destroy_fn) {
583 header->destroy_fn(context->worklist[i]);
584 }
585 }
586 for (size_t i = 0; i < ws_count; ++i) {
587 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
588 if ((header->gc_mark & ZAP_ARC_GC_GARBAGE) && header->weak_count == 0) {
589 zap_arc_deallocate(context, context->worklist[i]);
590 } else if (header->gc_mark & ZAP_ARC_GC_GARBAGE) {
591 header->gc_mark &= (uint8_t)~ZAP_ARC_GC_GARBAGE;
592 }
593 }
594
595 context->collecting = 0;
596}
