297static int zap_arc_ptrmap_get(const zap_arc_ptrmap_t *m, void *key,
298 uint32_t *out_index) {
299 size_t mask = m->cap - 1;
300 size_t i = zap_arc_hash_ptr(key) & mask;
301 while (m->keys[i]) {
302 if (m->keys[i] == key) {
303 *out_index = m->vals[i];
304 return 1;
305 }
306 i = (i + 1) & mask;
307 }
308 return 0;

311static void zap_arc_ptrmap_grow(zap_arc_ptrmap_t *m) {
312 size_t ncap = zap_runtime_next_capacity(m->cap, 16);
313 void **nkeys = (void **)zap_runtime_calloc_array(ncap, sizeof(void *));
314 uint32_t *nvals =
315 (uint32_t *)zap_runtime_calloc_array(ncap, sizeof(uint32_t));
316 size_t nmask = ncap - 1;
317 for (size_t i = 0; i < m->cap; ++i) {
318 if (!m->keys[i]) {
319 continue;
320 }
321 size_t j = zap_arc_hash_ptr(m->keys[i]) & nmask;
322 while (nkeys[j]) {
323 j = (j + 1) & nmask;
324 }
325 nkeys[j] = m->keys[i];
326 nvals[j] = m->vals[i];
327 }
328 free(m->keys);
329 free(m->vals);
330 m->keys = nkeys;
331 m->vals = nvals;
332 m->cap = ncap;
333}

335static void zap_arc_ptrmap_put(zap_arc_ptrmap_t *m, void *key, uint32_t value) {
336 if ((m->len + 1) * 2 >= m->cap) {
337 zap_arc_ptrmap_grow(m);
338 }
339 size_t mask = m->cap - 1;
340 size_t i = zap_arc_hash_ptr(key) & mask;
341 while (m->keys[i]) {
342 if (m->keys[i] == key) {
343 m->vals[i] = value;

346 i = (i + 1) & mask;
347 }

372static void zap_arc_ensure_scratch(zap_arc_runtime_context_t *context,
373 size_t n) {
374 if (n <= context->scratch_cap) {
375 return;
376 }
377#if defined(ZAP_RUNTIME_INSTRUMENTATION)
378 if (zap_runtime_fail_arc_scratch_allocation) {

380 zap_runtime_out_of_memory();
381 }
382#endif
383 size_t ncap = context->scratch_cap ? context->scratch_cap : 32;
384 while (ncap < n) {
385 ncap = zap_runtime_next_capacity(ncap, 32);

387 int *ni =
388 (int *)zap_runtime_realloc_array(context->incoming, ncap, sizeof(int));

390 sizeof(uint8_t));

392 sizeof(uint32_t));
393 context->incoming = ni;
394 context->reachable = nr;
395 context->stack = ns;
396 context->scratch_cap = ncap;

444void zap_arc_cycle_collect(zap_arc_runtime_context_t *context) {
445 if (!context || context->collecting || context->root_count == 0) {
446 return;
447 }
448 context->collecting = 1;
449#if defined(ZAP_RUNTIME_INSTRUMENTATION)
450 ++zap_runtime_ownership_counters.collection_runs;
451#endif
452
453 if (context->root_count > context->snap_cap) {
454 void **next = (void **)zap_runtime_realloc_array(
455 context->snap, context->root_count, sizeof(void *));
456 context->snap = next;
457 context->snap_cap = context->root_count;
458 }
459 void **roots = context->snap;
460 size_t root_count = 0;
461 for (size_t i = 0; i < context->root_count; ++i) {
462 void *object = context->roots[i];
463 if (!object) {
464 continue;
465 }
466 zap_arc_header_t *header = (zap_arc_header_t *)object;
467 header->gc_mark &= (uint8_t)~ZAP_ARC_GC_BUFFERED;
468 roots[root_count++] = object;
469 }
470 context->root_count = 0;
471 context->collection_pending = 0;
472 if (root_count == 0) {
473 context->collecting = 0;
474 return;
475 }
476
477 if (context->map.cap == 0) {
478 zap_arc_ptrmap_init(&context->map, 64);
479 }
480 zap_arc_ptrmap_clear(&context->map);
481
482 size_t ws_count = 0;
483 for (size_t i = 0; i < root_count; ++i) {
484 zap_arc_ws_push(&context->worklist, &ws_count, &context->worklist_cap,
485 &context->map, roots[i]);
486 }
487 for (size_t cursor = 0; cursor < ws_count; ++cursor) {
488 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[cursor];
489 if (!header->alive || !header->metadata) {
490 continue;
491 }
492 if (header->metadata->trace_fn) {
493 zap_arc_discover_context_t discovery = {
494 &context->worklist, &ws_count, &context->worklist_cap, &context->map};
495 header->metadata->trace_fn(context->worklist[cursor],
496 zap_arc_discover_child, &discovery);
497 }
498 }
499
500#if defined(ZAP_RUNTIME_INSTRUMENTATION)
501 zap_runtime_ownership_counters.visited_objects += ws_count;
502#endif
503
504 zap_arc_ensure_scratch(context, ws_count);
505 int *incoming = context->incoming;
506 uint8_t *reachable = context->reachable;
507 uint32_t *stack = context->stack;
508 memset(incoming, 0, ws_count * sizeof(int));
509 memset(reachable, 0, ws_count * sizeof(uint8_t));
510
511 for (size_t i = 0; i < ws_count; ++i) {
512 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
513 if (!header->alive || !header->metadata) {
514 continue;
515 }
516 if (header->metadata->trace_fn) {
517 zap_arc_incoming_context_t incoming_context = {&context->map, incoming};
518 header->metadata->trace_fn(context->worklist[i], zap_arc_count_incoming,
519 &incoming_context);
520 }
521 }
522
523 size_t sp = 0;
524 for (size_t i = 0; i < ws_count; ++i) {
525 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
526 if (header->alive && header->strong_count > incoming[i] && !reachable[i]) {
527 reachable[i] = 1;
528 stack[sp++] = (uint32_t)i;
529 }
530 }
531 while (sp) {
532 uint32_t idx = stack[--sp];
533 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[idx];
534 if (!header->alive || !header->metadata) {
535 continue;
536 }
537 if (header->metadata->trace_fn) {
538 zap_arc_reachable_context_t reachable_context = {&context->map, reachable,
539 stack, &sp};
540 header->metadata->trace_fn(context->worklist[idx], zap_arc_mark_reachable,
541 &reachable_context);
542 }
543 }
544
545 for (size_t i = 0; i < ws_count; ++i) {
546 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
547 if (!header->alive || reachable[i]) {
548 continue;
549 }
550 header->gc_mark = ZAP_ARC_GC_GARBAGE;
551 header->alive = 0;
552#if defined(ZAP_RUNTIME_INSTRUMENTATION)
553 ++zap_runtime_ownership_counters.reclaimed_objects;
554#endif
555 }
556 for (size_t i = 0; i < ws_count; ++i) {
557 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
558 if ((header->gc_mark & ZAP_ARC_GC_GARBAGE) && header->destroy_fn) {
559 header->destroy_fn(context->worklist[i]);
560 }
561 }
562 for (size_t i = 0; i < ws_count; ++i) {
563 zap_arc_header_t *header = (zap_arc_header_t *)context->worklist[i];
564 if ((header->gc_mark & ZAP_ARC_GC_GARBAGE) && header->weak_count == 0) {
565 zap_arc_deallocate(context, context->worklist[i]);
566 } else if (header->gc_mark & ZAP_ARC_GC_GARBAGE) {
567 header->gc_mark &= (uint8_t)~ZAP_ARC_GC_GARBAGE;
568 }
569 }
570
571 context->collecting = 0;
572}
