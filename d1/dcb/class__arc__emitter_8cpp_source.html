66void ClassArcEmitter::ensureNestedClassArcSupport(
67 const std::shared_ptr<zir::Type> &type) {
68 if (!type) {
69 return;
70 }
71
72 switch (type->getKind()) {
73 case zir::TypeKind::Class:
74 ensureClassArcSupport(std::static_pointer_cast<zir::ClassType>(type));
75 return;
76 case zir::TypeKind::Record: {
77 const auto &record = static_cast<const zir::RecordType &>(*type);
78 for (const auto &field : record.getFields()) {
79 ensureNestedClassArcSupport(field.type);
80 }
81 return;
82 }
83 case zir::TypeKind::Array:
84 ensureNestedClassArcSupport(
85 static_cast<const zir::ArrayType &>(*type).getBaseType());
86 return;
87 case zir::TypeKind::TaggedUnion: {
88 const auto &taggedUnion = static_cast<const zir::TaggedUnionType &>(*type);
89 for (const auto &variant : taggedUnion.getVariants()) {
90 ensureNestedClassArcSupport(variant.payloadType);
91 }
92 return;
93 }
94 default:
95 return;
96 }
97}

147void ClassArcEmitter::emitTraceChildren(
148 const std::shared_ptr<zir::Type> &type, llvm::Value *address,
149 llvm::Value *visitor, llvm::Value *context) {
150 if (!type || !address) {
151 return;
152 }
153
154 if (isClassType(type)) {
155 if (isWeakClassType(type)) {
156 return;
157 }
158 auto *child = codegen_.builder_.CreateLoad(codegen_.toLLVMType(*type),
159 address, "trace.child");
160 auto *rawPtrTy = llvm::PointerType::getUnqual(codegen_.ctx_);
161 auto *visitorTy = llvm::FunctionType::get(
162 llvm::Type::getVoidTy(codegen_.ctx_), {rawPtrTy, rawPtrTy}, false);
163 codegen_.builder_.CreateCall(visitorTy, visitor, {context, child});
164 return;
165 }
166
167 if (type->getKind() == zir::TypeKind::Record) {
168 const auto &record = static_cast<const zir::RecordType &>(*type);
169 auto *recordTy = llvm::cast<llvm::StructType>(codegen_.toLLVMType(*type));
170 for (size_t i = 0; i < record.getFields().size(); ++i) {
171 auto *fieldAddr = codegen_.builder_.CreateStructGEP(
172 recordTy, address, static_cast<unsigned>(i), "trace.field.addr");
173 emitTraceChildren(record.getFields()[i].type, fieldAddr, visitor,
174 context);
175 }
176 return;
177 }
178
179 if (type->getKind() == zir::TypeKind::Array) {
180 const auto &array = static_cast<const zir::ArrayType &>(*type);
181 auto *arrayTy = llvm::cast<llvm::ArrayType>(codegen_.toLLVMType(*type));
182 auto *i32Ty = llvm::Type::getInt32Ty(codegen_.ctx_);
183 for (size_t i = 0; i < array.getSize(); ++i) {
184 llvm::Value *indices[] = {
185 llvm::ConstantInt::get(i32Ty, 0), llvm::ConstantInt::get(i32Ty, i)};
186 auto *elementAddr = codegen_.builder_.CreateInBoundsGEP(
187 arrayTy, address, indices, "trace.element.addr");
188 emitTraceChildren(array.getBaseType(), elementAddr, visitor, context);
189 }
190 return;
191 }
192
193 if (type->getKind() != zir::TypeKind::TaggedUnion) {
194 return;
195 }
196
197 const auto &taggedUnion = static_cast<const zir::TaggedUnionType &>(*type);
198 auto *unionTy = llvm::cast<llvm::StructType>(codegen_.toLLVMType(*type));
199 auto *tagAddr =
200 codegen_.builder_.CreateStructGEP(unionTy, address, 0, "trace.tag.addr");
201 auto *tag = codegen_.builder_.CreateLoad(llvm::Type::getInt32Ty(codegen_.ctx_),
202 tagAddr, "trace.tag");
203 auto *done = llvm::BasicBlock::Create(codegen_.ctx_, "trace.union.done",
204 codegen_.currentFn_);
205
206 for (const auto &variant : taggedUnion.getVariants()) {
207 if (!variant.payloadType) {
208 continue;
209 }
210
211 auto *active = llvm::BasicBlock::Create(codegen_.ctx_, "trace.union.active",
212 codegen_.currentFn_);
213 auto *next = llvm::BasicBlock::Create(codegen_.ctx_, "trace.union.next",
214 codegen_.currentFn_);
215 auto *matches = codegen_.builder_.CreateICmpEQ(
216 tag, llvm::ConstantInt::getSigned(llvm::Type::getInt32Ty(codegen_.ctx_),
217 variant.tag),
218 "trace.union.is_active");
219 codegen_.builder_.CreateCondBr(matches, active, next);
220
221 codegen_.builder_.SetInsertPoint(active);
222 auto *payloadAddr = codegen_.builder_.CreateStructGEP(
223 unionTy, address, 1, "trace.payload.addr");
224 emitTraceChildren(variant.payloadType, payloadAddr, visitor, context);
225 codegen_.builder_.CreateBr(done);
226
227 codegen_.builder_.SetInsertPoint(next);
228 }
229
230 codegen_.builder_.CreateBr(done);
231 codegen_.builder_.SetInsertPoint(done);
232}

554 auto *typedPtr = codegen_.builder_.CreateBitCast(
555 value, llvm::PointerType::getUnqual(codegen_.ctx_), "arc.weak.alive.cast");
556 auto *aliveAddr = codegen_.builder_.CreateStructGEP(
557 objectTy, typedPtr, kClassAliveIndex, "arc.weak.alive.addr");
558 auto *aliveValue = codegen_.builder_.CreateLoad(
559 llvm::Type::getInt8Ty(codegen_.ctx_), aliveAddr, "arc.weak.alive");
560 auto *aliveBool = codegen_.builder_.CreateICmpNE(
561 aliveValue,
562 llvm::ConstantInt::get(llvm::Type::getInt8Ty(codegen_.ctx_), 0));
563 auto *aliveBB = codegen_.builder_.GetInsertBlock();
564 codegen_.builder_.CreateBr(mergeBB);
565
566 codegen_.builder_.SetInsertPoint(mergeBB);
567 auto *phi = codegen_.builder_.CreatePHI(llvm::Type::getInt1Ty(codegen_.ctx_),
568 2, "arc.weak.alive.res");
569 phi->addIncoming(
570 llvm::ConstantInt::get(llvm::Type::getInt1Ty(codegen_.ctx_), 0), entryBB);
571 phi->addIncoming(aliveBool, aliveBB);
572 return phi;
573}

575llvm::Value *
576ClassArcEmitter::emitWeakLock(llvm::Value *value,
577 const std::shared_ptr<zir::Type> &type) {
578 if (!value || !isWeakClassType(type)) {
579 return llvm::ConstantPointerNull::get(
580 llvm::cast<llvm::PointerType>(codegen_.toLLVMType(*type)));
581 }
582
583 auto weakClassType = std::static_pointer_cast<zir::ClassType>(type);
584 auto strongClassType = std::make_shared<zir::ClassType>(*weakClassType);
585 strongClassType->setWeak(false);
586 auto *resultTy =
587 llvm::cast<llvm::PointerType>(codegen_.toLLVMType(*strongClassType));
588 auto *objectTy =
589 codegen_.structCache_.at(weakClassType->getCodegenName() + ".obj");
590 auto *valuePtrTy = llvm::dyn_cast<llvm::PointerType>(value->getType());
591 if (!valuePtrTy) {
592 return llvm::ConstantPointerNull::get(resultTy);
593 }
594
595 auto *aliveCheckBB = llvm::BasicBlock::Create(
596 codegen_.ctx_, "arc.weak.lock.check", codegen_.currentFn_);
597 auto *retainBB = llvm::BasicBlock::Create(
598 codegen_.ctx_, "arc.weak.lock.retain", codegen_.currentFn_);
599 auto *mergeBB = llvm::BasicBlock::Create(codegen_.ctx_, "arc.weak.lock.merge",
600 codegen_.currentFn_);
601 auto *isNull = codegen_.builder_.CreateICmpEQ(
602 value, llvm::ConstantPointerNull::get(valuePtrTy),
603 "arc.weak.lock.isnull");
604 auto *entryBB = codegen_.builder_.GetInsertBlock();
605 codegen_.builder_.CreateCondBr(isNull, mergeBB, aliveCheckBB);
606
607 codegen_.builder_.SetInsertPoint(aliveCheckBB);
608 auto *typedPtr = codegen_.builder_.CreateBitCast(
609 value, llvm::PointerType::getUnqual(codegen_.ctx_), "arc.weak.lock.cast");
610 auto *aliveAddr = codegen_.builder_.CreateStructGEP(
611 objectTy, typedPtr, kClassAliveIndex, "arc.weak.lock.alive.addr");
612 auto *aliveValue = codegen_.builder_.CreateLoad(
613 llvm::Type::getInt8Ty(codegen_.ctx_), aliveAddr, "arc.weak.lock.alive");
614 auto *aliveBool = codegen_.builder_.CreateICmpNE(
615 aliveValue,
616 llvm::ConstantInt::get(llvm::Type::getInt8Ty(codegen_.ctx_), 0));
617 auto *aliveCheckBlock = codegen_.builder_.GetInsertBlock();
618 codegen_.builder_.CreateCondBr(aliveBool, retainBB, mergeBB);
619
620 codegen_.builder_.SetInsertPoint(retainBB);
621 auto *strongValue =
622 codegen_.builder_.CreateBitCast(value, resultTy, "arc.weak.lock.value");
623 emitRetainIfNeeded(strongValue, strongClassType);
624 auto *retainBlock = codegen_.builder_.GetInsertBlock();
625 codegen_.builder_.CreateBr(mergeBB);
626
627 codegen_.builder_.SetInsertPoint(mergeBB);
628 auto *phi = codegen_.builder_.CreatePHI(resultTy, 3, "arc.weak.lock.res");
629 phi->addIncoming(llvm::ConstantPointerNull::get(resultTy), entryBB);
630 phi->addIncoming(llvm::ConstantPointerNull::get(resultTy), aliveCheckBlock);
631 phi->addIncoming(strongValue, retainBlock);
632 return phi;
633}

635void ClassArcEmitter::emitStoreWithArc(llvm::Value *addr, llvm::Value *value,
636 const std::shared_ptr<zir::Type> &type,
637 zir::ValueOwnership valueOwnership,
638 bool skipReleaseOld) {
639 if (!isClassType(type)) {
640 codegen_.builder_.CreateStore(value, addr);
641 return;
642 }
643
644 if (isWeakClassType(type)) {
645 if (valueOwnership != zir::ValueOwnership::OwnedWeak) {
646 emitRetainWeakIfNeeded(value, type);
647 }
648 if (!skipReleaseOld) {
649 auto *oldValue = codegen_.builder_.CreateLoad(codegen_.toLLVMType(*type),
650 addr, "arc.weak.store.old");
651 codegen_.builder_.CreateStore(value, addr);
652 emitReleaseWeakIfNeeded(oldValue, type);
653 } else {
654 codegen_.builder_.CreateStore(value, addr);
655 }
656 if (valueOwnership == zir::ValueOwnership::OwnedStrong) {
657 auto strongType = std::make_shared<zir::ClassType>(
658 *std::static_pointer_cast<zir::ClassType>(type));
659 strongType->setWeak(false);
660 emitReleaseIfNeeded(value, strongType);
661 }
662 return;
663 }
664
665 if (valueOwnership != zir::ValueOwnership::OwnedStrong) {
666 emitRetainIfNeeded(value, type);
667 }
668 if (!skipReleaseOld) {
669 auto *oldValue = codegen_.builder_.CreateLoad(codegen_.toLLVMType(*type),
670 addr, "arc.store.old");
671 codegen_.builder_.CreateStore(value, addr);
672 emitReleaseIfNeeded(oldValue, type);
673 } else {
674 codegen_.builder_.CreateStore(value, addr);
675 }
676}

678void ClassArcEmitter::ensureClassArcSupport(
679 const std::shared_ptr<zir::ClassType> &classType) {
680 if (!classType || classType->isInterface() ||
681 codegen_.classReleaseFns_.count(classType->getCodegenName())) {
682 return;
683 }
684
685 codegen_.toLLVMType(*classType);
686 auto *objectTy =
687 codegen_.structCache_.at(classType->getCodegenName() + ".obj");
688 auto *rawPtrTy =
689 llvm::PointerType::getUnqual(codegen_.ctx_);
690 auto *helperTy = llvm::FunctionType::get(llvm::Type::getVoidTy(codegen_.ctx_),
691 {rawPtrTy}, false);
692 auto *releaseHelper = llvm::Function::Create(
693 helperTy, llvm::Function::InternalLinkage,
694 "__zap_arc_release_" + classType->getCodegenName(), *codegen_.module_);
695 auto *destroyHelper = llvm::Function::Create(
696 helperTy, llvm::Function::InternalLinkage,
697 "__zap_arc_destroy_" + classType->getCodegenName(), *codegen_.module_);
698 codegen_.classReleaseFns_[classType->getCodegenName()] = releaseHelper;
699 codegen_.classDestroyFns_[classType->getCodegenName()] = destroyHelper;
700
701 for (const auto &field : classType->getFields()) {
702 ensureNestedClassArcSupport(field.type);
703 }
704
705 if (!codegen_.classVTables_.count(classType->getCodegenName())) {
706 std::vector<llvm::Constant *> entries;
707 if (auto base = classType->getBase()) {
708 ensureClassArcSupport(base);
709 auto *baseVTable = codegen_.classVTables_.at(base->getCodegenName());
710 if (auto *baseInit = baseVTable->getInitializer()) {
711 for (unsigned i = 0; i < baseInit->getNumOperands(); ++i) {
712 entries.push_back(
713 llvm::cast<llvm::Constant>(baseInit->getOperand(i)));
714 }
715 }
716 }
717
718 auto methodsIt =
719 codegen_.classVirtualMethodFns_.find(classType->getCodegenName());
720 if (methodsIt != codegen_.classVirtualMethodFns_.end()) {
721 auto *i8PtrTy =
722 llvm::PointerType::getUnqual(codegen_.ctx_);
723 for (const auto &[slot, fn] : methodsIt->second) {
724 if (slot >= static_cast<int>(entries.size())) {
725 entries.resize(static_cast<size_t>(slot + 1),
726 llvm::ConstantPointerNull::get(i8PtrTy));
727 }
728 entries[slot] = llvm::ConstantExpr::getBitCast(fn, i8PtrTy);
729 }
730 }
731
732 auto *i8PtrTy =
733 llvm::PointerType::getUnqual(codegen_.ctx_);
734 auto *vtableTy =
735 llvm::ArrayType::get(i8PtrTy, static_cast<uint64_t>(entries.size()));
736 auto *init = llvm::ConstantArray::get(vtableTy, entries);
737 auto *gv = new llvm::GlobalVariable(
738 *codegen_.module_, vtableTy, true, llvm::GlobalValue::InternalLinkage,
739 init, "__zap_vtable_" + classType->getCodegenName());
740 codegen_.classVTables_[classType->getCodegenName()] = gv;
741 }
742
743 if (!classType->getInterfaceConformances().empty() &&
744 !codegen_.classInterfaceTables_.count(classType->getCodegenName())) {
745 auto *i8PtrTy = llvm::PointerType::getUnqual(codegen_.ctx_);
746 auto *i64Ty = llvm::Type::getInt64Ty(codegen_.ctx_);
747 auto *entryTy = llvm::StructType::get(codegen_.ctx_, {i8PtrTy, i8PtrTy});
748
749 std::vector<llvm::Constant *> entries;
750 for (const auto &conformance : classType->getInterfaceConformances()) {
751 std::vector<llvm::Constant *> slotConsts;
752 slotConsts.reserve(conformance.methodVtableSlots.size());
753 for (int slot : conformance.methodVtableSlots) {
754 slotConsts.push_back(
755 llvm::ConstantInt::get(i64Ty, static_cast<uint64_t>(slot)));
756 }
757 auto *slotsArrTy =
758 llvm::ArrayType::get(i64Ty, slotConsts.size());
759 auto *slotsInit = llvm::ConstantArray::get(slotsArrTy, slotConsts);
760 auto *slotsGlobal = new llvm::GlobalVariable(
761 *codegen_.module_, slotsArrTy, true,
762 llvm::GlobalValue::InternalLinkage, slotsInit,
763 "__zap_iface_slots_" + classType->getCodegenName() + "_" +
764 conformance.interfaceCodegenName);
765
766 auto *nameConst = llvm::ConstantDataArray::getString(
767 codegen_.ctx_, conformance.interfaceCodegenName, true);
768 auto *nameGlobal = new llvm::GlobalVariable(
769 *codegen_.module_, nameConst->getType(), true,
770 llvm::GlobalValue::InternalLinkage, nameConst,
771 "__zap_iface_name_" + conformance.interfaceCodegenName);
772
773 llvm::Constant *fields[] = {
774 llvm::ConstantExpr::getBitCast(nameGlobal, i8PtrTy),
775 llvm::ConstantExpr::getBitCast(slotsGlobal, i8PtrTy)};
776 entries.push_back(llvm::ConstantStruct::get(entryTy, fields));
777 }
778 llvm::Constant *sentinelFields[] = {
779 llvm::ConstantPointerNull::get(i8PtrTy),
780 llvm::ConstantPointerNull::get(i8PtrTy)};
781 entries.push_back(llvm::ConstantStruct::get(entryTy, sentinelFields));
782
783 auto *tableTy = llvm::ArrayType::get(entryTy, entries.size());
784 auto *tableInit = llvm::ConstantArray::get(tableTy, entries);
785 auto *tableGlobal = new llvm::GlobalVariable(
786 *codegen_.module_, tableTy, true, llvm::GlobalValue::InternalLinkage,
787 tableInit, "__zap_iface_table_" + classType->getCodegenName());
788 codegen_.classInterfaceTables_[classType->getCodegenName()] = tableGlobal;
789 }
790
791 if (!codegen_.classMetadataGlobals_.count(classType->getCodegenName())) {
792 auto *traceHelper = emitClassTraceFunction(classType, objectTy);
793 auto *i32PtrTy = llvm::PointerType::getUnqual(codegen_.ctx_);
794 auto *metaTy = llvm::StructType::create(
795 codegen_.ctx_, "__zap_arc_meta_" + classType->getCodegenName());
796 metaTy->setBody({i32PtrTy});
797 auto *metaInit = llvm::ConstantStruct::get(
798 metaTy, llvm::ConstantExpr::getBitCast(traceHelper, i32PtrTy));
799 auto *metaGlobal = new llvm::GlobalVariable(
800 *codegen_.module_, metaTy, true, llvm::GlobalValue::InternalLinkage,
801 metaInit, "__zap_arc_meta_" + classType->getCodegenName());
802 codegen_.classMetadataGlobals_[classType->getCodegenName()] = metaGlobal;
803 }
804
805 auto savedFn = codegen_.currentFn_;
806 auto savedBlock = codegen_.builder_.GetInsertBlock();
807
808 codegen_.currentFn_ = destroyHelper;
809 auto *destroyEntry =
810 llvm::BasicBlock::Create(codegen_.ctx_, "entry", destroyHelper);
811 auto *destroyReturnBB =
812 llvm::BasicBlock::Create(codegen_.ctx_, "arc.ret", destroyHelper);
813 codegen_.builder_.SetInsertPoint(destroyEntry);
814
815 auto *destroyRawObject = &*destroyHelper->arg_begin();
816 destroyRawObject->setName("object.raw");
817 auto *destroyIsNull = codegen_.builder_.CreateICmpEQ(
818 destroyRawObject,
819 llvm::ConstantPointerNull::get(llvm::cast<llvm::PointerType>(rawPtrTy)));
820 auto *destroyBodyBB =
821 llvm::BasicBlock::Create(codegen_.ctx_, "arc.destroy", destroyHelper);
822 codegen_.builder_.CreateCondBr(destroyIsNull, destroyReturnBB, destroyBodyBB);
823
824 codegen_.builder_.SetInsertPoint(destroyBodyBB);
825#if defined(ZAP_RUNTIME_INSTRUMENTATION)
826 codegen_.emitRuntimeOwnershipEvent("zap_runtime_ownership_note_destroy");
827#endif
828 auto *destroyTypedObject = codegen_.builder_.CreateBitCast(
829 destroyRawObject, llvm::PointerType::getUnqual(codegen_.ctx_), "object");
830 auto *aliveAddr = codegen_.builder_.CreateStructGEP(
831 objectTy, destroyTypedObject, kClassAliveIndex, "alive.addr");
832 codegen_.builder_.CreateStore(
833 llvm::ConstantInt::get(llvm::Type::getInt8Ty(codegen_.ctx_), 0),
834 aliveAddr);
835 auto *gcMarkAddr = codegen_.builder_.CreateStructGEP(
836 objectTy, destroyTypedObject, kClassGcMarkIndex, "gcmark.addr");
837 auto *gcMark = codegen_.builder_.CreateLoad(
838 llvm::Type::getInt8Ty(codegen_.ctx_), gcMarkAddr, "gcmark");
839 auto *finalizingMark = codegen_.builder_.CreateOr(
840 gcMark, llvm::ConstantInt::get(llvm::Type::getInt8Ty(codegen_.ctx_),
841 kClassGcFinalizingMask));
842 codegen_.builder_.CreateStore(finalizingMark, gcMarkAddr);
843 auto dtorIt =
844 codegen_.classDestructorFns_.find(classType->getCodegenName());
845 if (dtorIt != codegen_.classDestructorFns_.end()) {
846 auto *dtorPtr = codegen_.builder_.CreateBitCast(
847 destroyTypedObject, dtorIt->second->getArg(0)->getType(), "dtor.self");
848 codegen_.builder_.CreateCall(dtorIt->second, {dtorPtr});
849 }
850
851 for (size_t i = 0; i < classType->getFields().size(); ++i) {
852 const auto &field = classType->getFields()[i];
853 if (!field.type || !codegen_.containsManagedValues(field.type)) {
854 continue;
855 }
856 auto *fieldAddr = codegen_.builder_.CreateStructGEP(
857 objectTy, destroyTypedObject,
858 static_cast<unsigned>(i + kClassFieldStartIndex));
859 auto *fieldValue = codegen_.builder_.CreateLoad(
860 codegen_.toLLVMType(*field.type), fieldAddr, field.name);
861 codegen_.emitManagedRelease(fieldValue, field.type);
862 }
863
864 auto *markAfterDrop = codegen_.builder_.CreateLoad(
865 llvm::Type::getInt8Ty(codegen_.ctx_), gcMarkAddr, "gcmark.after.drop");
866 auto *finalizedMark = codegen_.builder_.CreateAnd(
867 markAfterDrop,
868 llvm::ConstantInt::get(llvm::Type::getInt8Ty(codegen_.ctx_),
869 static_cast<uint8_t>(~kClassGcFinalizingMask)));
870 codegen_.builder_.CreateStore(finalizedMark, gcMarkAddr);
871 codegen_.builder_.CreateBr(destroyReturnBB);
872
873 codegen_.builder_.SetInsertPoint(destroyReturnBB);
874 codegen_.builder_.CreateRetVoid();
875
876 codegen_.currentFn_ = releaseHelper;
877 auto *entry = llvm::BasicBlock::Create(codegen_.ctx_, "entry", releaseHelper);
878 auto *decrementedBB =
879 llvm::BasicBlock::Create(codegen_.ctx_, "arc.dec.check", releaseHelper);
880 auto *validReleaseBB =
881 llvm::BasicBlock::Create(codegen_.ctx_, "arc.dec", releaseHelper);
882 auto *underflowBB = llvm::BasicBlock::Create(codegen_.ctx_,
883 "arc.dec.underflow",
884 releaseHelper);
885 auto *destroyBB =
886 llvm::BasicBlock::Create(codegen_.ctx_, "arc.destroy", releaseHelper);
887 auto *deallocateBB =
888 llvm::BasicBlock::Create(codegen_.ctx_, "arc.deallocate", releaseHelper);
889 bool canBeCyclic =
890 codegen_.cyclicClasses_.count(classType->getCodegenName()) != 0;
891 auto *cycleBB =
892 canBeCyclic
893 ? llvm::BasicBlock::Create(codegen_.ctx_, "arc.cycle", releaseHelper)
894 : nullptr;
895 auto *returnBB =
896 llvm::BasicBlock::Create(codegen_.ctx_, "arc.ret", releaseHelper);
897 codegen_.builder_.SetInsertPoint(entry);
898
899 auto *rawObject = &*releaseHelper->arg_begin();
900 rawObject->setName("object.raw");
901 auto *isNull = codegen_.builder_.CreateICmpEQ(
902 rawObject,
903 llvm::ConstantPointerNull::get(llvm::cast<llvm::PointerType>(rawPtrTy)));
904 codegen_.builder_.CreateCondBr(isNull, returnBB, decrementedBB);
905
906 codegen_.builder_.SetInsertPoint(decrementedBB);
907 auto *typedObject = codegen_.builder_.CreateBitCast(
908 rawObject, llvm::PointerType::getUnqual(codegen_.ctx_), "object");
909 auto *countAddr = codegen_.builder_.CreateStructGEP(
910 objectTy, typedObject, kClassStrongCountIndex, "refcount.addr");
911 auto *count = codegen_.builder_.CreateLoad(
912 llvm::Type::getInt64Ty(codegen_.ctx_), countAddr, "refcount");
913 auto *isUnderflow = codegen_.builder_.CreateICmpSLE(
914 count, llvm::ConstantInt::get(llvm::Type::getInt64Ty(codegen_.ctx_), 0),
915 "refcount.underflowed");
916 codegen_.builder_.CreateCondBr(isUnderflow, underflowBB, validReleaseBB);
917
918 codegen_.builder_.SetInsertPoint(underflowBB);
919 emitRefcountFailure("zap_arc_strong_refcount_underflow");
920
921 codegen_.builder_.SetInsertPoint(validReleaseBB);
922 auto *nextCount = codegen_.builder_.CreateSub(
923 count, llvm::ConstantInt::get(llvm::Type::getInt64Ty(codegen_.ctx_), 1),
924 "refcount.next");
925 codegen_.builder_.CreateStore(nextCount, countAddr);
926#if defined(ZAP_RUNTIME_INSTRUMENTATION)
927 codegen_.emitRuntimeOwnershipEvent(
928 "zap_runtime_ownership_note_strong_release");
929#endif
930 auto *isZero = codegen_.builder_.CreateICmpEQ(
931 nextCount,
932 llvm::ConstantInt::get(llvm::Type::getInt64Ty(codegen_.ctx_), 0));
933 codegen_.builder_.CreateCondBr(isZero, destroyBB,
934 canBeCyclic ? cycleBB : returnBB);
935
936 codegen_.builder_.SetInsertPoint(destroyBB);
937 if (codegen_.functionMap_.count("zap_arc_remove_possible_root") == 0) {
938 auto *removeTy = llvm::FunctionType::get(
939 llvm::Type::getVoidTy(codegen_.ctx_), {rawPtrTy, rawPtrTy}, false);
940 auto *removeFn = llvm::Function::Create(
941 removeTy, llvm::Function::ExternalLinkage,
942 "zap_arc_remove_possible_root", *codegen_.module_);
943 codegen_.functionMap_["zap_arc_remove_possible_root"] = removeFn;
944 }
945 codegen_.builder_.CreateCall(
946 codegen_.functionMap_.at("zap_arc_remove_possible_root"),
947 {emitArcRuntimeContext(), rawObject});
948 auto *destroyAddr = codegen_.builder_.CreateStructGEP(
949 objectTy, typedObject, kClassDestroyFnIndex, "destroy.addr");
950 auto *destroyFn = codegen_.builder_.CreateLoad(
951 llvm::PointerType::getUnqual(codegen_.ctx_),
952 destroyAddr, "destroy.fn");
953 auto *destroyTy = llvm::FunctionType::get(
954 llvm::Type::getVoidTy(codegen_.ctx_), {rawPtrTy}, false);
955 codegen_.builder_.CreateCall(destroyTy, destroyFn, {rawObject});
956 auto *weakAddr = codegen_.builder_.CreateStructGEP(
957 objectTy, typedObject, kClassWeakCountIndex, "weakcount.addr");
958 auto *weakCount = codegen_.builder_.CreateLoad(
959 llvm::Type::getInt64Ty(codegen_.ctx_), weakAddr, "weakcount");
960 auto *isWeakZero = codegen_.builder_.CreateICmpEQ(
961 weakCount,
962 llvm::ConstantInt::get(llvm::Type::getInt64Ty(codegen_.ctx_), 0));
963 codegen_.builder_.CreateCondBr(isWeakZero, deallocateBB, returnBB);
964
965 codegen_.builder_.SetInsertPoint(deallocateBB);
966 codegen_.builder_.CreateCall(getOrCreateArcDeallocateFunction(),
967 {emitArcRuntimeContext(), rawObject});
968 codegen_.builder_.CreateBr(returnBB);
969
970 if (canBeCyclic) {
971 codegen_.builder_.SetInsertPoint(cycleBB);
972 if (codegen_.functionMap_.count("zap_arc_add_possible_root") == 0) {
973 auto *addTy = llvm::FunctionType::get(
974 llvm::Type::getVoidTy(codegen_.ctx_), {rawPtrTy, rawPtrTy}, false);
975 auto *addFn = llvm::Function::Create(
976 addTy, llvm::Function::ExternalLinkage, "zap_arc_add_possible_root",
977 *codegen_.module_);
978 codegen_.functionMap_["zap_arc_add_possible_root"] = addFn;
979 }
980 codegen_.builder_.CreateCall(
981 codegen_.functionMap_.at("zap_arc_add_possible_root"),
982 {emitArcRuntimeContext(), rawObject});
983 codegen_.builder_.CreateBr(returnBB);
984 }
985
986 codegen_.builder_.SetInsertPoint(returnBB);
987 codegen_.builder_.CreateRetVoid();
988
989 codegen_.currentFn_ = savedFn;
990 if (savedBlock) {
991 codegen_.builder_.SetInsertPoint(savedBlock);
992 }
993}
