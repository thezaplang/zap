89 signatureItems.reserve(signatures.size());
90 for (const auto &signature : signatures) {
91 JsonObject::List parameters;
92 parameters.reserve(signature.parameters.size());
93 for (const auto &param : signature.parameters) {
94 JsonObject::Object parameter;
95 parameter.emplace("label", JsonObject(param));
96 parameters.push_back(JsonObject(std::move(parameter)));
97 }
98
99 JsonObject::Object sig;
100 sig.emplace("label", JsonObject(signature.label));
101 sig.emplace("parameters", JsonObject(std::move(parameters)));
102 signatureItems.push_back(JsonObject(std::move(sig)));
103 }
104
105 int64_t clampedSignature =
106 signatures.empty()
107 ? 0
108 : std::max<int64_t>(0, std::min<int64_t>(activeSignature,
109 static_cast<int64_t>(
110 signatures.size() - 1)));
111 int64_t clampedParameter = 0;
112 if (!signatures.empty()) {
113 clampedParameter = std::max<int64_t>(
114 0, std::min<int64_t>(
115 activeParameter,
116 static_cast<int64_t>(
117 signatures[clampedSignature].parameters.empty()
118 ? 0
119 : signatures[clampedSignature].parameters.size() - 1)));
120 }
121
122 JsonObject::Object result;
123 result.emplace("signatures", JsonObject(std::move(signatureItems)));
124 result.emplace("activeSignature", JsonObject(clampedSignature));
125 result.emplace("activeParameter", JsonObject(clampedParameter));
126 return JsonObject(std::move(result));
