32void collectStrongReferencedClasses(
33 const std::shared_ptr<zir::Type> &type,
34 std::unordered_set<std::string> &classNames) {
35 if (!type) {
36 return;
37 }
38
39 if (type->getKind() == zir::TypeKind::Class) {
40 auto classType = std::static_pointer_cast<zir::ClassType>(type);
41 if (!classType->isWeak()) {
42 classNames.insert(classType->getCodegenName());
43 }
44 return;
45 }
46
47 if (type->getKind() == zir::TypeKind::Record) {
48 const auto &record = static_cast<const zir::RecordType &>(*type);
49 for (const auto &field : record.getFields()) {
50 collectStrongReferencedClasses(field.type, classNames);
51 }
52 } else if (type->getKind() == zir::TypeKind::Array) {
53 collectStrongReferencedClasses(
54 static_cast<const zir::ArrayType &>(*type).getBaseType(), classNames);
55 } else if (type->getKind() == zir::TypeKind::TaggedUnion) {
56 const auto &taggedUnion = static_cast<const zir::TaggedUnionType &>(*type);
57 for (const auto &variant : taggedUnion.getVariants()) {
58 collectStrongReferencedClasses(variant.payloadType, classNames);
59 }
60 }

63std::string lowerGccAsmTemplateToLLVM(const std::string &assembly) {
64 std::string lowered;
65 lowered.reserve(assembly.size());
66
67 for (size_t i = 0; i < assembly.size(); ++i) {
68 if (assembly[i] != '%' || i + 1 == assembly.size()) {
69 lowered.push_back(assembly[i]);
70 continue;
71 }
72
73 char next = assembly[i + 1];
74 if (std::isdigit(static_cast<unsigned char>(next))) {
75 lowered.push_back('$');
76 while (i + 1 < assembly.size() &&
77 std::isdigit(static_cast<unsigned char>(assembly[i + 1]))) {
78 lowered.push_back(assembly[++i]);
79 }
80 } else if (next == '%') {
81 lowered.push_back('%');
82 ++i;
83 } else {
84 lowered.push_back(assembly[i]);
85 }
86 }
87
88 return lowered;
89}

91std::string lowerX86GccConstraintToLLVM(const std::string &constraint) {
92 std::string lowered;
93 lowered.reserve(constraint.size() + 8);
94
95 for (size_t i = 0; i < constraint.size(); ++i) {
96 if (constraint[i] == '{') {
97 while (i < constraint.size()) {
98 lowered.push_back(constraint[i]);
99 if (constraint[i] == '}')
100 break;
101 ++i;
102 }
103 continue;
104 }
105
106 switch (constraint[i]) {
107 case 'a':
108 lowered += "{ax}";
109 break;
110 case 'b':
111 lowered += "{bx}";
112 break;
113 case 'c':
114 lowered += "{cx}";
115 break;
116 case 'd':
117 lowered += "{dx}";
118 break;
119 case 'S':
120 lowered += "{si}";
121 break;
122 case 'D':
123 lowered += "{di}";
124 break;
125 default:
126 lowered.push_back(constraint[i]);
127 break;
128 }
129 }
130
131 return lowered;

189void optimizeModule(llvm::Module &module, int optimizationLevel) {
190 if (optimizationLevel <= 0) {
191 return;
192 }
193
194 llvm::LoopAnalysisManager loopAnalysisManager;
195 llvm::FunctionAnalysisManager functionAnalysisManager;
196 llvm::CGSCCAnalysisManager cgsccAnalysisManager;
197 llvm::ModuleAnalysisManager moduleAnalysisManager;
198 llvm::PassBuilder passBuilder;
199 passBuilder.registerModuleAnalyses(moduleAnalysisManager);
200 passBuilder.registerCGSCCAnalyses(cgsccAnalysisManager);
201 passBuilder.registerFunctionAnalyses(functionAnalysisManager);
202 passBuilder.registerLoopAnalyses(loopAnalysisManager);
203 passBuilder.crossRegisterProxies(loopAnalysisManager, functionAnalysisManager,
204 cgsccAnalysisManager, moduleAnalysisManager);
205

276void LLVMCodeGen::computeCyclicClasses(const zir::Module &module) {
277 cyclicClasses_.clear();
278
279 std::unordered_map<std::string, std::shared_ptr<zir::ClassType>> classes;
280 for (const auto &type : module.getTypes()) {
281 if (type->getKind() == zir::TypeKind::Class) {
282 auto cls = std::static_pointer_cast<zir::ClassType>(type);
283 classes[cls->getCodegenName()] = cls;
284 }
285 }
286
287 std::unordered_map<std::string, std::vector<std::string>> subtypesOf;
288 for (const auto &[name, cls] : classes) {
289 for (auto cur = cls; cur; cur = cur->getBase()) {
290 subtypesOf[cur->getCodegenName()].push_back(name);
291 }
292 }
293
294 std::unordered_map<std::string, std::unordered_set<std::string>> edges;
295 for (const auto &[name, cls] : classes) {
296 auto &out = edges[name];
297 for (auto cur = cls; cur; cur = cur->getBase()) {
298 for (const auto &field : cur->getFields()) {
299 std::unordered_set<std::string> referencedClasses;
300 collectStrongReferencedClasses(field.type, referencedClasses);
301 for (const auto &referencedClass : referencedClasses) {
302 auto it = subtypesOf.find(referencedClass);
303 if (it == subtypesOf.end()) {
304 continue;
305 }
306 for (const auto &sub : it->second) {
307 out.insert(sub);
308 }
309 }
310 }
311 }
312 }
313
314 for (const auto &[start, _] : classes) {
315 std::vector<std::string> stack = {start};
316 std::unordered_set<std::string> seen;
317 bool cyclic = false;
318 while (!stack.empty()) {
319 std::string node = std::move(stack.back());
320 stack.pop_back();
321 auto it = edges.find(node);
322 if (it == edges.end()) {
323 continue;
324 }
325 for (const auto &next : it->second) {
326 if (next == start) {
327 cyclic = true;
328 break;
329 }
330 if (seen.insert(next).second) {
331 stack.push_back(next);
332 }
333 }
334 if (cyclic) {
335 break;
336 }
337 }
338 if (cyclic) {
339 cyclicClasses_.insert(start);
340 }
341 }

466llvm::Type *LLVMCodeGen::toLLVMType(const zir::Type &ty) {
467 switch (ty.getKind()) {
468 case zir::TypeKind::Void:
469 return llvm::Type::getVoidTy(ctx_);
470 case zir::TypeKind::Bool:
471 return llvm::Type::getInt1Ty(ctx_);
472 case zir::TypeKind::Char:
473 return llvm::Type::getInt8Ty(ctx_);
474 case zir::TypeKind::Int8:
475 case zir::TypeKind::UInt8:
476 return llvm::Type::getInt8Ty(ctx_);
477 case zir::TypeKind::Int16:
478 case zir::TypeKind::UInt16:
479 return llvm::Type::getInt16Ty(ctx_);
480 case zir::TypeKind::Int32:
481 case zir::TypeKind::UInt32:
482 return llvm::Type::getInt32Ty(ctx_);
483 case zir::TypeKind::Int:
484 case zir::TypeKind::UInt:
485 return nativeIntegerType();
486 case zir::TypeKind::Int64:
487 case zir::TypeKind::UInt64:
488 return llvm::Type::getInt64Ty(ctx_);
489 case zir::TypeKind::Float:
490 case zir::TypeKind::Float32:
491 return llvm::Type::getFloatTy(ctx_);
492 case zir::TypeKind::Float64:
493 return llvm::Type::getDoubleTy(ctx_);
494 case zir::TypeKind::Pointer: {
495 const auto &pt = static_cast<const zir::PointerType &>(ty);
496 auto *baseTy = toLLVMType(*pt.getBaseType());
497 if (baseTy->isVoidTy()) {
498 baseTy = llvm::Type::getInt8Ty(ctx_);
499 }
500 return llvm::PointerType::getUnqual(ctx_);
501 }
502 case zir::TypeKind::NullPtr:
503 return llvm::PointerType::getUnqual(ctx_);
504 case zir::TypeKind::Enum:
505 if (static_cast<const zir::EnumType &>(ty).hasReprC)
506 return llvm::Type::getInt32Ty(ctx_);
507 return nativeIntegerType();
508 case zir::TypeKind::TaggedUnion: {
509 const auto &tu = static_cast<const zir::TaggedUnionType &>(ty);
510 auto it = structCache_.find(tu.getCodegenName());
511 if (it != structCache_.end())
512 return it->second;
513
514 auto *structTy = llvm::StructType::create(ctx_, tu.getCodegenName());
515 structCache_[tu.getCodegenName()] = structTy;
516
517 llvm::Type *payloadStorageTy = llvm::Type::getInt8Ty(ctx_);
518 uint64_t payloadStorageSize = 0;
519 uint64_t payloadStorageAlign = 0;
520 for (const auto &variant : tu.getVariants()) {
521 if (!variant.payloadType)
522 continue;
523 auto *candidateTy = toLLVMType(*variant.payloadType);
524 auto candidateSize =
525 module_->getDataLayout().getTypeAllocSize(candidateTy);
526 auto candidateAlign =
527 module_->getDataLayout().getABITypeAlign(candidateTy).value();
528 if (candidateSize > payloadStorageSize ||
529 (candidateSize == payloadStorageSize &&
530 candidateAlign > payloadStorageAlign)) {
531 payloadStorageTy = candidateTy;
532 payloadStorageSize = candidateSize;
533 payloadStorageAlign = candidateAlign;
534 }
535 }
536
537 structTy->setBody({llvm::Type::getInt32Ty(ctx_), payloadStorageTy});
538 return structTy;
539 }
540 case zir::TypeKind::Record: {
541 const auto &rt = static_cast<const zir::RecordType &>(ty);
542 std::string cacheKey = rt.getCodegenName();
543 if (zir::isIntrinsicStringType(rt)) {
544 cacheKey = rt.getIntrinsicKind() == zir::IntrinsicTypeKind::String
545 ? "zap.intrinsic.String"
546 : "zap.intrinsic.StringView";
547 }
548
549 auto it = structCache_.find(cacheKey);
550 if (it != structCache_.end())
551 return it->second;
552
553 if (zir::isIntrinsicStringType(rt)) {
554 auto *structTy = llvm::StructType::create(ctx_, cacheKey);
555 structCache_[cacheKey] = structTy;
556 std::vector<llvm::Type *> fieldTypes;
557 fieldTypes.push_back(llvm::PointerType::getUnqual(ctx_));
558 fieldTypes.push_back(llvm::Type::getInt64Ty(ctx_));
559 structTy->setBody(fieldTypes, rt.isPacked);
560 return structTy;
561 }
562
563 auto *structTy = llvm::StructType::create(ctx_, rt.getCodegenName());
564 structCache_[cacheKey] = structTy;
565 std::vector<llvm::Type *> fieldTypes;
566 for (const auto &f : rt.getFields())
567 fieldTypes.push_back(toLLVMAggregateFieldType(f.type));
568 structTy->setBody(fieldTypes, rt.isPacked);
569 return structTy;
570 }
571 case zir::TypeKind::Class:
572 return llvm::PointerType::getUnqual(ctx_);
573 case zir::TypeKind::Array: {
574 const auto &at = static_cast<const zir::ArrayType &>(ty);
575 return llvm::ArrayType::get(toLLVMType(*at.getBaseType()), at.getSize());
576 }
577 case zir::TypeKind::FunctionPointer:
578 return llvm::PointerType::getUnqual(ctx_);
579 }
580 throw std::runtime_error("Unknown ZIR type: " + ty.toString());
581}

591llvm::FunctionType *LLVMCodeGen::buildFunctionType(const zir::Function &fn) {
592 std::vector<llvm::Type *> paramTypes;
593 if (!freestanding_ && fn.name == "main") {
594 paramTypes.push_back(llvm::Type::getInt32Ty(ctx_));
595 paramTypes.push_back(llvm::PointerType::getUnqual(ctx_));
596 }
597 for (const auto &param : fn.getArguments()) {
598 if (param->isVariadicPack()) {
599 paramTypes.push_back(llvm::Type::getInt32Ty(ctx_));
600 paramTypes.push_back(llvm::PointerType::getUnqual(ctx_));
601 continue;
602 }
603 paramTypes.push_back(toLLVMType(*param->getType()));
604 }
605 llvm::Type *retTy = toLLVMType(*fn.getReturnType());
606 if (fn.returnsRef)
607 retTy = llvm::PointerType::getUnqual(ctx_);
608 return llvm::FunctionType::get(retTy, paramTypes, fn.isCVariadic);
609}

685 params, false);
686 auto *fn = llvm::Function::Create(ft, llvm::Function::ExternalLinkage,
687 "string_concat_ptrlen", *module_);
688 concatIt = functionMap_.emplace("string_concat_ptrlen", fn).first;
689 }
690
691 auto *concatFn = concatIt->second;
692 auto *call = builder_.CreateCall(concatFn, {lhsPtr, lhsLen, rhsPtr, rhsLen});
693 auto *sumLen = builder_.CreateAdd(lhsLen, rhsLen);
694
695 auto *structTy = static_cast<llvm::StructType *>(toLLVMType(*resultType));
696 llvm::Value *res = llvm::UndefValue::get(structTy);
697 res = builder_.CreateInsertValue(res, call, {0});
698 res = builder_.CreateInsertValue(res, sumLen, {1});
699 return res;
700}

702void LLVMCodeGen::buildInlineAsmCall(
703 const std::string &assembly, const std::vector<std::string> &outConstraints,
704 const std::vector<llvm::Value *> &outAddrs,
705 const std::vector<llvm::Type *> &outValueTypes,
706 const std::vector<std::string> &inConstraints,
707 const std::vector<llvm::Value *> &inValues,
708 const std::vector<std::string> &clobbers) {
709 llvm::Triple triple(module_->getTargetTriple());
710 std::vector<std::string> parts;
711 for (const auto &c : outConstraints)
712 parts.push_back(lowerAsmConstraintToLLVM(c, triple));
713 for (const auto &c : inConstraints)
714 parts.push_back(lowerAsmConstraintToLLVM(c, triple));
715 for (const auto &c : clobbers)
716 parts.push_back("~{" + c + "}");
717
718 std::string constraints;
719 for (size_t i = 0; i < parts.size(); ++i) {
720 if (i > 0)
721 constraints += ",";
722 constraints += parts[i];
723 }
724
725 llvm::Type *retTy;
726 if (outValueTypes.empty()) {
727 retTy = llvm::Type::getVoidTy(ctx_);
728 } else if (outValueTypes.size() == 1) {
729 retTy = outValueTypes[0];
730 } else {
731 retTy = llvm::StructType::get(ctx_, outValueTypes);
732 }
733
734 std::vector<llvm::Type *> paramTypes;
735 for (auto *v : inValues)
736 paramTypes.push_back(v->getType());
737
738 auto *fnTy = llvm::FunctionType::get(retTy, paramTypes, /*isVarArg=*/false);
739 auto *inlineAsm =
740 llvm::InlineAsm::get(fnTy, lowerGccAsmTemplateToLLVM(assembly),
741 constraints, /*hasSideEffects=*/true);
742 auto *call = builder_.CreateCall(inlineAsm, inValues);
743
744 if (outValueTypes.size() == 1) {
745 builder_.CreateStore(call, outAddrs[0]);
746 } else if (outValueTypes.size() > 1) {
747 for (unsigned i = 0; i < outValueTypes.size(); ++i) {
748 auto *field = builder_.CreateExtractValue(call, {i});
749 builder_.CreateStore(field, outAddrs[i]);
750 }
751 }
752}
