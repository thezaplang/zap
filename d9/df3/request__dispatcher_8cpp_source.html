100 bool isCancelled(const std::optional<std::string> &key) {

109 if (!key) {

112 std::lock_guard lock(cancellationMutex_);

124 void execute(const JsonObject &request) {
125 auto method = getStringField(request, {"method"});
126 const JsonObject *id = getField(request, "id");
127 if (!method) {
128 if (id) {
129 server_.sendMessage(
130 makeErrorResponse(id, JsonRPC::InvalidRequest, "Missing method"));
131 }
132 return;
133 }
134
135 if (*method == "initialize") {
136 shutdownRequested_ = false;
137 workspace_.configure();
138 server_.sendMessage(makeResponse(id, makeCapabilities()));
139 } else if (*method == "initialized") {
140 return;
141 } else if (*method == "shutdown") {
142 shutdownRequested_ = true;
143 server_.sendMessage(makeResponse(id, JsonObject(nullptr)));
144 } else if (*method == "exit") {
145 return;
146 } else if (*method == "textDocument/didOpen") {
147 if (auto params = decodeOpenDocument(request)) {
148 if (auto path = uriToPath(params->uri)) {
149 workspace_.open(params->uri, *path, std::move(params->text),
150 params->version);
151 publishAnalysis(server_, workspace_.analyze(params->uri));
152 }
153 }
154 } else if (*method == "textDocument/didChange") {
155 if (auto params = decodeChangeDocument(request);
156 params && workspace_.contains(params->uri)) {
157 workspace_.update(params->uri, std::move(params->text),
158 params->version);
159 publishAnalysis(server_, workspace_.analyze(params->uri));
160 }
161 } else if (*method == "textDocument/didClose") {
162 if (auto uri = decodeCloseDocument(request)) {
163 workspace_.close(*uri);
164 server_.sendMessage(makePublishDiagnostics(*uri, {}));
165 }
166 } else if (*method == "workspace/didChangeWatchedFiles") {
167 if (auto changes = decodeWatchedFiles(request)) {
168 std::vector<std::filesystem::path> paths;
169 paths.reserve(changes->size());
170 for (const auto &change : *changes) {
171 if (auto path = uriToPath(change.uri)) {
172 paths.push_back(std::move(*path));
173 }
174 }
175 publishAnalysis(server_, workspace_.watchedFilesChanged(paths));
176 }
177 } else if (*method == "workspace/didChangeWorkspaceFolders") {
178 publishAnalysis(server_, workspace_.workspaceFoldersChanged());
179 } else if (*method == "textDocument/completion") {
180 if (id) {
181 if (auto context = documentRequestContext(workspace_, request)) {
182 auto items =
183 makeCompletionItems(context->uri, context->query.document->text,
184 *context->query.project, context->offset);
185 server_.sendMessage(makeResponse(id, JsonObject(std::move(items))));
186 } else {
187 server_.sendMessage(makeErrorResponse(id, JsonRPC::InvalidParams,
188 "Invalid document position"));
189 }
190 }
191 } else if (*method == "textDocument/definition") {
192 JsonObject result(nullptr);
193 if (id) {
194 if (auto context = documentRequestContext(workspace_, request)) {
195 if (auto symbol =
196 resolveDefinition(context->query.document->text, context->uri,
197 *context->query.project, context->offset)) {
198 if (auto source = workspace_.sourceForUri(symbol->uri)) {
199 result = makeLocation(symbol->uri, *source, symbol->span);
200 }
201 }
202 server_.sendMessage(makeResponse(id, std::move(result)));
203 } else {
204 server_.sendMessage(makeErrorResponse(id, JsonRPC::InvalidParams,
205 "Invalid document position"));
206 }
207 }
208 } else if (*method == "textDocument/hover") {
209 JsonObject result(nullptr);
210 if (id) {
211 if (auto context = documentRequestContext(workspace_, request)) {
212 if (auto hover =
213 resolveHover(context->query.document->text, context->uri,
214 *context->query.project, context->offset)) {
215 result = makeHover(*hover);
216 }
217 server_.sendMessage(makeResponse(id, std::move(result)));
218 } else {
219 server_.sendMessage(makeErrorResponse(id, JsonRPC::InvalidParams,
220 "Invalid document position"));
221 }
222 }
223 } else if (*method == "textDocument/signatureHelp") {
224 JsonObject result(nullptr);
225 if (id) {
226 if (auto context = documentRequestContext(workspace_, request)) {
227 int64_t activeParameter = 0;
228 auto signatures = resolveSignatures(
229 context->query.document->text, context->uri,
230 *context->query.project, context->offset, activeParameter);
231 if (!signatures.empty()) {
232 result = makeSignatureHelp(
233 signatures, chooseActiveSignature(signatures, activeParameter),
234 activeParameter);
235 }
236 server_.sendMessage(makeResponse(id, std::move(result)));
237 } else {
238 server_.sendMessage(makeErrorResponse(id, JsonRPC::InvalidParams,
239 "Invalid document position"));
240 }
241 }
242 } else if (id) {
243 server_.sendMessage(
244 makeErrorResponse(id, JsonRPC::MethodNotFound, "Method not found"));
245 }

248 void workerLoop() {
249 while (true) {
250 JsonObject request;
251 {
252 std::unique_lock lock(queueMutex_);
253 queueReady_.wait(lock, [this] { return stopping_ || !queue_.empty(); });
254 if (queue_.empty()) {
255 return;
256 }
257 request = std::move(queue_.front());
258 queue_.pop_front();
259 }
260
261 const JsonObject *id = getField(request, "id");
262 auto key = requestKey(id);
263 if (isCancelled(key)) {
264 finishRequest(key);
265 sendCancelled(id);
266 continue;
267 }
268
269 execute(request);
270 if (finishRequest(key)) {
271 sendCancelled(id);
272 } else {
273 server_.send();
274 }
275 }
276 }

288 bool submit(JsonObject request) {
289 auto method = getStringField(request, {"method"});
290 if (method && *method == "$/cancelRequest") {
291 auto key = requestKey(getPath(request, {"params", "id"}));
292 if (key) {
293 std::lock_guard lock(cancellationMutex_);
294 if (outstanding_.find(*key) != outstanding_.end()) {
295 cancelled_.insert(std::move(*key));
296 }
297 }
298 return true;
299 }
300
301 const bool keepReading = !method || *method != "exit";
302 if (auto key = requestKey(getField(request, "id"))) {
303 std::lock_guard lock(cancellationMutex_);
304 outstanding_.insert(std::move(*key));
305 }
306 {
307 std::lock_guard lock(queueMutex_);
308 queue_.push_back(std::move(request));
309 }
310 queueReady_.notify_one();
311 return keepReading;
312 }
313
314 void stop() {
315 {
316 std::lock_guard lock(queueMutex_);
317 if (stopping_) {
318 return;
319 }
320 stopping_ = true;
321 }
322 queueReady_.notify_one();
323 if (worker_.joinable()) {
324 worker_.join();
325 }
326 }
327
328 bool shutdownRequested() const { return shutdownRequested_; }
329};

333int runRequestDispatcher() {
334 std::setvbuf(stdout, nullptr, _IONBF, 0);
335 Server server;
336 RequestScheduler scheduler(server);
337 std::string line;
338
339 while (true) {
340 std::string message = server.processMessage(line);
341 if (message.empty()) {
342 break;
343 }
344 JsonRPC rpc(message);
345 if (!scheduler.submit(rpc.object())) {
346 break;
347 }
348 }
349
350 scheduler.stop();
351 return scheduler.shutdownRequested() ? 0 : 1;
352}
