26std::shared_ptr<zir::Type>
27joinNumericTypes(const std::shared_ptr<zir::Type> &left,
28 const std::shared_ptr<zir::Type> &right,
29 uint16_t nativeBitWidth) {
30 if (left->isFloatingPoint() || right->isFloatingPoint()) {
31 if (left->getKind() == zir::TypeKind::Float64 ||
32 right->getKind() == zir::TypeKind::Float64) {
33 return primitive(zir::TypeKind::Float64);
34 }
35 auto kind = left->getKind() == zir::TypeKind::Float ||
36 right->getKind() == zir::TypeKind::Float
37 ? zir::TypeKind::Float
38 : zir::TypeKind::Float32;
39 return primitive(kind);
40 }
41
42 int leftWidth = bitWidth(left, nativeBitWidth);
43 int rightWidth = bitWidth(right, nativeBitWidth);
44 int width = std::max(leftWidth, rightWidth);
45 bool leftUnsigned = left->isUnsigned();
46 bool rightUnsigned = right->isUnsigned();
47
48 bool useUnsigned = leftUnsigned && rightUnsigned;
49 if (leftUnsigned != rightUnsigned) {
50 int unsignedWidth = leftUnsigned ? leftWidth : rightWidth;
51 int signedWidth = leftUnsigned ? rightWidth : leftWidth;
52 useUnsigned = unsignedWidth >= signedWidth;
53 }
54
55 auto leftInfo = *zir::numericTypeInfo(left->getKind());
56 auto rightInfo = *zir::numericTypeInfo(right->getKind());
57 bool preferNative = width == nativeBitWidth &&
58 ((leftInfo.isNative && leftUnsigned == useUnsigned) ||
59 (rightInfo.isNative && rightUnsigned == useUnsigned));
60 if (preferNative) {
61 return primitive(useUnsigned ? zir::TypeKind::UInt : zir::TypeKind::Int);
62 }
63
64 if (useUnsigned) {
65 if (width <= 8) {
66 return primitive(zir::TypeKind::UInt8);
67 }
68 if (width <= 16) {
69 return primitive(zir::TypeKind::UInt16);
70 }
71 if (width <= 32) {
72 return primitive(zir::TypeKind::UInt32);
73 }
74 return primitive(zir::TypeKind::UInt64);
75 }
76
77 if (width <= 8) {
78 return primitive(zir::TypeKind::Int8);
79 }
80 if (width <= 16) {
81 return primitive(zir::TypeKind::Int16);
82 }
83 if (width <= 32) {
84 return primitive(zir::TypeKind::Int32);
85 }
86 return primitive(zir::TypeKind::Int64);
87}

177bool ConversionClassifier::isSubtype(
178 const std::shared_ptr<zir::Type> &source,
179 const std::shared_ptr<zir::Type> &target) const {
180 if (!source || !target) {
181 return false;
182 }
183 if (types_.same(source, target)) {
184 return true;
185 }
186 if (source->getKind() != zir::TypeKind::Class ||
187 target->getKind() != zir::TypeKind::Class) {
188 return false;
189 }
190
191 const auto sourceClass = std::static_pointer_cast<zir::ClassType>(source);
192 const auto targetClass = std::static_pointer_cast<zir::ClassType>(target);
193 return sourceClass->isWeak() == targetClass->isWeak() &&
194 hasClassAncestor(sourceClass, targetClass->getCodegenName());

198 const std::shared_ptr<zir::Type> &source,
199 const std::shared_ptr<zir::Type> &target) const {
200 if (!source || !target) {
201 return std::nullopt;
202 }
203 const TypePair key{types_.intern(source), types_.intern(target)};
204 if (auto cached = implicitCache_.find(key); cached != implicitCache_.end()) {
205 auto result = cached->second;
206 if (result) {
207 result->targetType = target;
208 }
209 return result;
210 }
211 auto result = classifyImplicitUncached(source, target);
212 implicitCache_.emplace(key, result);

216std::optional<Conversion> ConversionClassifier::classifyImplicitUncached(
217 const std::shared_ptr<zir::Type> &source,
218 const std::shared_ptr<zir::Type> &target) const {
219 if (types_.same(source, target)) {
220 return conversion(ConversionKind::Identity, ConversionRank::Exact, target);
221 }
222
223 if (source->getKind() == zir::TypeKind::NullPtr) {
224 if (target->getKind() == zir::TypeKind::Pointer) {
225 return conversion(ConversionKind::NullToPointer,
226 ConversionRank::NullPointer, target);
227 }
228 if (target->getKind() == zir::TypeKind::Class) {
229 return conversion(ConversionKind::NullToClass,
230 ConversionRank::Structural, target);
231 }
232 }
233
234 if (zir::isIntrinsicStringType(source) &&
235 zir::isIntrinsicStringType(target)) {
236 if (zir::isIntrinsicStringViewType(target)) {
237 return conversion(ConversionKind::StringToView, ConversionRank::Exact,
238 target);
239 }
240 return conversion(ConversionKind::StringToOwned, ConversionRank::Promotion,
241 target);
242 }
243
244 if (zir::isIntrinsicStringType(source) &&
245 target->getKind() == zir::TypeKind::Pointer) {
246 const auto &pointer = static_cast<const zir::PointerType &>(*target);
247 if (pointer.getBaseType()->getKind() == zir::TypeKind::Char) {
248 return conversion(ConversionKind::StringToCharPointer,
249 ConversionRank::Structural, target);
250 }
251 }
252
253 if (source->getKind() == zir::TypeKind::Class &&
254 target->getKind() == zir::TypeKind::Class) {
255 auto sourceClass = std::static_pointer_cast<zir::ClassType>(source);
256 auto targetClass = std::static_pointer_cast<zir::ClassType>(target);
257 if (isSubtype(source, target)) {
258 return conversion(ConversionKind::ClassUpcast,
259 ConversionRank::Structural, target);
260 }
261 if (!sourceClass->isWeak() && targetClass->isWeak() &&
262 hasClassAncestor(sourceClass, targetClass->getCodegenName())) {
263 return conversion(ConversionKind::StrongToWeak,
264 ConversionRank::Structural, target);
265 }
266 }
267
268 if (source->getKind() == zir::TypeKind::Array &&
269 isVariadicViewType(target)) {
270 const auto &array = static_cast<const zir::ArrayType &>(*source);
271 const auto &view = static_cast<const zir::RecordType &>(*target);
272 if (!view.getFields().empty() && view.getFields()[0].type &&
273 view.getFields()[0].type->getKind() == zir::TypeKind::Pointer) {
274 const auto &data =
275 static_cast<const zir::PointerType &>(*view.getFields()[0].type);
276 if (types_.same(array.getBaseType(), data.getBaseType())) {
277 return conversion(ConversionKind::ArrayToVariadicView,
278 ConversionRank::Structural, target);
279 }
280 }
281 }
282
283 auto sourceFailable = zir::getFailableTypeLayout(source);
284 auto targetFailable = zir::getFailableTypeLayout(target);
285 if (sourceFailable && targetFailable &&
286 classifyImplicit(sourceFailable->valueType, targetFailable->valueType) &&
287 classifyImplicit(sourceFailable->errorType, targetFailable->errorType)) {
288 return conversion(ConversionKind::Failable, ConversionRank::Structural,
289 target);
290 }
291
292 if (source->getKind() == zir::TypeKind::Enum && target->isInteger()) {
293 const auto &enumType = static_cast<const zir::EnumType &>(*source);
294 int sourceWidth = enumType.hasReprC
295 ? 32
296 : targetInfo_.nativeIntegerBitWidth();
297 int targetWidth =
298 bitWidth(target, targetInfo_.nativeIntegerBitWidth());
299 auto targetInfo = zir::numericTypeInfo(target->getKind());
300 auto rank = ConversionRank::Narrowing;
301 if (target->isUnsigned()) {
302 rank = ConversionRank::Lossy;
303 } else if (targetInfo && targetInfo->isNative) {
304 rank = ConversionRank::Promotion;
305 } else if (targetWidth >= sourceWidth) {
306 // A fixed-width integer can represent the enum, but the enum's natural
307 // representation is the native signed integer.
308 rank = ConversionRank::Numeric;
309 }
310 return conversion(ConversionKind::EnumToInteger, rank, target);
311 }
312
313 if (source->isFloatingPoint() && target->isFloatingPoint()) {
314 bool widening = bitWidth(target, targetInfo_.nativeIntegerBitWidth()) >=
315 bitWidth(source, targetInfo_.nativeIntegerBitWidth());
316 return conversion(widening ? ConversionKind::FloatingWidening
317 : ConversionKind::FloatingNarrowing,
318 widening ? ConversionRank::Promotion
319 : ConversionRank::Lossy,
320 target);
321 }
322
323 if (isSignedInteger(source) && isSignedInteger(target)) {
324 bool widening = bitWidth(target, targetInfo_.nativeIntegerBitWidth()) >=
325 bitWidth(source, targetInfo_.nativeIntegerBitWidth());
326 return conversion(widening ? ConversionKind::SignedWidening
327 : ConversionKind::SignedNarrowing,
328 widening ? ConversionRank::Promotion
329 : ConversionRank::Narrowing,
330 target);
331 }
332
333 if (source->isInteger() && source->isUnsigned() && target->isInteger() &&
334 target->isUnsigned()) {
335 bool widening = bitWidth(target, targetInfo_.nativeIntegerBitWidth()) >=
336 bitWidth(source, targetInfo_.nativeIntegerBitWidth());
337 return conversion(widening ? ConversionKind::UnsignedWidening
338 : ConversionKind::UnsignedNarrowing,
339 widening ? ConversionRank::Promotion
340 : ConversionRank::Narrowing,
341 target);
342 }
343
344 if (source->isInteger() && target->isFloatingPoint()) {
345 bool preservesWidth =
346 bitWidth(target, targetInfo_.nativeIntegerBitWidth()) >=
347 bitWidth(source, targetInfo_.nativeIntegerBitWidth());
348 return conversion(ConversionKind::IntegerToFloat,
349 preservesWidth ? ConversionRank::Numeric
350 : ConversionRank::Lossy,
351 target);
352 }
353
354 if (source->isFloatingPoint() && target->isInteger()) {
355 return conversion(ConversionKind::FloatToInteger,
356 ConversionRank::FloatToInteger, target);
357 }
358
359 if (source->isInteger() && target->isInteger()) {
360 return conversion(ConversionKind::SignednessChange, ConversionRank::Lossy,
361 target);
362 }
363
364 return std::nullopt;
365}

379std::optional<TypeJoin> ConversionClassifier::joinTypes(
380 const std::shared_ptr<zir::Type> &left,
381 const std::shared_ptr<zir::Type> &right) const {
382 if (!left || !right) {
383 return std::nullopt;
384 }
385 if (types_.same(left, right)) {
386 return makeJoin(left, right, left);
387 }
388
389 bool leftNumeric = left->isInteger() || left->isFloatingPoint();
390 bool rightNumeric = right->isInteger() || right->isFloatingPoint();
391 if (leftNumeric && rightNumeric) {
392 return makeJoin(left, right,
393 joinNumericTypes(left, right,
394 targetInfo_.nativeIntegerBitWidth()));
395 }
396
397 if (zir::isIntrinsicStringType(left) &&
398 zir::isIntrinsicStringType(right)) {
399 auto target = zir::makeStringViewType();
400 if (zir::isIntrinsicStringViewType(left)) {
401 target = std::static_pointer_cast<zir::RecordType>(left);
402 } else if (zir::isIntrinsicStringViewType(right)) {
403 target = std::static_pointer_cast<zir::RecordType>(right);
404 }
405 return makeJoin(left, right, target);
406 }
407
408 if (left->getKind() == zir::TypeKind::Class &&
409 right->getKind() == zir::TypeKind::Class) {
410 auto leftClass = std::static_pointer_cast<zir::ClassType>(left);
411 auto rightClass = std::static_pointer_cast<zir::ClassType>(right);
412 std::shared_ptr<zir::ClassType> commonClass;
413 for (auto leftAncestor = leftClass; leftAncestor && !commonClass;
414 leftAncestor = leftAncestor->getBase()) {
415 for (auto rightAncestor = rightClass; rightAncestor;
416 rightAncestor = rightAncestor->getBase()) {
417 if (leftAncestor->getCodegenName() ==
418 rightAncestor->getCodegenName()) {
419 commonClass = leftAncestor;
420 break;
421 }
422 }
423 }
424 if (commonClass) {
425 bool weak = leftClass->isWeak() || rightClass->isWeak();
426 std::shared_ptr<zir::ClassType> target = commonClass;
427 if (weak && !target->isWeak()) {
428 if (leftClass->isWeak() &&
429 leftClass->getCodegenName() == target->getCodegenName()) {
430 target = leftClass;
431 } else if (rightClass->isWeak() &&
432 rightClass->getCodegenName() == target->getCodegenName()) {
433 target = rightClass;
434 } else {
435 target = std::make_shared<zir::ClassType>(*commonClass);
436 target->setWeak(true);
437 }
438 }
439 return makeJoin(left, right, target);
440 }
441 }
442
443 auto leftFailable = zir::getFailableTypeLayout(left);
444 auto rightFailable = zir::getFailableTypeLayout(right);
445 if (leftFailable && rightFailable) {
446 auto valueJoin =
447 joinTypes(leftFailable->valueType, rightFailable->valueType);
448 auto errorJoin =
449 joinTypes(leftFailable->errorType, rightFailable->errorType);
450 if (valueJoin && errorJoin) {
451 return makeJoin(left, right, zir::makeFailableRecordType(
452 valueJoin->type, errorJoin->type));
453 }
454 }
455
456 auto leftToRight = classifyImplicit(left, right);
457 auto rightToLeft = classifyImplicit(right, left);
458 if (leftToRight && !rightToLeft) {
459 return makeJoin(left, right, right);
460 }
461 if (rightToLeft && !leftToRight) {
462 return makeJoin(left, right, left);
463 }
464 if (leftToRight && rightToLeft) {
465 if (leftToRight->cost() < rightToLeft->cost()) {
466 return makeJoin(left, right, right);
467 }
468 if (rightToLeft->cost() < leftToRight->cost()) {
469 return makeJoin(left, right, left);
470 }
471 }
472 return std::nullopt;
473}

475std::optional<Conversion> ConversionClassifier::classifyExplicit(
476 const std::shared_ptr<zir::Type> &source,
477 const std::shared_ptr<zir::Type> &target) const {
478 if (auto implicit = classifyImplicit(source, target)) {
479 return implicit;
480 }
481 if (!source || !target) {
482 return std::nullopt;
483 }
484
485 bool sourceChar = source->getKind() == zir::TypeKind::Char;
486 bool targetChar = target->getKind() == zir::TypeKind::Char;
487 if ((sourceChar && target->isInteger()) ||
488 (source->isInteger() && targetChar)) {
489 return conversion(ConversionKind::ExplicitCharInteger,
490 ConversionRank::Explicit, target);
491 }
492 if (source->getKind() == zir::TypeKind::Enum && target->isInteger()) {
493 return conversion(ConversionKind::EnumToInteger, ConversionRank::Explicit,
494 target);
495 }
496
497 bool sourcePointer = source->getKind() == zir::TypeKind::Pointer;
498 bool targetPointer = target->getKind() == zir::TypeKind::Pointer;
499 if ((sourcePointer || source->getKind() == zir::TypeKind::NullPtr) &&
500 targetPointer) {
501 return conversion(ConversionKind::ExplicitPointer,
502 ConversionRank::Explicit, target);
503 }
504 if (zir::isIntrinsicStringType(source) && targetPointer) {
505 return conversion(ConversionKind::ExplicitStringPointer,
506 ConversionRank::Explicit, target);
507 }
508 if (sourcePointer && zir::isIntrinsicStringType(target)) {
509 const auto &pointer = static_cast<const zir::PointerType &>(*source);
510 auto baseKind = pointer.getBaseType()->getKind();
511 if (baseKind == zir::TypeKind::Void || baseKind == zir::TypeKind::Char) {
512 return conversion(ConversionKind::ExplicitPointerString,
513 ConversionRank::Explicit, target);
514 }
515 }
516 if ((sourcePointer && target->isInteger()) ||
517 (source->isInteger() && targetPointer)) {
518 return conversion(ConversionKind::ExplicitPointerInteger,
519 ConversionRank::Explicit, target);
520 }
521 return std::nullopt;
522}
