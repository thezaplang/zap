47 statementStack_.pop();

54 }
55

58 if (!expressionStack_.empty()) {

60 expressionStack_.pop();

118 bool expressionHadDiagnostic = false;
119 if (node.returnValue) {

122 node.returnValue.get(),
123 currentFunction_ ? currentFunction_->returnType : nullptr);
124 expressionHadDiagnostic = (hadError_ != hadErrorBefore);
125 if (!expr) {
126 // The return expression already produced a diagnostic.
127 // Avoid cascading with a secondary "received Void" return-type error.
128 statementStack_.push(std::make_unique<BoundReturnStatement>(
129 nullptr, currentFunction_ && currentFunction_->returnsRef));
130 return;
131 }
132 }
133
134 if (currentFunction_ && currentFunction_->returnType) {

136 auto actualType =

138 : std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
139

141 auto expectedValueType = failableValueType(expectedType);

144 if (!conversion) {

151 expr = makeFailableValueExpr(std::move(expr), expectedType);
152 } else {

155 }
156 } else {

158 if (!conversion) {

161 std::move(expr),
