89OwnerMap collectValueOwners(
90 const Module &module, const Function &function,
91 const OwnerSet &localStorage,
92 std::unordered_map<const Value *, std::shared_ptr<Value>> &derivedFrom,
93 OwnerMap owners = {}) {
94 const auto resultSource = function.resultBorrow.sourceParameter();
95 for (size_t i = 0; i < function.getArguments().size(); ++i) {
96 const auto &argument = function.getArguments()[i];
97 const bool noescapeView =
98 argument &&
99 argument->getParameterEscape() == ParameterEscape::NoEscape &&
100 argument->getType() &&
101 argument->getType()->getIntrinsicKind() ==
102 IntrinsicTypeKind::StringView;
103 if (argument &&
104 (noescapeView || (resultSource && *resultSource == i))) {
105 owners[argument.get()].insert(argument.get());
106 }
107 }
108 bool changed = true;
109 while (changed) {
110 changed = false;
111 for (const auto &blockOwner : function.getBlocks()) {
112 if (!blockOwner) {
113 continue;
114 }
115 for (const auto &instruction : blockOwner->getInstructions()) {
116 if (!instruction) {
117 continue;
118 }
119 switch (instruction->getOpCode()) {
120 case OpCode::Borrow: {
121 const auto &borrow = static_cast<const BorrowInst &>(*instruction);
122 if (borrow.getResult()) {
123 changed = addBorrowSourcesFromValue(
124 owners, borrow.getResult().get(), borrow.getOwner()) ||
125 changed;
126 }
127 break;
128 }
129 case OpCode::Phi: {
130 const auto &phi = static_cast<const PhiInst &>(*instruction);
131 if (!phi.getResult()) {
132 break;
133 }
134 for (const auto &[_, incoming] : phi.getIncoming()) {
135 changed =
136 addOwnersFromValue(owners, phi.getResult().get(), incoming) ||
137 changed;
138 }
139 break;
140 }
141 case OpCode::Store: {
142 const auto &store = static_cast<const StoreInst &>(*instruction);
143 if (store.getDestination() &&
144 localStorage.count(store.getDestination().get()) != 0) {
145 changed = addOwnersFromValue(owners, store.getDestination().get(),
146 store.getSource()) ||
147 changed;
148 }
149 break;
150 }
151 case OpCode::Load: {
152 const auto &load = static_cast<const LoadInst &>(*instruction);
153 if (load.getResult()) {
154 changed = addOwnersFromValue(owners, load.getResult().get(),
155 load.getSource()) ||
156 changed;
157 }
158 break;
159 }
160 case OpCode::GetElementPtr: {
161 const auto &gep =
162 static_cast<const GetElementPtrInst &>(*instruction);
163 if (gep.getResult()) {
164 changed = addOwnersFromValue(owners, gep.getResult().get(),
165 gep.getPointer()) ||
166 changed;
167 }
168 break;
169 }
170 case OpCode::Cast: {
171 const auto &cast = static_cast<const CastInst &>(*instruction);
172 if (cast.getResult() && cast.getSource() &&
173 cast.getResult()->getType()->getIntrinsicKind() ==
174 IntrinsicTypeKind::StringView &&
175 cast.getSource()->getType()->getIntrinsicKind() ==
176 IntrinsicTypeKind::StringView) {
177 derivedFrom[cast.getResult().get()] = cast.getSource();
178 changed = addOwnersFromValue(owners, cast.getResult().get(),
179 cast.getSource()) ||
180 changed;
181 }
182 break;
183 }
184 case OpCode::Call: {
185 const auto &call = static_cast<const CallInst &>(*instruction);
186 const auto resultBorrow =
187 resolveCallResultBorrowContract(module, call);
188 if (!call.getResult() || !resultBorrow.hasSource()) {
189 break;
190 }
191 const size_t sourceIndex = *resultBorrow.sourceParameter();
192 if (sourceIndex < call.getArguments().size()) {
193 changed = addBorrowSourcesFromValue(
194 owners, call.getResult().get(),
195 call.getArguments()[sourceIndex]) ||
196 changed;
197 }
198 break;
199 }
200 default:
201 break;
202 }
203 }
204 }
205 }
206 return owners;

245StorageStates analyzeStorageOwners(const Function &function,
246 const ControlFlowGraph &cfg,
247 const OwnerSet &localStorage,
248 const OwnerMap &valueOwners,
249 OwnerMap &loadOwners) {
250 StorageStates entryStates;
251 StorageStates exitStates;
252 bool changed = true;
253 while (changed) {
254 changed = false;
255 for (const auto &blockOwner : function.getBlocks()) {
256 if (!blockOwner) {
257 continue;
258 }
259 StorageState entry;
260 for (const auto *predecessor : cfg.predecessors().at(blockOwner.get())) {
261 unionStorageStates(entry, exitStates[predecessor]);
262 }
263 if (entryStates[blockOwner.get()] != entry) {
264 entryStates[blockOwner.get()] = entry;
265 changed = true;
266 }
267
268 StorageState state = std::move(entry);
269 for (const auto &instruction : blockOwner->getInstructions()) {
270 if (!instruction) {
271 continue;
272 }
273 if (instruction->getOpCode() == OpCode::Load) {
274 const auto &load = static_cast<const LoadInst &>(*instruction);
275 if (load.getResult() && load.getSource() &&
276 localStorage.count(load.getSource().get()) != 0) {
277 loadOwners[load.getResult().get()] = state[load.getSource().get()];
278 }
279 continue;
280 }
281 if (instruction->getOpCode() != OpCode::Store) {
282 continue;
283 }
284 const auto &store = static_cast<const StoreInst &>(*instruction);
285 if (!store.getDestination() ||
286 localStorage.count(store.getDestination().get()) == 0) {
287 continue;
288 }
289 OwnerSet owners;
290 if (store.getSource()) {
291 if (tracksOwnership(store.getSource())) {
292 owners.insert(store.getSource().get());
293 }
294 const auto ownerIt = valueOwners.find(store.getSource().get());
295 if (ownerIt != valueOwners.end()) {
296 owners = ownerIt->second;
297 }
298 }
299 if (state[store.getDestination().get()] != owners) {
300 state[store.getDestination().get()] = std::move(owners);
301 }
302 }
303 if (exitStates[blockOwner.get()] != state) {
304 exitStates[blockOwner.get()] = std::move(state);
305 changed = true;
306 }
307 }
308 }
309 return exitStates;
310}

323BorrowProvenance::OwnerSet BorrowProvenance::ownersAtDefinition(
324 const std::shared_ptr<Value> &value) const {
325 OwnerSet result;
326 OwnerSet visited;
327 std::vector<std::shared_ptr<Value>> pending{value};
328 while (!pending.empty()) {
329 auto current = std::move(pending.back());
330 pending.pop_back();
331 if (!current || !visited.insert(current.get()).second) {
332 continue;
333 }
334 const auto load = loadOwners_.find(current.get());
335 if (load != loadOwners_.end()) {
336 result.insert(load->second.begin(), load->second.end());
337 continue;
338 }

340 if (derived != derivedFrom_.end()) {
341 pending.push_back(derived->second);
342 continue;
343 }
344 const auto phi = phiIncoming_.find(current.get());
345 if (phi != phiIncoming_.end()) {
346 pending.insert(pending.end(), phi->second.begin(), phi->second.end());
347 continue;
348 }
349 const auto &owners = ownersOf(current);

351 }

356BorrowProvenance::ownersOnEdge(const std::shared_ptr<Value> &value,
357 const BasicBlock &source,
358 const BasicBlock &destination) const {
359 std::shared_ptr<Value> current = value;
360 OwnerSet visited;
361 while (current && visited.insert(current.get()).second) {
362 for (const auto &instruction : destination.getInstructions()) {
363 if (!instruction) {
364 continue;
365 }
366 if (instruction->getOpCode() == OpCode::Phi) {
367 const auto &phi = static_cast<const PhiInst &>(*instruction);
368 if (phi.getResult() != current) {
369 continue;
370 }
371 for (const auto &[label, incoming] : phi.getIncoming()) {
372 if (label == source.label) {
373 return ownersOf(incoming);
374 }
375 }
376 return {};
377 }
378 if (instruction->getOpCode() == OpCode::Load) {
379 const auto &load = static_cast<const LoadInst &>(*instruction);
380 if (load.getResult() == current &&
381 isEntryLoad(destination, load.getResult())) {
382 return ownersStoredAtExit(source, load.getSource());
383 }
384 }
385 }
386 const auto derived = derivedFrom_.find(current.get());
387 if (derived == derivedFrom_.end()) {
388 return ownersOf(current);
389 }
390 current = derived->second;
391 }
392 return {};

420BorrowProvenance analyzeBorrowProvenance(const Module &module,
421 const Function &function,
422 const ControlFlowGraph &cfg) {
423 BorrowProvenance result;
424 result.localStorage_ = collectLocalStorage(function);
425 result.owners_ = collectValueOwners(module, function, result.localStorage_,
426 result.derivedFrom_);
427 for (const auto &blockOwner : function.getBlocks()) {
428 if (!blockOwner) {
429 continue;
430 }
431 for (const auto &instruction : blockOwner->getInstructions()) {
432 if (!instruction || instruction->getOpCode() != OpCode::Phi) {
433 continue;
434 }
435 const auto &phi = static_cast<const PhiInst &>(*instruction);
436 if (!phi.getResult()) {
437 continue;
438 }
439 auto &incoming = result.phiIncoming_[phi.getResult().get()];
440 for (const auto &[_, value] : phi.getIncoming()) {
441 incoming.push_back(value);
442 }
443 }
444 }
445 collectEntryLoads(function, result.localStorage_, result.entryLoads_);
446 result.exitStorage_ = analyzeStorageOwners(
447 function, cfg, result.localStorage_, result.owners_, result.loadOwners_);
448 for (const auto &blockOwner : function.getBlocks()) {
449 if (!blockOwner) {
450 continue;
451 }
452 for (const auto &instruction : blockOwner->getInstructions()) {
453 if (!instruction || instruction->getOpCode() != OpCode::Borrow) {
454 continue;
455 }
456 const auto &borrow = static_cast<const BorrowInst &>(*instruction);
457 const auto loadedOwners = result.loadOwners_.find(borrow.getOwner().get());
458 if (borrow.getResult() && loadedOwners != result.loadOwners_.end()) {
459 addOwners(result.owners_, borrow.getResult().get(),
460 loadedOwners->second);
461 }
462 }
463 }
464 result.owners_ = collectValueOwners(
465 module, function, result.localStorage_, result.derivedFrom_,
466 std::move(result.owners_));
467 result.loadOwners_.clear();
468 result.exitStorage_ = analyzeStorageOwners(
469 function, cfg, result.localStorage_, result.owners_, result.loadOwners_);
470 return result;
