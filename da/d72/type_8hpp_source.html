465inline bool containsManagedValues(const std::shared_ptr<Type> &type) {
466 if (!type) {
467 return false;
468 }
469 if (type->getKind() == TypeKind::Class ||
470 type->getIntrinsicKind() == IntrinsicTypeKind::String) {
471 return true;
472 }
473 if (type->getKind() == TypeKind::Record) {
474 const auto &record = static_cast<const RecordType &>(*type);
475 for (const auto &field : record.getFields()) {
476 if (containsManagedValues(field.type)) {
477 return true;
478 }
479 }
480 return false;
481 }
482 if (type->getKind() == TypeKind::Array) {
483 const auto &array = static_cast<const ArrayType &>(*type);
484 return containsManagedValues(array.getBaseType());
485 }
486 if (type->getKind() == TypeKind::TaggedUnion) {
487 const auto &taggedUnion = static_cast<const TaggedUnionType &>(*type);
488 for (const auto &variant : taggedUnion.getVariants()) {
489 if (containsManagedValues(variant.payloadType)) {
490 return true;
491 }
492 }
493 }
494 return false;
495}
