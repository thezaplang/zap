781 classMetadataGlobals_.at(name), llvm::PointerType::getUnqual(ctx_));
782 auto *isCandidate =
783 builder_.CreateICmpEQ(metadata, candidate, "classis");
784 result = builder_.CreateOr(result, isCandidate, "classis.any");
785 }
786 zirValueMap_[classIs.getResult().get()] = result;
787 return;
788 }
789 case OpCode::Br: {
790 const auto &branchInst = static_cast<const BranchInst &>(inst);
791 builder_.CreateBr(zirBlockMap_.at(branchInst.getTarget()));
792 return;
793 }
794 case OpCode::CondBr: {
795 const auto &branchInst = static_cast<const CondBranchInst &>(inst);
796 builder_.CreateCondBr(lowerZIRRValue(branchInst.getCondition()),
797 zirBlockMap_.at(branchInst.getTrueLabel()),
798 zirBlockMap_.at(branchInst.getFalseLabel()));
799 return;
800 }
801 case OpCode::Ret: {
802 const auto &returnInst = static_cast<const ReturnInst &>(inst);
803 if (returnInst.getValue()) {
804 auto *retValue = lowerZIRRValue(returnInst.getValue());
805 auto retType = returnInst.getValue()->getType();
806 if (returnInst.getValue()->getOwnership() ==
807 zir::ValueOwnership::Borrowed) {
808 if (isClassType(retType)) {
809 emitRetainIfNeeded(retValue, retType);
810 } else if (isOwnedStringType(retType)) {
811 retValue = emitStringRetainIfNeeded(retValue, retType);
812 } else if (containsManagedValues(retType)) {
813 emitManagedRetain(retValue, retType);
814 }
815 }
816 emitZIRFunctionReleases();
817 if (!currentZIRFunction_->isDestructor) {
818 emitArcCollectionSafePoint();
819 }
820 builder_.CreateRet(retValue);
821 } else {
822 emitZIRFunctionReleases();
823 if (!currentZIRFunction_->isDestructor) {
824 emitArcCollectionSafePoint();
825 }
826 builder_.CreateRetVoid();
827 }
828 return;
829 }
830 case OpCode::Call: {
831 const auto &callInst = static_cast<const CallInst &>(inst);
832 std::vector<llvm::Value *> args;
833
834 if (callInst.isIndirect()) {
835 auto *calleePtr = lowerZIRRValue(callInst.getCalleeValue());
836 const auto &fpType = static_cast<const zir::FunctionPointerType &>(
837 *callInst.getCalleeValue()->getType());
838 std::vector<llvm::Type *> paramTypes;
839 for (const auto &p : fpType.getParams())
840 paramTypes.push_back(toLLVMType(*p));
841 auto *returnType = fpType.returnsRef()
842 ? llvm::PointerType::getUnqual(ctx_)
843 : toLLVMType(*fpType.getReturnType());
844 auto *fnTy = llvm::FunctionType::get(returnType, paramTypes, false);
845 for (size_t i = 0; i < callInst.getArguments().size(); ++i) {
846 const auto &arg = callInst.getArguments()[i];
847 auto *lowered = lowerZIRRValue(arg);
848 args.push_back(lowered);
849 }
850 auto *call = builder_.CreateCall(fnTy, calleePtr, args);
851 if (callInst.getResult()) {
852 zirValueMap_[callInst.getResult().get()] = call;
853 if (fpType.returnsRef()) {
854 refReturnValues_.insert(callInst.getResult().get());
855 }
856 }
857 return;
858 }
859
860 auto calleeIt = functionMap_.find(callInst.getFunctionName());
861 if (calleeIt == functionMap_.end()) {
862 auto zirDeclIt = zirFunctionMap_.find(callInst.getFunctionName());
863 if (zirDeclIt != zirFunctionMap_.end()) {
864 declareZIRFunction(*zirDeclIt->second, true);
865 calleeIt = functionMap_.find(callInst.getFunctionName());
866 }
867 }
868 if (calleeIt == functionMap_.end()) {
869 auto *existing = module_->getFunction(callInst.getFunctionName());
870 if (!existing) {
871 std::vector<llvm::Type *> inferredParamTypes;
872 inferredParamTypes.reserve(callInst.getArguments().size());
873 for (size_t i = 0; i < callInst.getArguments().size(); ++i) {
874 bool isRef = i < callInst.getArgumentIsRef().size() &&
875 callInst.getArgumentIsRef()[i];
876 auto *argTy =
877 isRef ? lowerZIRValue(callInst.getArguments()[i])->getType()
878 : lowerZIRRValue(callInst.getArguments()[i])->getType();
879 inferredParamTypes.push_back(argTy);
880 }
881
882 llvm::Type *retTy = llvm::Type::getVoidTy(ctx_);
883 if (callInst.getResult()) {
884 retTy = toLLVMType(*callInst.getResult()->getType());
885 }
886
887 auto *fnTy = llvm::FunctionType::get(retTy, inferredParamTypes, false);
888 existing = llvm::Function::Create(fnTy, llvm::Function::ExternalLinkage,
889 callInst.getFunctionName(), *module_);
890 }
891
892 functionMap_[callInst.getFunctionName()] = existing;
893 calleeIt = functionMap_.find(callInst.getFunctionName());
894 }
895 if (calleeIt == functionMap_.end()) {
896 throw std::runtime_error("missing callee in LLVM function map: " +
897 callInst.getFunctionName());
898 }
899 auto *callee = calleeIt->second;
900 auto *calleeTy = callee->getFunctionType();
901 auto zirIt = zirFunctionMap_.find(callInst.getFunctionName());
902 size_t fixedParamCount = callInst.getArguments().size();
903 bool hasVariadicParameter = false;
904 std::shared_ptr<zir::Type> variadicElementType = nullptr;
905 bool isCVariadic = calleeTy->isVarArg();
906 if (zirIt != zirFunctionMap_.end()) {
907 fixedParamCount = 0;
908 for (const auto &param : zirIt->second->getArguments()) {
909 if (param->isVariadicPack()) {
910 hasVariadicParameter = true;
911 variadicElementType = param->getVariadicElementType();
912 break;
913 }
914 ++fixedParamCount;
915 }
916 isCVariadic = zirIt->second->isCVariadic;
917 }
918 for (size_t i = 0; i < callInst.getArguments().size(); ++i) {
919 bool isRef = i < callInst.getArgumentIsRef().size() &&
920 callInst.getArgumentIsRef()[i];
921 auto *arg = isRef ? lowerZIRValue(callInst.getArguments()[i])
922 : lowerZIRRValue(callInst.getArguments()[i]);
923 std::shared_ptr<zir::Type> calleeParamType = nullptr;
924 if (zirIt != zirFunctionMap_.end() && i < fixedParamCount &&
925 i < zirIt->second->getArguments().size()) {
926 calleeParamType = zirIt->second->getArguments()[i]->getType();
927 }
928 llvm::Type *paramTy = nullptr;
929 if (i < fixedParamCount &&
930 static_cast<unsigned>(i) < calleeTy->getNumParams()) {
931 paramTy = calleeTy->getParamType(static_cast<unsigned>(i));
932 }
933 if (paramTy && arg->getType() != paramTy) {
934 auto argumentType = callInst.getArguments()[i]
935 ? callInst.getArguments()[i]->getType()
936 : nullptr;
937 if (isStringType(argumentType) && isStringType(calleeParamType)) {
938 auto *ptr = builder_.CreateExtractValue(arg, {0}, "zir.call.str.ptr");
939 auto *len = builder_.CreateExtractValue(arg, {1}, "zir.call.str.len");
940 llvm::Value *converted = llvm::UndefValue::get(paramTy);
941 converted = builder_.CreateInsertValue(converted, ptr, {0},
942 "zir.call.str.ptr.i");
943 converted = builder_.CreateInsertValue(converted, len, {1},
944 "zir.call.str.len.i");
945 arg = converted;
946 }
947 auto *argPtrTy = llvm::dyn_cast<llvm::PointerType>(arg->getType());
948 if (argPtrTy && !paramTy->isPointerTy()) {
949 arg = builder_.CreateLoad(paramTy, arg, "zir.call.arg");
950 }
951 } else if (isCVariadic && i >= fixedParamCount) {
952 // C varargs default argument promotions:
953 // - integer types narrower than int -> int
954 // - float -> double
955 // (already-typed varargs keep their value category)
956 auto *argTy = arg->getType();
957 if (argTy->isIntegerTy()) {
958 unsigned bits = argTy->getIntegerBitWidth();
959 if (bits < 32) {
960 auto argType = callInst.getArguments()[i]
961 ? callInst.getArguments()[i]->getType()
962 : nullptr;
963 bool isUnsignedArg =
964 argType && (argType->isUnsigned() ||
965 argType->getKind() == zir::TypeKind::Bool ||
966 argType->getKind() == zir::TypeKind::Char);
967 auto *i32Ty = llvm::Type::getInt32Ty(ctx_);
968 arg = isUnsignedArg
969 ? builder_.CreateZExt(arg, i32Ty, "cvararg.zext")
970 : builder_.CreateSExt(arg, i32Ty, "cvararg.sext");
971 }
972 } else if (argTy->isFloatTy()) {
973 auto *f64Ty = llvm::Type::getDoubleTy(ctx_);
974 arg = builder_.CreateFPExt(arg, f64Ty, "cvararg.fpext");
975 }
976 }
977 args.push_back(arg);
978 }
979 if (hasVariadicParameter) {
980 auto *elemTy = toLLVMType(*variadicElementType);
981 auto *elemPtrTy = llvm::PointerType::getUnqual(ctx_);
982 size_t explicitVariadicCount =
983 callInst.getArguments().size() - fixedParamCount;
984
985 llvm::Value *forwardedCount =
986 llvm::ConstantInt::get(llvm::Type::getInt32Ty(ctx_), 0);
987 llvm::Value *forwardedData = llvm::ConstantPointerNull::get(elemPtrTy);
988 if (callInst.getVariadicPack()) {
989 auto *packValue = lowerZIRRValue(callInst.getVariadicPack());
990 forwardedData =
991 builder_.CreateExtractValue(packValue, {0}, "varargs.forward.data");
992 forwardedCount =
993 builder_.CreateExtractValue(packValue, {1}, "varargs.forward.len");
994 if (forwardedCount->getType() != llvm::Type::getInt32Ty(ctx_)) {
995 forwardedCount = builder_.CreateIntCast(
996 forwardedCount, llvm::Type::getInt32Ty(ctx_), /*isSigned=*/true);
997 }
998 }
999
1000 llvm::Value *explicitCount =
1001 llvm::ConstantInt::get(llvm::Type::getInt32Ty(ctx_),
1002 static_cast<uint64_t>(explicitVariadicCount));
1003
1004 if (explicitVariadicCount == 0 && !callInst.getVariadicPack()) {
1005 args.push_back(explicitCount);
1006 args.push_back(llvm::ConstantPointerNull::get(elemPtrTy));
1007 } else if (explicitVariadicCount == 0) {
1008 args.push_back(forwardedCount);
1009 args.push_back(forwardedData);
1010 } else {
1011 llvm::Value *totalCount = explicitCount;
1012 if (callInst.getVariadicPack()) {
1013 totalCount = builder_.CreateAdd(explicitCount, forwardedCount,
1014 "varargs.total");
1015 }
1016
1017 auto *buffer = builder_.CreateAlloca(elemTy, totalCount, "varargs.buf");
1018 for (size_t i = 0; i < explicitVariadicCount; ++i) {
1019 auto *dst = builder_.CreateInBoundsGEP(
1020 elemTy, buffer,
1021 llvm::ConstantInt::get(llvm::Type::getInt32Ty(ctx_),
1022 static_cast<uint64_t>(i)));
1023 builder_.CreateStore(args[fixedParamCount + i], dst);
1024 }
1025
1026 if (callInst.getVariadicPack()) {
1027 auto *copyCondBB =
1028 llvm::BasicBlock::Create(ctx_, "varargs.copy.cond", currentFn_);
1029 auto *copyBodyBB =
1030 llvm::BasicBlock::Create(ctx_, "varargs.copy.body", currentFn_);
1031 auto *copyDoneBB =
1032 llvm::BasicBlock::Create(ctx_, "varargs.copy.done", currentFn_);
1033 auto *copyPreheaderBB = builder_.GetInsertBlock();
1034
1035 builder_.CreateBr(copyCondBB);
1036 builder_.SetInsertPoint(copyCondBB);
1037 auto *indexPhi =
1038 builder_.CreatePHI(llvm::Type::getInt32Ty(ctx_), 2, "varargs.i");
1039 indexPhi->addIncoming(
1040 llvm::ConstantInt::get(llvm::Type::getInt32Ty(ctx_), 0),
1041 copyPreheaderBB);
1042 auto *cond =
1043 builder_.CreateICmpULT(indexPhi, forwardedCount, "varargs.cond");
1044 builder_.CreateCondBr(cond, copyBodyBB, copyDoneBB);
1045
1046 builder_.SetInsertPoint(copyBodyBB);
1047 auto *src = builder_.CreateInBoundsGEP(elemTy, forwardedData,
1048 indexPhi, "varargs.src");
1049 auto *srcVal = builder_.CreateLoad(elemTy, src, "varargs.load");
1050 auto *dstIndex = builder_.CreateAdd(indexPhi, explicitCount);
1051 auto *dst = builder_.CreateInBoundsGEP(elemTy, buffer, dstIndex,
1052 "varargs.dst");
1053 builder_.CreateStore(srcVal, dst);
1054 auto *next = builder_.CreateAdd(
1055 indexPhi,
1056 llvm::ConstantInt::get(llvm::Type::getInt32Ty(ctx_), 1));
1057 builder_.CreateBr(copyCondBB);
1058 indexPhi->addIncoming(next, copyBodyBB);
1059
1060 builder_.SetInsertPoint(copyDoneBB);
1061 }
1062
1063 args.erase(args.begin() + static_cast<std::ptrdiff_t>(fixedParamCount),
1064 args.end());
1065 args.push_back(totalCount);
1066 args.push_back(buffer);
1067 }
1068 } else if (isCVariadic) {
1069 // Extra arguments are passed through unchanged after the fixed params.
1070 }
1071 llvm::Value *call = nullptr;
1072 if (zirIt != zirFunctionMap_.end() && zirIt->second->vtableSlot >= 0 &&
1073 !args.empty()) {
1074 auto receiverType = callInst.getArguments().front()->getType();
1075 std::shared_ptr<zir::ClassType> classType = nullptr;
1076 if (receiverType->getKind() == zir::TypeKind::Class) {
1077 classType = std::static_pointer_cast<zir::ClassType>(receiverType);
1078 } else if (receiverType->getKind() == zir::TypeKind::Pointer) {
1079 auto baseType = std::static_pointer_cast<zir::PointerType>(receiverType)
1080 ->getBaseType();
1081 if (baseType->getKind() == zir::TypeKind::Class) {
1082 classType = std::static_pointer_cast<zir::ClassType>(baseType);
1083 }
1084 }
1085
1086 if (classType) {
1087 auto *objectTy = structCache_.at(classType->getCodegenName() + ".obj");
1088 auto *selfPtr = builder_.CreateBitCast(
1089 args[0], llvm::PointerType::getUnqual(ctx_), "zir.method.self");
1090 auto *vtableAddr = builder_.CreateStructGEP(
1091 objectTy, selfPtr, kClassVTableIndex, "zir.method.vtable.addr");
1092 auto *i8PtrTy = llvm::PointerType::getUnqual(ctx_);
1093 auto *vtablePtrTy = llvm::PointerType::getUnqual(ctx_);
1094 auto *vtablePtr =
1095 builder_.CreateLoad(vtablePtrTy, vtableAddr, "zir.method.vtable");
1096 auto *slotAddr = builder_.CreateInBoundsGEP(
1097 i8PtrTy, vtablePtr,
1098 llvm::ConstantInt::get(
1099 llvm::Type::getInt32Ty(ctx_),
1100 static_cast<uint64_t>(zirIt->second->vtableSlot)));
1101 auto *fnRaw =
1102 builder_.CreateLoad(i8PtrTy, slotAddr, "zir.method.fn.raw");
1103 auto *fnPtr = builder_.CreateBitCast(
1104 fnRaw, llvm::PointerType::getUnqual(ctx_), "zir.method.fn");
1105 call = builder_.CreateCall(calleeTy, fnPtr, args);
1106 }
1107 }
1108 if (!call) {
1109 call = builder_.CreateCall(callee, args);
1110 }
1111 if (callInst.getResult()) {
1112 zirValueMap_[callInst.getResult().get()] = call;
1113 if (zirIt != zirFunctionMap_.end() && zirIt->second->returnsRef) {
1114 refReturnValues_.insert(callInst.getResult().get());
1115 }
1116 }
1117 return;
1118 }
1119 case OpCode::GetElementPtr: {
1120 const auto &gepInst = static_cast<const GetElementPtrInst &>(inst);
1121 auto *ptr = lowerZIRValue(gepInst.getPointer());
1122 auto operandType = gepInst.getPointer()->getType();
1123 auto pointerType = std::dynamic_pointer_cast<zir::PointerType>(operandType);
1124 auto baseType = pointerType ? pointerType->getBaseType() : operandType;
1125 llvm::Value *gep = nullptr;
1126
1127 if (baseType->getKind() == zir::TypeKind::Record) {
1128 auto recordType = std::static_pointer_cast<zir::RecordType>(baseType);
1129 auto *structTy = llvm::cast<llvm::StructType>(toLLVMType(*recordType));
1130 llvm::Value *structPtr = ptr;
1131 if (!pointerType) {
1132 auto *tmp = createEntryAlloca(
1133 currentFn_,
1134 static_cast<const Register &>(*gepInst.getResult()).getRawName() +
1135 ".addr",
1136 structTy);

1151 currentFn_,
1152 static_cast<const Register &>(*gepInst.getResult()).getRawName() +
1153 ".addr",
1154 structTy);

1157 }

1159 structTy, structPtr, static_cast<unsigned>(gepInst.getIndex()),
1160 static_cast<const Register &>(*gepInst.getResult()).getRawName());

1199 static_cast<const Register &>(*gepInst.getResult()).getRawName());
1200 }
1201 }
1202 zirValueMap_[gepInst.getResult().get()] = gep;
1203 return;
1204 }
1205 case OpCode::Phi: {

1207 auto *phi = builder_.CreatePHI(
1208 toLLVMType(*phiInst.getResult()->getType()),
1209 phiInst.getIncoming().size(),
1210 static_cast<const Register &>(*phiInst.getResult()).getRawName());
1211 pendingPhiIncoming_.push_back({phi, phiInst.getIncoming()});

1214 }
1215 case OpCode::Cast: {

1222 }
1223 case OpCode::Copy: {

1232 }
1233 case OpCode::Move: {

1236 lowerZIRRValue(moveInst.getSource());

1238 }
1239 case OpCode::Borrow: {

1242 auto *result = lowerZIRCast(source, borrowInst.getOwner()->getType(),
1243 borrowInst.getResult()->getType());
1244 zirValueMap_[borrowInst.getResult().get()] = result;
1245 return;
1246 }
1247 case OpCode::WeakLock: {
1248 const auto &weakLockInst = static_cast<const WeakLockInst &>(inst);

1250 weakLockInst.getWeakValue()->getType());
1251 zirValueMap_[weakLockInst.getResult().get()] = result;
1252 return;
1253 }
1254 case OpCode::WeakAlive: {
1255 const auto &weakAliveInst = static_cast<const WeakAliveInst &>(inst);

1257 weakAliveInst.getWeakValue()->getType());
1258 zirValueMap_[weakAliveInst.getResult().get()] = result;
1259 return;
1260 }
1261 case OpCode::Alloc: {
1262 const auto &allocInst = static_cast<const AllocInst &>(inst);
1263 auto allocType = allocInst.getAllocatedType();
1264 if (allocType->getKind() != zir::TypeKind::Class) {
1265 throw std::runtime_error("ZIR alloc currently supports only class types");
1266 }
1267
1268 auto classType = std::static_pointer_cast<zir::ClassType>(allocType);
1269 auto *ptrTy = llvm::cast<llvm::PointerType>(toLLVMType(*classType));
1270 auto *objectTy = structCache_.at(classType->getCodegenName() + ".obj");
1271 auto *sizeOfObj = llvm::ConstantExpr::getSizeOf(objectTy);
1272 auto *sizeTy = nativeIntegerType();
1273 llvm::Value *sizeValue = sizeOfObj;
1274 if (sizeValue->getType() != sizeTy) {
1275 sizeValue = builder_.CreateIntCast(sizeValue, sizeTy, /*isSigned=*/false);
1276 }
1277
1278 auto allocationIt = functionMap_.find("zap_runtime_alloc");
1279 if (allocationIt == functionMap_.end()) {
1280 auto *allocationTy = llvm::FunctionType::get(
1281 llvm::PointerType::getUnqual(ctx_), {sizeTy}, false);
1282 auto *allocationFn =
1283 llvm::Function::Create(allocationTy, llvm::Function::ExternalLinkage,
1284 "zap_runtime_alloc", *module_);
1285 allocationIt =
1286 functionMap_.emplace("zap_runtime_alloc", allocationFn).first;
1287 }

1292
1293 auto *refCountAddr = builder_.CreateStructGEP(

1295 builder_.CreateStore(

1301 auto *aliveAddr = builder_.CreateStructGEP(objectTy, typedPtr,
1302 kClassAliveIndex, "alive.addr");

1304 aliveAddr);

1306 objectTy, typedPtr, kClassGcMarkIndex, "gcmark.addr");

1308 gcMarkAddr);

1310 objectTy, typedPtr, kClassReleaseFnIndex, "release.fn.addr");
1311 auto *releaseFnPtr =
1312 builder_.CreateBitCast(classReleaseFns_.at(classType->getCodegenName()),
1313 llvm::PointerType::getUnqual(ctx_));
1314 builder_.CreateStore(releaseFnPtr, releaseFnAddr);

1316 objectTy, typedPtr, kClassDestroyFnIndex, "destroy.fn.addr");
