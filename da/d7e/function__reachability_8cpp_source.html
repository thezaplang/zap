185void collectClassTypes(const sema::BoundRootNode &root, ClassTypes &classes) {

201void collectClassesInType(const std::shared_ptr<zir::Type> &type,
202 FunctionReachability &result,
203 std::unordered_set<const zir::Type *> &visited) {
204 if (!type || !visited.insert(type.get()).second)
205 return;
206
207 switch (type->getKind()) {
208 case zir::TypeKind::Class: {
209 auto classType = std::static_pointer_cast<zir::ClassType>(type);
210 result.liveClassCodegenNames.insert(classType->getCodegenName());
211 for (const auto &field : classType->getFields())
212 collectClassesInType(field.type, result, visited);
213 return;
214 }
215 case zir::TypeKind::Record: {
216 auto recordType = std::static_pointer_cast<zir::RecordType>(type);
217 for (const auto &field : recordType->getFields())
218 collectClassesInType(field.type, result, visited);
219 return;
220 }
221 case zir::TypeKind::Array:
222 collectClassesInType(
223 std::static_pointer_cast<zir::ArrayType>(type)->getBaseType(), result,
224 visited);
225 return;
226 case zir::TypeKind::Pointer:
227 collectClassesInType(
228 std::static_pointer_cast<zir::PointerType>(type)->getBaseType(), result,
229 visited);
230 return;
231 default:
232 return;
233 }

267FunctionReachabilityAnalyzer::analyze(sema::BoundRootNode &root) {
268 FunctionReachability result;
269 std::unordered_map<const sema::FunctionSymbol *,
270 sema::BoundFunctionDeclaration *>
271 declarations;
272 for (const auto &function : root.functions) {
273 if (function && function->symbol)
274 declarations.emplace(function->symbol.get(), function.get());
275 }
276
277 std::deque<const sema::FunctionSymbol *> pending;
278 ReachabilityVisitor visitor(result);
279 visitor.setPending(&pending);
280
281 ClassTypes classes;
282 collectClassTypes(root, classes);
283
284 std::unordered_set<const zir::Type *> globalTypes;
285 for (const auto &global : root.globals) {
286 visitor.visit(*global);
287 if (global && global->symbol)
288 collectClassesInType(global->symbol->type, result, globalTypes);
289 }
290
291 for (const auto &function : root.functions) {
292 const auto &symbol = function->symbol;
293 if (symbol && (symbol->isEntryModule || symbol->name == "main" ||
294 symbol->hasEntry || symbol->hasNoMangle)) {
295 if (result.functions.insert(symbol.get()).second)
296 pending.push_back(symbol.get());
297 }
298 }
299
300 bool changed = true;
301 while (changed) {
302 changed = false;
303 while (!pending.empty()) {
304 const auto *symbol = pending.front();
305 pending.pop_front();
306 const auto declaration = declarations.find(symbol);
307 if (declaration != declarations.end())
308 visitor.visit(*declaration->second);
309 }
310
311 changed = addLiveBaseClasses(classes, result) || changed;
312 for (const auto &function : root.functions) {
313 const auto &symbol = function->symbol;
314 if (!symbol || symbol->ownerTypeCodegenName.empty() ||
315 result.liveClassCodegenNames.count(symbol->ownerTypeCodegenName) == 0)
316 continue;
317
318 bool mustKeep = symbol->isConstructor || symbol->isDestructor;
319 if (symbol->vtableSlot >= 0) {
320 for (const auto &[dispatchClass, slots] : result.liveVtableSlots) {
321 const auto classIt = classes.find(symbol->ownerTypeCodegenName);
322 if (classIt != classes.end() &&
323 isSubclassOf(classIt->second, dispatchClass) &&
324 slots.count(symbol->vtableSlot) != 0) {
325 mustKeep = true;
326 break;
327 }
328 }
329 }
330 if (mustKeep && result.functions.insert(symbol.get()).second) {
331 pending.push_back(symbol.get());
332 changed = true;
333 }
334 }
335 }
336
337 return result;
338}
