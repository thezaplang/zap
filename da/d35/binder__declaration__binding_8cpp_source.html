30void Binder::visit(ClassDecl &node) {
31 auto symbol =
32 std::dynamic_pointer_cast<TypeSymbol>(currentScope_->lookup(node.name_));
33 if (!symbol || !symbol->isClass || !symbol->genericParameterNames.empty()) {
34 return;
35 }
36
37 auto classType = std::static_pointer_cast<zir::ClassType>(symbol->type);
38 auto &classInfo = classInfos_[classType->getCodegenName()];
39 currentClassStack_.push_back(classType->getName());
40
41 if (node.baseType_) {
42 bool hasOwnCtor = false;
43 bool hasOwnDtor = false;
44 for (const auto &method : node.methods_) {
45 hasOwnCtor = hasOwnCtor || method->name_ == "init";
46 hasOwnDtor = hasOwnDtor || method->name_ == "deinit";
47 }
48 auto baseType = mapType(*node.baseType_);
49 if (!baseType || baseType->getKind() != zir::TypeKind::Class) {
50 error(node.baseType_->span, "Base type must be a class.");
51 } else {
52 auto baseClass = std::static_pointer_cast<zir::ClassType>(baseType);
53 classType->setBase(baseClass);
54 auto baseIt = classInfos_.find(baseClass->getCodegenName());
55 if (baseIt != classInfos_.end()) {
56 if (!hasOwnCtor) {
57 classInfo.constructor = baseIt->second.constructor;
58 }
59 if (!hasOwnDtor) {
60 classInfo.destructor = baseIt->second.destructor;
61 }
62 classInfo.fields = baseIt->second.fields;
63 classInfo.methods.insert(baseIt->second.methods.begin(),
64 baseIt->second.methods.end());
65 for (const auto &field : baseClass->getFields()) {
66 classType->addField(field.name, field.type, field.visibility);
67 }
68 }
69 }
70 }
71
72 for (const auto &field : node.fields_) {
73 auto fieldType = mapType(*field->type);
74 if (!fieldType) {
75 error(field->span, "Unknown type: " + field->type->qualifiedName());
76 fieldType = std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
77 }
78 classType->addField(field->name, fieldType,
79 static_cast<int>(field->visibility_));
80 classInfo.fields[field->name] = std::make_shared<VariableSymbol>(
81 field->name, fieldType, BindingKind::Mutable, false, field->name,
82 modules_[currentModuleId_].info->moduleName, field->visibility_);
83 }
84
85 auto boundRecord = std::make_unique<BoundRecordDeclaration>();
86 boundRecord->type = classType;
87 boundRoot_->records.push_back(std::move(boundRecord));
88
89 for (const auto &method : node.methods_) {
90 method->accept(*this);
91 }
92
93 currentClassStack_.pop_back();
94}

96void Binder::visit(FunDecl &node) {
97 auto symbolIt = declaredFunctionSymbols_.find(&node);
98 auto symbol =
99 symbolIt == declaredFunctionSymbols_.end() ? nullptr : symbolIt->second;
100 if (!symbol) {
101 return;
102 }
103
104 if (!symbol->genericParameterNames.empty() &&
105 !symbol->isGenericInstantiation) {
106 return;
107 }
108
109 pushScope();
110 auto oldFunction = currentFunction_;
111 currentFunction_ = symbol;
112 int oldUnsafeDepth = unsafeDepth_;
113 if (node.isUnsafe_) {
114 ++unsafeDepth_;
115 }
116
117 for (size_t i = 0; i < symbol->parameters.size(); ++i) {
118 const auto &param = symbol->parameters[i];
119 if (!currentScope_->declare(param->name, param)) {
120 error(node.span, "Parameter '" + param->name + "' already declared.");
121 }
122 if (semanticInfo_ && i < node.params_.size() && node.params_[i]) {
123 semanticInfo_->recordSymbol(node.params_[i].get(), param);
124 semanticInfo_->recordDeclaration(node.params_[i].get(), param);
125 semanticInfo_->recordType(node.params_[i].get(), param->type);
126 }
127 }
128
129 auto boundBody = bindBody(node.body_.get(), false);
130
131 popScope();
132 currentFunction_ = oldFunction;
133 unsafeDepth_ = oldUnsafeDepth;
134
135 if (!symbol->returnType) {
136 std::vector<BoundReturnStatement *> returns;
137 std::function<void(BoundBlock *)> collectReturns =
138 [&](BoundBlock *block) {
139 if (!block)
140 return;
141 for (auto &stmt : block->statements) {
142 if (auto *ret = dynamic_cast<BoundReturnStatement *>(stmt.get())) {
143 returns.push_back(ret);
144 } else if (auto *ifStmt =
145 dynamic_cast<BoundIfStatement *>(stmt.get())) {
146 collectReturns(ifStmt->thenBody.get());
147 collectReturns(ifStmt->elseBody.get());
148 } else if (auto *whileStmt =
149 dynamic_cast<BoundWhileStatement *>(stmt.get())) {
150 collectReturns(whileStmt->body.get());
151 } else if (auto *forStmt =
152 dynamic_cast<BoundForStatement *>(stmt.get())) {
153 collectReturns(forStmt->body.get());
154 } else if (auto *nested =
155 dynamic_cast<BoundBlock *>(stmt.get())) {
156 collectReturns(nested);
157 }
158 }
159 };
160 collectReturns(boundBody.get());
161
162 if (returns.empty()) {
163 symbol->returnType =
164 std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
165 } else {
166 auto returnType = [](const BoundReturnStatement *ret) {
167 return ret->expression
168 ? ret->expression->type
169 : std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
170 };
171 auto inferred = returnType(returns[0]);
172 bool conflict = false;
173 for (size_t i = 1; i < returns.size(); ++i) {
174 auto candidate = returnType(returns[i]);
175 auto join = conversions_.joinTypes(inferred, candidate);
176 if (!join) {
177 error(node.span, "Cannot infer return type of function '" +
178 node.name_ + "': conflicting return types '" +
179 renderTypeForUser(inferred) + "' and '" +
180 renderTypeForUser(candidate) +
181 "'. Add an explicit return type annotation.");
182 conflict = true;
183 break;
184 }
185 inferred = join->type;
186 }
187 if (!conflict) {
188 for (auto *ret : returns) {
189 if (!ret->expression) {
190 continue;
191 }
192 auto conversion =
193 conversions_.classifyImplicit(ret->expression->type, inferred);
194 if (!conversion) {
195 error(node.span,
196 "Internal error: inferred return type cannot represent all "
197 "return expressions.");
198 conflict = true;
199 break;
200 }
201 ret->expression =
202 applyConversion(std::move(ret->expression), *conversion);
203 }
204 }
205 symbol->returnType =
206 conflict ? std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void)
207 : inferred;
208 }
209 }
210
211 bool hasReturn = blockAlwaysReturns(boundBody.get());
212
213 if (!hasReturn && symbol->linkName == "main" &&
214 symbol->returnType->isInteger()) {
215 auto intType = std::make_shared<zir::PrimitiveType>(zir::TypeKind::Int);
216 auto lit = std::make_unique<BoundLiteral>("0", intType);
217 boundBody->statements.push_back(
218 std::make_unique<BoundReturnStatement>(std::move(lit)));
219 hasReturn = true;
220 }
221
222 if (!hasReturn && symbol->returnType->getKind() != zir::TypeKind::Void) {
223 auto kind = symbol->returnType->getKind();
224 if (symbol->returnType->isInteger() ||
225 symbol->returnType->isFloatingPoint() ||
226 kind == zir::TypeKind::Bool) {
227 std::string litVal = "0";
228 if (symbol->returnType->isFloatingPoint())
229 litVal = "0.0";
230 else if (kind == zir::TypeKind::Bool)
231 litVal = "false";
232 auto lit = std::make_unique<BoundLiteral>(litVal, symbol->returnType);
233 boundBody->statements.push_back(
234 std::make_unique<BoundReturnStatement>(std::move(lit)));
235 }
236
237 _diag.report(node.span, zap::DiagnosticLevel::Warning,
238 "Function '" + node.name_ +
239 "' has non-void return type but no return on some paths.");
240 }
241
242 boundRoot_->functions.push_back(
243 std::make_unique<BoundFunctionDeclaration>(symbol, std::move(boundBody)));
244}

264void Binder::visit(BindingDecl &node) {
265 auto existing = currentScope_->lookupLocal(node.name_);
266 std::shared_ptr<VariableSymbol> symbol;
267 if (existing) {
268 symbol = std::dynamic_pointer_cast<VariableSymbol>(existing);
269 }
270
271 const bool isConstant = node.kind_ == BindingKind::CompileTimeConstant;
272 const bool isImmutable = node.kind_ == BindingKind::Immutable;
273 const bool isRef = !isConstant && node.type_ && node.type_->isReference;
274 const std::string declarationKind =
275 isConstant ? "constant"
276 : (isImmutable ? "immutable binding" : "variable");
277
278 std::shared_ptr<zir::Type> type;
279 std::unique_ptr<BoundExpression> initializer = nullptr;
280
281 if (node.type_) {
282 type = mapType(*node.type_);
283 if (!type) {
284 error(node.span, "Unknown type: " + node.type_->qualifiedName());
285 type = std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
286 }
287
288 if (node.initializer_) {
289 initializer = bindExpressionWithExpected(node.initializer_.get(), type);
290 if (initializer && !isRef) {
291 auto conversion =
292 conversions_.classifyImplicit(initializer->type, type);
293 if (!conversion) {
294 error(node.initializer_->span,
295 "Cannot assign expression of type '" +
296 renderTypeForUser(initializer->type) + "' to " +
297 declarationKind + " of type '" + renderTypeForUser(type) +
298 "'");
299 } else {
300 initializer = applyConversion(std::move(initializer), *conversion);
301 }
302 }
303 } else if (isConstant || isImmutable) {
304 error(node.span,
305 std::string(isConstant ? "Constant '" : "Immutable binding '") +
306 node.name_ + "' must be initialized.");
307 } else if (isRef) {
308 error(node.span,
309 "Reference variable '" + node.name_ + "' must be initialized.");
310 }
311 } else {
312 if (!node.initializer_) {
313 if (isConstant || isImmutable) {
314 error(node.span,
315 std::string(isConstant ? "Constant '" : "Immutable binding '") +
316 node.name_ + "' must be initialized.");
317 } else {
318 error(node.span, "Variable '" + node.name_ +
319 "' needs a type annotation or an initializer.");
320 }
321 type = std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
322 } else {
323 initializer =
324 bindExpressionWithExpected(node.initializer_.get(), nullptr);
325 if (initializer && initializer->type &&
326 initializer->type->getKind() != zir::TypeKind::Void) {
327 type = initializer->type;
328 } else {
329 error(node.initializer_->span, "Cannot infer type of " +
330 declarationKind + " '" + node.name_ +
331 "' from a void expression.");
332 type = std::make_shared<zir::PrimitiveType>(zir::TypeKind::Void);
333 }
334 }
335 }
336
337 if (isRef && initializer &&
338 !requireMutablePlace(*initializer, node.span,
339 MutablePlaceUse::MutableReference)) {
340 initializer.reset();
341 }
342
343 if (!symbol) {
344 std::string linkName = node.name_;
345 if (isConstant || node.isGlobal_) {
346 linkName = node.isExternal_
347 ? node.name_
348 : mangleName(currentModuleLinkPath(), node.name_);
349 }
350 symbol = std::make_shared<VariableSymbol>(
351 node.name_, type, node.kind_, isRef, std::move(linkName),
352 modules_[currentModuleId_].info->moduleName, node.visibility_);
353 symbol->is_external = node.isExternal_;
354 if (!currentScope_->declare(node.name_, symbol)) {
355 if (isConstant) {
356 error(node.span, "Identifier '" + node.name_ + "' already declared.");
357 } else {
358 error(node.span, "Variable '" + node.name_ + "' already declared.");
359 }
360 }
361 } else if (!node.type_ && type) {
362 symbol->type = type;
363 }
364 symbol->is_ref = isRef;
365 symbol->is_external = node.isExternal_;
366 symbol->is_global = !currentBlock_;
367
368 if (isConstant && initializer) {
369 std::string failureReason;
370 if (!ConstantEvaluator::isConstant(*initializer, &failureReason)) {
371 error(node.initializer_->span,
372 "Constant '" + node.name_ +
373 "' must be initialized with a compile-time expression: " +
374 failureReason + ".");
375 } else {
376 symbol->constant_value =
377 std::shared_ptr<BoundExpression>(initializer->clone());
378 }
379 }
380 if (semanticInfo_) {
381 semanticInfo_->recordSymbol(&node, symbol);
382 semanticInfo_->recordDeclaration(&node, symbol);
383 semanticInfo_->recordType(&node, symbol->type);
384 }
385
386 auto boundDecl = std::make_unique<BoundVariableDeclaration>(
387 symbol, std::move(initializer));
388
389 if (currentBlock_ && !node.isGlobal_) {
390 statementStack_.push(std::move(boundDecl));
391 } else {
392 boundRoot_->globals.push_back(std::move(boundDecl));
393 }
